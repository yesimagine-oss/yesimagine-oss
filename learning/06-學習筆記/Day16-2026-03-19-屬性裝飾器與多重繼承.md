## 1️⃣ 屬性裝飾器深入 (@property)

### 1.1 基礎用法

```python
class Person:
    def __init__(self, name, age):
        self._name = name
        self._age = age
    
    @property
    def name(self):
        """getter - 訪問時調用"""
        print(f"獲取 name")
        return self._name
    
    @name.setter
    def name(self, value):
        """setter - 設置時調用"""
        print(f"設置 name: {value}")
        if not value:
            raise ValueError("名字不能為空")
        self._name = value
    
    @name.deleter
    def name(self):
        """deleter - 刪除時調用"""
        print(f"刪除 name")
        del self._name
    
    @property
    def age(self):
        """只讀屬性 - 沒有 setter"""
        return self._age
    
    @property
    def info(self):
        """計算屬性 - 動態計算"""
        return f"{self._name}, {self._age}歲"

# 使用
person = Person("小明", 18)

# 訪問屬性 (像訪問普通屬性一樣)
print(person.name)  # 獲取 name \n 小明
print(person.age)   # 18 (只讀)
print(person.info)  # 小明，18 歲 (計算屬性)

# 設置屬性
person.name = "小紅"  # 設置 name: 小紅

# 刪除屬性
del person.name  # 刪除 name

# 嘗試設置只讀屬性會報錯
# person.age = 20  # ❌ AttributeError: can't set attribute
```

### 1.2 實際應用：數據驗證

```python
class Product:
    def __init__(self, name, price, quantity):
        self.name = name
        self.price = price
        self.quantity = quantity
    
    @property
    def price(self):
        return self._price
    
    @price.setter
    def price(self, value):
        if value < 0:
            raise ValueError("價格不能為負數")
        if value > 10000:
            print(f"警告：價格 {value} 超過 10000")
        self._price = value
    
    @property
    def quantity(self):
        return self._quantity
    
    @quantity.setter
    def quantity(self, value):
        if value < 0:
            raise ValueError("數量不能為負數")
        if not isinstance(value, int):
            raise TypeError("數量必須是整數")
        self._quantity = value
    
    @property
    def total_value(self):
        """計算總價值 (只讀)"""
        return self._price * self._quantity
    
    @property
    def availability(self):
        """庫存狀態"""
        if self._quantity == 0:
            return "缺貨"
        elif self._quantity < 10:
            return "庫存緊張"
        else:
            return "貨源充足"

# 使用
product = Product("Python 書籍", 59.9, 100)
print(product.total_value)  # 5990.0
print(product.availability)  # 貨源充足

# 設置價格 (會進行驗證)
product.price = 79.9  # ✅
# product.price = -10  # ❌ ValueError

# 設置數量 (會進行驗證)
product.quantity = 5  # ✅
print(product.availability)  # 庫存緊張
# product.quantity = -1  # ❌ ValueError
# product.quantity = 3.5  # ❌ TypeError
```

### 1.3 使用 property 實現緩存

```python
import time

class DataProcessor:
    def __init__(self, data):
        self._data = data
        self._cached_result = None
        self._cache_valid = False
    
    @property
    def processed_data(self):
        """帶緩存的計算屬性"""
        if not self._cache_valid:
            print("正在處理數據...")
            time.sleep(1)  # 模擬耗時操作
            self._cached_result = [x * 2 for x in self._data]
            self._cache_valid = True
            print("處理完成！")
        else:
            print("使用緩存結果")
        return self._cached_result
    
    def invalidate_cache(self):
        """使緩存失效"""
        self._cache_valid = False
        print("緩存已失效")

# 使用
processor = DataProcessor([1, 2, 3, 4, 5])

print(processor.processed_data)  # 第一次：處理數據...
print(processor.processed_data)  # 第二次：使用緩存結果
print(processor.processed_data)  # 第三次：使用緩存結果

processor.invalidate_cache()  # 使緩存失效
print(processor.processed_data)  # 第四次：處理數據...
```

```python
class TypedProperty:
    """類型檢查描述器"""
    def __init__(self, name, expected_type):
        self.name = name
        self.expected_type = expected_type
    
    def __get__(self, instance, owner):
        if instance is None:
            return self
        return instance.__dict__.get(self.name)
    
    def __set__(self, instance, value):
        if not isinstance(value, self.expected_type):
            raise TypeError(f"{self.name} 必須是 {self.expected_type.__name__} 類型")
        instance.__dict__[self.name] = value
    
    def __delete__(self, instance):
        if self.name in instance.__dict__:
            del instance.__dict__[self.name]

class Person:
    name = TypedProperty("name", str)
    age = TypedProperty("age", int)
    height = TypedProperty("height", float)
    
    def __init__(self, name, age, height):
        self.name = name
        self.age = age
        self.height = height

# 使用
person = Person("小明", 18, 175.5)
print(person.name)    # 小明
print(person.age)     # 18
print(person.height)  # 175.5

# person.age = "二十"  # ❌ TypeError: age 必須是 int 類型
```

## 2️⃣ 多重繼承與 MRO

### 2.1 多重繼承基礎

```python
class A:
    def method(self):
        return "A"

class B:
    def method(self):
        return "B"

class C:
    def method(self):
        return "C"

# 多重繼承
class D(A, B, C):
    pass

d = D()
print(d.method())  # 輸出什麼？答案是 "A"

# 查看 MRO (方法解析順序)
print(D.__mro__)
# (<class '__main__.D'>, <class '__main__.A'>, 
#  <class '__main__.B'>, <class '__main__.C'>, 
#  <class 'object'>)

# 或者使用 mro() 方法
print(D.mro())
```

### 2.2 鑽石繼承問題

```python
class A:
    def method(self):
        print("A.method")
        return "A"

class B(A):
    def method(self):
        print("B.method")
        return super().method()

class C(A):
    def method(self):
        print("C.method")
        return super().method()

class D(B, C):
    def method(self):
        print("D.method")
        return super().method()

# 使用
d = D()
result = d.method()
print(f"最終結果：{result}")

# 查看 MRO
print("\nMRO:")
for cls in D.__mro__:
    print(f"  {cls.__name__}")

# 輸出:
# D.method
# B.method
# C.method
# A.method
# 最終結果：A
#
# MRO:
#   D
#   B
#   C
#   A
#   object
```

```python
class Animal:
    def __init__(self, name):
        print(f"Animal.__init__({name})")

    def speak(self):
        return "動物聲音"

    def __init__(self, name, warm_blooded=True):

        return "哺乳動物聲音"

    def __init__(self, name, can_fly=True):

        self.can_fly = can_fly
    
    def speak(self):
        return "鳥類聲音"

    def __init__(self, name):
        print(f"Bat.__init__({name})")

        # MRO: Bat -> Mammal -> Bird -> Animal
    
    def speak(self):
        return "蝙蝠叫聲"

# 使用
bat = Bat("小蝙蝠")
print(f"\n名字：{bat.name}")
print(f"溫血：{bat.warm_blooded}")
print(f"會飛：{bat.can_fly}")

    print(f"  {cls.__name__}")
```

### 2.4 混合類 (Mixin) 模式

```python
class JSONMixin:
    """提供 JSON 序列化功能"""
    def to_json(self):
        import json
        return json.dumps(self.__dict__, ensure_ascii=False, indent=2)

class LoggingMixin:
    """提供日誌功能"""
    def log(self, message):
        print(f"[LOG] {self.__class__.__name__}: {message}")

class ValidationMixin:
    """提供驗證功能"""
    def validate(self):
        errors = []
        for attr, value in self.__dict__.items():
            if value is None:
                errors.append(f"{attr} 不能為空")
        return errors

# 使用 Mixin
class Person(JSONMixin, LoggingMixin, ValidationMixin):
    def __init__(self, name, age, email):
        self.name = name
        self.age = age
        self.email = email

# 使用
person = Person("小明", 18, "xiaoming@example.com")

# 使用 Mixin 提供的功能
person.log("創建了一個新人")
print(person.to_json())
errors = person.validate()
if errors:
    print(f"驗證錯誤：{errors}")
else:
    print("驗證通過！")
```

### 3.2 上下文管理器進階

```python
from contextlib import contextmanager

class Timer:
    """計時器上下文管理器"""
    def __init__(self, name="操作"):
        self.name = name
        self.start = None
        self.end = None
    
    def __enter__(self):
        import time
        self.start = time.time()
        print(f"開始 {self.name}")
        return self
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        import time
        self.end = time.time()
        elapsed = self.end - self.start
        print(f"結束 {self.name}, 耗時：{elapsed:.4f}秒")
        
        # 如果有異常，可以選擇吞掉或繼續傳播
        if exc_type:
            print(f"發生異常：{exc_val}")
        return False  # False 讓異常繼續傳播

# 使用
with Timer("計算"):
    result = sum(range(1000000))
    print(f"結果：{result}")

# 使用 contextmanager 裝飾器實現
@contextmanager
def timer(name="操作"):
    import time
    start = time.time()
    print(f"開始 {name}")
    try:
        yield
    finally:
        end = time.time()
        elapsed = end - start
        print(f"結束 {name}, 耗時：{elapsed:.4f}秒")

# 使用
with timer("另一個計算"):
    result = sum(range(1000000))
```

### 3.3 自定義迭代器

```python
class Fibonacci:
    """斐波那契數列迭代器"""
    def __init__(self, max_value=None):
        self.max_value = max_value
    
    def __iter__(self):
        self.a, self.b = 0, 1
        return self
    
    def __next__(self):
        if self.max_value is not None and self.a > self.max_value:
            raise StopIteration
        result = self.a
        self.a, self.b = self.b, self.a + self.b
        return result

# 使用
print("斐波那契數列 (前 10 項):")
for i, num in enumerate(Fibonacci()):
    if i >= 10:
        break
    print(num, end=" ")
print()

print("\n斐波那契數列 (<= 100):")
for num in Fibonacci(100):
    print(num, end=" ")
print()
```

```python
from functools import total_ordering

@total_ordering  # 自動生成其他比較方法
class Version:
    def __init__(self, major, minor, patch):
        self.major = major
        self.minor = minor
        self.patch = patch
    
    def __str__(self):
        return f"{self.major}.{self.minor}.{self.patch}"
    
    def __repr__(self):
        return f"Version({self.major}, {self.minor}, {self.patch})"
    
    def __eq__(self, other):
        if not isinstance(other, Version):
            return NotImplemented
        return (self.major, self.minor, self.patch) == \
               (other.major, other.minor, other.patch)
    
    def __lt__(self, other):
        if not isinstance(other, Version):
            return NotImplemented
        return (self.major, self.minor, self.patch) < \
               (other.major, other.minor, other.patch)

# 使用
v1 = Version(1, 0, 0)
v2 = Version(1, 2, 0)
v3 = Version(2, 0, 0)

print(f"v1: {v1}")
print(f"v2: {v2}")
print(f"v3: {v3}")

print(f"\nv1 < v2: {v1 < v2}")  # True
print(f"v2 < v3: {v2 < v3}")  # True
print(f"v1 == v1: {v1 == v1}")  # True
print(f"v1 <= v2: {v1 <= v2}")  # True (自動生成)
print(f"v2 > v1: {v2 > v1}")  # True (自動生成)
print(f"v3 >= v2: {v3 >= v2}")  # True (自動生成)
```

## 4️⃣ 實戰：圖書管理系統進階版

```python
from datetime import datetime, timedelta
from typing import List, Optional, Dict

class Book:
    def __init__(self, title: str, author: str, isbn: str, total_copies: int = 1):
        self.title = title
        self.author = author
        self.isbn = isbn
        self.total_copies = total_copies
        self.available_copies = total_copies
    
    @property
    def availability_status(self) -> str:
        if self.available_copies == 0:
            return "缺貨"
        elif self.available_copies < self.total_copies // 2:
            return "庫存緊張"
        else:
            return "貨源充足"
    
    def __str__(self) -> str:
        return f"《{self.title}》- {self.author} [{self.available_copies}/{self.total_copies}]"
    
    def __repr__(self) -> str:
        return f"Book('{self.title}', '{self.author}', '{self.isbn}')"
    
    def __eq__(self, other) -> bool:
        if not isinstance(other, Book):
            return NotImplemented
        return self.isbn == other.isbn

class BorrowRecord:
    """借閱記錄"""
    def __init__(self, book: Book, borrower: str, days: int = 14):
        self.book = book
        self.borrower = borrower
        self.borrow_date = datetime.now()
        self.due_date = self.borrow_date + timedelta(days=days)
        self.return_date: Optional[datetime] = None
    
    @property
    def is_overdue(self) -> bool:
        if self.return_date:
            return False
        return datetime.now() > self.due_date
    
    @property
    def days_remaining(self) -> int:
        if self.return_date:
            return 0
        delta = self.due_date - datetime.now()
        return max(0, delta.days)
    
    def __str__(self) -> str:
        status = "已歸還" if self.return_date else ("逾期" if self.is_overdue else "借閱中")
        return f"{self.borrower} 借閱《{self.book.title}》- {status}"

class Library:
    def __init__(self, name: str):
        self.name = name
        self.books: Dict[str, Book] = {}  # ISBN -> Book
        self.records: List[BorrowRecord] = []
    
    def add_book(self, book: Book) -> str:
        if book.isbn in self.books:
            self.books[book.isbn].total_copies += book.total_copies
            self.books[book.isbn].available_copies += book.total_copies
            return f"已增加庫存：{book.title}"
        else:
            self.books[book.isbn] = book
            return f"已添加新書：{book.title}"
    
    def borrow_book(self, isbn: str, borrower: str, days: int = 14) -> str:
        if isbn not in self.books:
            return f"找不到 ISBN: {isbn}"
        
        book = self.books[isbn]
        if book.available_copies <= 0:
            return f"《{book.title}》已借完"
        
        book.available_copies -= 1
        record = BorrowRecord(book, borrower, days)
        self.records.append(record)
        
        return f"借閱成功：{borrower} 借走《{book.title}》，應歸還日期：{record.due_date.strftime('%Y-%m-%d')}"
    
    def return_book(self, isbn: str, borrower: str) -> str:
        if isbn not in self.books:
            return f"找不到 ISBN: {isbn}"
        
        # 查找借閱記錄
        for record in self.records:
            if record.book.isbn == isbn and record.borrower == borrower and not record.return_date:
                record.return_date = datetime.now()
                self.books[isbn].available_copies += 1
                
                if record.is_overdue:
                    return f"歸還成功，但已逾期 {record.days_remaining} 天"
                return f"歸還成功：{borrower} 歸還《{record.book.title}》"
        
        return f"未找到 {borrower} 的借閱記錄"
    
    def search_books(self, keyword: str) -> List[Book]:
        """搜索圖書"""
        results = []
        for book in self.books.values():
            if keyword.lower() in book.title.lower() or \
               keyword.lower() in book.author.lower():
                results.append(book)
        return results
    
    def list_overdue_books(self) -> List[BorrowRecord]:
        """列出所有逾期圖書"""
        return [record for record in self.records if record.is_overdue and not record.return_date]
    
    def __len__(self) -> int:
        return len(self.books)
    
    def __str__(self) -> str:
        return f"{self.name} - 館藏：{len(self)} 種圖書"

# 測試
library = Library("市立圖書館")

# 添加圖書
book1 = Book("Python 編程", "Guido", "978-0001", 3)
book2 = Book("數據結構", "Knuth", "978-0002", 2)
book3 = Book("算法導論", "Cormen", "978-0003", 1)

library.add_book(book1)
library.add_book(book2)
library.add_book(book3)

print(f"圖書館：{library}")
print()

# 借閱
print(library.borrow_book("978-0001", "小明"))
print(library.borrow_book("978-0001", "小紅"))
print(library.borrow_book("978-0002", "小剛"))
print()

# 查看圖書狀態
for book in library.books.values():
    print(f"{book} - {book.availability_status}")
print()

# 搜索
print("搜索 'Python':")
results = library.search_books("Python")
for book in results:
    print(f"  {book}")
print()

# 歸還
print(library.return_book("978-0001", "小明"))
print()

# 查看館藏
print(f"館藏總數：{len(library)}")
```

### 練習 1: 實現帶緩存的屬性

```python
class CachedProperty:
    def __init__(self, func):
        self.func = func
        self.name = func.__name__
    
    def __get__(self, instance, owner):
        if instance is None:
            return self
        
        # 檢查是否有緩存
        cache_name = f"_cached_{self.name}"
        if hasattr(instance, cache_name):
            return getattr(instance, cache_name)
        
        # 計算並緩存
        result = self.func(instance)
        setattr(instance, cache_name, result)
        return result

class DataAnalyzer:
    def __init__(self, data):
        self.data = data
    
    @CachedProperty
    def sum(self):
        print("計算總和...")
        return sum(self.data)
    
    @CachedProperty
    def average(self):
        print("計算平均值...")
        return sum(self.data) / len(self.data)

analyzer = DataAnalyzer([1, 2, 3, 4, 5])
print(analyzer.sum)      # 第一次計算
print(analyzer.sum)      # 使用緩存
print(analyzer.average)  # 第一次計算
```

### 練習 2: 實現一個簡單的 ORM

```python
class Field:
    def __init__(self, field_type):
        self.field_type = field_type
    
    def __get__(self, instance, owner):
        if instance is None:
            return self
        return instance.__dict__.get(self.name)
    
    def __set__(self, instance, value):
        if not isinstance(value, self.field_type):
            raise TypeError(f"必須是 {self.field_type.__name__} 類型")
        instance.__dict__[self.name] = value

class ModelMeta(type):
    def __new__(mcs, name, bases, attrs):
        # 自動設置字段名稱
        for key, value in attrs.items():
            if isinstance(value, Field):
                value.name = key
        return super().__new__(mcs, name, bases, attrs)

class Model(metaclass=ModelMeta):
    id = Field(int)
    name = Field(str)
    
    def __init__(self, **kwargs):
        for key, value in kwargs.items():
            setattr(self, key, value)
    
    def save(self):
        print(f"保存 {self.__class__.__name__}: {self.__dict__}")
    
    def __str__(self):
        return f"{self.__class__.__name__}({self.__dict__})"

class User(Model):
    email = Field(str)
    age = Field(int)

# 使用
```
