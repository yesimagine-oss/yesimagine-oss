### 1.2 上下文管理器協議

```python
class ManagedFile:
    """自定義文件上下文管理器"""
    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode
        self.file = None
    
    def __enter__(self):
        """進入上下文時調用"""
        print(f"打開文件：{self.filename}")
        self.file = open(self.filename, self.mode)
        return self.file
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        """退出上下文時調用"""
        print(f"關閉文件：{self.filename}")
        self.file.close()
        
        # 返回值決定是否吞掉異常
        # True = 吞掉異常，False/None = 讓異常繼續傳播
        if exc_type:
            print(f"發生異常：{exc_type.__name__}: {exc_val}")
        return False  # 讓異常繼續傳播

# 使用
with ManagedFile('test.txt', 'w') as f:
    f.write("Hello, World!")

# 處理異常
with ManagedFile('test.txt', 'r') as f:
    content = f.read()  # 如果文件不存在，異常會被捕獲但不會被吞掉
```

### 1.3 異常處理細節

```python
class SafeFile:
    """安全文件操作 - 吞掉特定異常"""
    def __init__(self, filename, mode, ignore_errors=False):
        self.filename = filename
        self.mode = mode
        self.ignore_errors = ignore_errors
        self.file = None
    
    def __enter__(self):
        try:
            self.file = open(self.filename, self.mode)
            return self.file
        except FileNotFoundError:
            if self.ignore_errors:
                return None
            raise
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.file:
            self.file.close()
        
        # 只吞掉 FileNotFoundError
        if self.ignore_errors and exc_type == FileNotFoundError:
            print(f"文件不存在：{self.filename}，已忽略")
            return True
        
        return False

# 使用
with SafeFile('nonexistent.txt', 'r', ignore_errors=True) as f:
    if f:
        content = f.read()
    else:
        print("文件不存在，使用默認值")
```

### 2.1 contextmanager 裝飾器

```python
from contextlib import contextmanager

@contextmanager
def managed_file(filename, mode):
    """生成器實現的文件上下文管理器"""
    print(f"打開文件：{filename}")
    f = open(filename, mode)
    try:
        yield f
    finally:
        print(f"關閉文件：{filename}")
        f.close()

# 使用
with managed_file('test.txt', 'w') as f:
    f.write("Hello!")

# 帶異常處理
@contextmanager
def safe_open(filename, mode):
    try:
        f = open(filename, mode)
        yield f
    except FileNotFoundError:
        print(f"文件不存在：{filename}")
        yield None
    finally:
        if 'f' in locals() and f:
            f.close()
```

### 2.2 contextmanager 帶參數

```python
from contextlib import contextmanager
import time

@contextmanager
def timer(name="操作"):
    """計時器上下文管理器"""
    start = time.time()
    try:
        yield  # 不返回值，也可以 yield value
    finally:
        end = time.time()
        elapsed = end - start
        print(f"{name} 耗時：{elapsed:.4f}秒")

# 使用
with timer("計算"):
    result = sum(range(1000000))

with timer("排序"):
    result = sorted(range(1000000), reverse=True)
```

### 2.3 contextmanager 高級應用

```python
from contextlib import contextmanager
import sqlite3

@contextmanager
def database_connection(db_path):
    """數據庫連接上下文管理器"""
    conn = sqlite3.connect(db_path)
    try:
        yield conn
        conn.commit()
    except Exception as e:
        conn.rollback()
        raise
    finally:
        conn.close()

# 使用

    cursor = conn.cursor()
    cursor.execute("SELECT * FROM users")
    results = cursor.fetchall()

@contextmanager
def transaction(conn):
    """事務上下文管理器"""
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise

# 使用
with database_connection('test.db') as conn:
```

### 2.5 ExitStack 實用技巧

```python
from contextlib import ExitStack, contextmanager

# 清理回調
with ExitStack() as stack:
    # 註冊清理回調
    stack.callback(print, "清理資源 1")
    stack.callback(print, "清理資源 2")
    
    # 退出時會按註冊的相反順序執行回調
    # 輸出:
    # 清理資源 2
    # 清理資源 1

# 推遲關閉
@contextmanager
def deferred_close(resources):
    with ExitStack() as stack:
        # 註冊所有資源的關閉回調
        for resource in resources:
            stack.callback(resource.close)
        yield resources

# 使用
files = [open(f, 'r') for f in ['file1.txt', 'file2.txt']]
with deferred_close(files):
    for f in files:
        print(f.read())
# 退出時所有文件自動關閉
```

```python
# 實用示例：靜默執行
from contextlib import redirect_stdout, redirect_stderr
import io

def silent_exec(func, *args, **kwargs):
    """靜默執行函數，捕獲所有輸出"""
    stdout = io.StringIO()
    stderr = io.StringIO()
    with redirect_stdout(stdout), redirect_stderr(stderr):
        result = func(*args, **kwargs)
    return result, stdout.getvalue(), stderr.getvalue()

# 使用
def noisy_function():
    print("正常輸出")
    print("錯誤輸出", file=__import__('sys').stderr)
    return "結果"

result, out, err = silent_exec(noisy_function)
print(f"結果：{result}")
print(f"輸出：{out}")
print(f"錯誤：{err}")
```

### 3.1 文件操作

```python
from contextlib import contextmanager
import os
import shutil

@contextmanager
def atomic_write(filename, mode='w'):
    """原子寫入 - 要麼全部寫入，要麼不寫"""
    temp_filename = filename + '.tmp'
    try:
        with open(temp_filename, mode) as f:
            yield f
        # 寫入成功，原子替換
        shutil.move(temp_filename, filename)
    except:
        # 寫入失敗，清理臨時文件
        if os.path.exists(temp_filename):
            os.remove(temp_filename)
        raise

# 使用
with atomic_write('data.txt', 'w') as f:
    f.write("重要數據")
# 要麼文件完整寫入，要麼原文件不變
```

### 3.3 鎖管理

```python
from contextlib import contextmanager
import threading

class ResourceManager:
    def __init__(self):
        self.lock = threading.Lock()
        self.resources = {}
    
    @contextmanager
    def acquire_resource(self, name):
        """獲取資源鎖"""
        self.lock.acquire()
        try:
            if name not in self.resources:
                self.resources[name] = {"count": 0, "data": None}
            self.resources[name]["count"] += 1
            yield self.resources[name]["data"]
        finally:
            self.resources[name]["count"] -= 1
            if self.resources[name]["count"] == 0:
                del self.resources[name]
            self.lock.release()

# 使用
manager = ResourceManager()
with manager.acquire_resource('database') as db:
    # 使用數據庫
    pass
```

### 3.4 臨時文件/目錄

```python
from contextlib import contextmanager
import tempfile
import shutil

@contextmanager
def temp_file(suffix='', prefix='tmp', dir=None):
    """臨時文件上下文管理器"""
    fd, path = tempfile.mkstemp(suffix=suffix, prefix=prefix, dir=dir)
    try:
        os.close(fd)  # 關閉文件描述符
        with open(path, 'r+b') as f:
            yield f, path
    finally:
        if os.path.exists(path):
            os.remove(path)

@contextmanager
def temp_directory(suffix='', prefix='tmp', dir=None):
    """臨時目錄上下文管理器"""
    path = tempfile.mkdtemp(suffix=suffix, prefix=prefix, dir=dir)
    try:
        yield path
    finally:
        shutil.rmtree(path)

# 使用
with temp_file() as (f, path):
    f.write(b"臨時數據")
    print(f"臨時文件：{path}")
# 退出後自動刪除

with temp_directory() as tmpdir:
    # 在臨時目錄中創建文件
    with open(os.path.join(tmpdir, 'test.txt'), 'w') as f:
        f.write("測試")
# 退出後整個目錄被刪除
```

## 4️⃣ 實戰：文件處理工具庫

### 4.1 批量文件處理器

```python
from contextlib import ExitStack, contextmanager
from pathlib import Path
import os

class FileProcessor:
    """批量文件處理器"""
    
    def __init__(self, input_dir, output_dir):
        self.input_dir = Path(input_dir)
        self.output_dir = Path(output_dir)
        self.output_dir.mkdir(parents=True, exist_ok=True)
    
    @contextmanager
    def process_batch(self, pattern='*.txt'):
        """批量處理文件"""
        with ExitStack() as stack:
            input_files = list(self.input_dir.glob(pattern))
            output_files = []
            
            for input_file in input_files:
                outfile = self.output_dir / input_file.name
                infile_ctx = stack.enter_context(open(input_file, 'r', encoding='utf-8'))
                outfile_ctx = stack.enter_context(open(outfile, 'w', encoding='utf-8'))
                output_files.append((infile_ctx, outfile_ctx, input_file.name))
            
            yield output_files
    
    def transform(self, transform_func, pattern='*.txt'):
        """應用轉換函數"""
        with self.process_batch(pattern) as files:
            for infile, outfile, filename in files:
                content = infile.read()
                transformed = transform_func(content)
                outfile.write(transformed)
                print(f"處理：{filename}")

# 使用
processor = FileProcessor('input', 'output')

# 轉換為大寫
processor.transform(lambda x: x.upper())

# 添加行號
def add_line_numbers(content):
    lines = content.split('\n')
    numbered = [f"{i+1}: {line}" for i, line in enumerate(lines)]
    return '\n'.join(numbered)

processor.transform(add_line_numbers)
```

### 4.2 日誌輪轉器

```python
from contextlib import contextmanager
from datetime import datetime
import os

class LogRotator:
    """日誌輪轉器"""
    
    def __init__(self, log_dir, max_size=10*1024*1024, max_files=5):
        self.log_dir = log_dir
        self.max_size = max_size
        self.max_files = max_files
        os.makedirs(log_dir, exist_ok=True)
    
    def _get_current_log(self):
        """獲取當前日誌文件"""
        today = datetime.now().strftime('%Y-%m-%d')
        return os.path.join(self.log_dir, f'app.{today}.log')
    
    def _rotate_if_needed(self):
        """檢查並輪轉"""
        log_file = self._get_current_log()
        if os.path.exists(log_file):
            size = os.path.getsize(log_file)
            if size > self.max_size:
                # 輪轉
                timestamp = datetime.now().strftime('%Y-%m-%d_%H-%M-%S')
                rotated = os.path.join(self.log_dir, f'app.{timestamp}.log')
                os.rename(log_file, rotated)
                
                # 清理舊文件
                self._cleanup_old_files()
    
    def _cleanup_old_files(self):
        """清理舊日誌文件"""
        files = sorted([
            os.path.join(self.log_dir, f)
            for f in os.listdir(self.log_dir)
            if f.startswith('app.') and f.endswith('.log')
        ])
        
        while len(files) > self.max_files:
            os.remove(files.pop(0))
    
    @contextmanager
    def open_log(self, mode='a'):
        """打開日誌文件"""
        self._rotate_if_needed()
        log_file = self._get_current_log()
        f = open(log_file, mode, encoding='utf-8')
        try:
            yield f
        finally:
            f.close()
    
    def write(self, message):
        """寫入日誌"""
        timestamp = datetime.now().strftime('%Y-%m-%d %H:%M:%S')
        with self.open_log() as f:
            f.write(f"[{timestamp}] {message}\n")

# 使用
logger = LogRotator('logs', max_size=1024*1024, max_files=7)
logger.write("應用啟動")
logger.write("處理請求")
```

### 4.3 CSV 數據處理器

```python
from contextlib import contextmanager
import csv
from pathlib import Path

class CSVProcessor:
    """CSV 數據處理器"""
    
    def __init__(self, input_file, output_file=None):
        self.input_file = Path(input_file)
        self.output_file = Path(output_file) if output_file else None
    
    @contextmanager
    def read(self, **kwargs):
        """讀取 CSV"""
        with open(self.input_file, 'r', encoding='utf-8', newline='') as f:
            reader = csv.DictReader(f, **kwargs)
            yield reader
    
    @contextmanager
    def write(self, fieldnames, **kwargs):
        """寫入 CSV"""
        if not self.output_file:
            raise ValueError("需要指定輸出文件")
        
        with open(self.output_file, 'w', encoding='utf-8', newline='') as f:
            writer = csv.DictWriter(f, fieldnames=fieldnames, **kwargs)
            writer.writeheader()
            yield writer
    
    @contextmanager
    def transform(self, transform_func, **kwargs):
        """轉換 CSV 數據"""
        if not self.output_file:
            raise ValueError("需要指定輸出文件")
        
        with open(self.input_file, 'r', encoding='utf-8', newline='') as infile, \
             open(self.output_file, 'w', encoding='utf-8', newline='') as outfile:
            
            reader = csv.DictReader(infile, **kwargs)
            writer = csv.DictWriter(outfile, fieldnames=reader.fieldnames)
            writer.writeheader()
            
            for row in reader:
                transformed = transform_func(row)
                if transformed:
                    writer.writerow(transformed)

# 使用
processor = CSVProcessor('input.csv', 'output.csv')

# 過濾數據
with processor.transform(lambda row: row if int(row['age']) >= 18 else None):
    pass

# 添加字段
def add_status(row):
    row['status'] = 'active' if int(row['age']) >= 18 else 'minor'
    return row

with processor.transform(add_status):
    pass
```

### 4.4 配置管理器

```python
from contextlib import contextmanager
import json
import os

class ConfigManager:
    """配置文件管理器"""
    
    def __init__(self, config_file):
        self.config_file = config_file
        self.config = self._load()
    
    def _load(self):
        """加載配置"""
        if os.path.exists(self.config_file):
            with open(self.config_file, 'r', encoding='utf-8') as f:
                return json.load(f)
        return {}
    
    def _save(self):
        """保存配置"""
        with open(self.config_file, 'w', encoding='utf-8') as f:
            json.dump(self.config, f, indent=2, ensure_ascii=False)
    
    @contextmanager
    def edit(self):
        """編輯配置"""
        yield self.config
        self._save()
    
    def get(self, key, default=None):
        """獲取配置"""
        return self.config.get(key, default)
    
    def set(self, key, value, save=True):
        """設置配置"""
        self.config[key] = value
        if save:
            self._save()

# 使用
config = ConfigManager('config.json')

# 讀取配置
db_host = config.get('database.host', 'localhost')

# 批量修改
with config.edit() as cfg:
    cfg['database.host'] = 'new-host'
    cfg['database.port'] = 5432
    cfg['cache.enabled'] = True
# 退出時自動保存
```
