package engine

import (
    "goex/engine/chromedp"
    "goex/engine/rod"

    "time"

type BrowserEngine interface {
    Navigate(url string) error
    Click(selector string) error
    Screenshot() ([]byte, error)
    ExtractContent() (string, error)
    ExecJS(script string) (interface{}, error)
    Wait(selector string) error
    Close()
}

type AutoSwitchEngine struct {
    currentEngine BrowserEngine
    usingRod bool
}

// 引擎预热：启动就初始化
func NewAutoSwitchEngine() *AutoSwitchEngine {
    e := &AutoSwitchEngine{}
    rodEngine := rod.New()
    e.currentEngine = rodEngine
    e.usingRod = true
    log.Info("【优化版】引擎预热完成：Rod")
    return e
}

func (e *AutoSwitchEngine) safeSwitch() {
    if !e.usingRod {
        return
    }
    log.Warn("【优化版】自动安全切换 → ChromeDP")
    e.currentEngine.Close()
    e.currentEngine = chromedp.New()
    e.usingRod = false
}

// 自动重试 3 次
func (e *AutoSwitchEngine) Navigate(url string) error {
    var err error
    for i := 0; i < 3; i++ {
        err = e.currentEngine.Navigate(url)
        if err == nil {
            return nil
        }
        time.Sleep(300 * time.Millisecond)
        e.safeSwitch()
    }
    return err
}

func (e *AutoSwitchEngine) Click(selector string) error {
    var err error
    for i := 0; i < 3; i++ {
        err = e.currentEngine.Click(selector)
        if err == nil {
            return nil
        }
        time.Sleep(300 * time.Millisecond)
        e.safeSwitch()
    }
    return err
}
