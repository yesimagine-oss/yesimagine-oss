## 🛡️ 主機安全

### 1. 系統更新

```bash
# Alibaba Cloud Linux
sudo yum update -y

# Ubuntu/Debian
sudo apt update && sudo apt upgrade -y

# 檢查內核版本
uname -r
```

### 2. 用戶權限管理

```bash
# 創建 Docker 用戶組
sudo groupadd docker

# 添加用戶到 Docker 組
sudo usermod -aG docker $USER

# 驗證
newgrp docker
docker run hello-world

# ⚠️ 警告：Docker 組用戶具有 root 等效權限
# 不要隨意將用戶加入 Docker 組
```

### 3. 防火牆配置

```bash
# 配置 iptables 規則
# 只允許特定端口
iptables -A INPUT -p tcp --dport 2376 -j ACCEPT  # Docker TLS
iptables -A INPUT -p tcp --dport 2375 -j DROP    # 禁用未加密

# 保存規則
iptables-save > /etc/iptables/rules.v4
```

## 🔐 守護進程安全

### 1. 啟用 TLS 認證

```bash
# 1. 創建 CA
mkdir -p ~/docker-ca
cd ~/docker-ca
openssl genrsa -aes256 -out ca-key.pem 4096
openssl req -new -x509 -days 365 -key ca-key.pem -sha256 -out ca.pem

# 2. 創建服務器證書
openssl genrsa -out server-key.pem 4096
openssl req -subj "/CN=iZm5ei3ekpe8wbnvf7snniZ" -sha256 \
  -new -key server-key.pem -out server.csr
echo subjectAltName = DNS:iZm5ei3ekpe8wbnvf7snniZ,IP:127.0.0.1 >> extfile.cnf
echo extendedKeyUsage = serverAuth >> extfile.cnf
openssl x509 -req -days 365 -sha256 -in server.csr \
  -CA ca.pem -CAkey ca-key.pem -CAcreateserial \
  -out server-cert.pem -extfile extfile.cnf

# 3. 創建客戶端證書
openssl genrsa -out key.pem 4096
openssl req -subj '/CN=client' -new -key key.pem -out client.csr
echo extendedKeyUsage = clientAuth > extfile-client.cnf
openssl x509 -req -days 365 -sha256 -in client.csr \
  -CA ca.pem -CAkey ca-key.pem -CAcreateserial \
  -out cert.pem -extfile extfile-client.cnf

# 4. 設置權限
chmod -v 0400 ca-key.pem key.pem server-key.pem
chmod -v 0444 ca.pem server-cert.pem cert.pem

# 5. 移動證書
sudo mkdir -p /etc/docker/certs
sudo cp ca.pem server-cert.pem server-key.pem /etc/docker/certs/

# 6. 配置 Docker Daemon
# /etc/docker/daemon.json
{
  "tls": true,
  "tlsverify": true,
  "tlscacert": "/etc/docker/certs/ca.pem",
  "tlscert": "/etc/docker/certs/server-cert.pem",
  "tlskey": "/etc/docker/certs/server-key.pem",
  "hosts": ["tcp://0.0.0.0:2376"]
}

# 7. 重啟 Docker
sudo systemctl daemon-reload
sudo systemctl restart docker
```

### 2. 配置 daemon.json 安全選項

```json
{
  "tls": true,
  "tlsverify": true,
  "live-restore": true,
  "userland-proxy": false,
  "no-new-privileges": true,
  "log-opts": {
    "max-size": "10m",
    "max-file": "3"
  },
  "default-address-pools": [
    {
      "base": "10.255.0.0/16",
      "size": 24
    }
  ]
}
```

## 📦 容器安全

### 1. 最小權限原則

```bash
# ✅ 推薦：刪除所有能力，只添加需要的
docker run -d \
  --cap-drop=ALL \
  --cap-add=NET_BIND_SERVICE \
  nginx

# ❌ 避免：特權模式
docker run --privileged nginx

# 常見能力說明
# NET_BIND_SERVICE  - 綁定低端口 (<1024)
# SYS_ADMIN         - 管理員權限 (危險)
# SYS_PTRACE        - 調試進程
# NET_ADMIN         - 網絡管理
# SYS_MODULE        - 加載內核模塊
```

### 2. 只讀文件系統

```bash
# 只讀根文件系統
docker run -d \
  --read-only \
  --tmpfs /tmp \
  --tmpfs /var/run \
  nginx

# 掛載特定可寫目錄
docker run -d \
  --read-only \
  -v /var/lib/mysql:/var/lib/mysql \
  mysql
```

### 3. 用戶命名空間

```bash
# 啟用用戶命名空間
# /etc/docker/daemon.json
{
  "userns-remap": "default"
}

# 創建映射用戶
sudo useradd dockremap

# 配置映射
echo dockremap:165536:65536 | sudo tee -a /etc/subuid
echo dockremap:165536:65536 | sudo tee -a /etc/subgid

# 重啟 Docker
sudo systemctl restart docker
```

### 4. 資源限制

```bash
# CPU 限制
docker run -d \
  --cpus="1" \
  --cpu-shares="512" \
  nginx

# 內存限制
docker run -d \
  --memory="512m" \
  --memory-swap="512m" \
  --memory-reservation="256m" \
  nginx

# 進程數限制
docker run -d \
  --pids-limit=100 \
  nginx

# 磁盤配額 (需要配額支持)
docker run -d \
  --device-read-bps="/dev/sda:1mb" \
  --device-write-bps="/dev/sda:1mb" \
  nginx
```

### 5. 安全選項

```bash
# no-new-privileges
docker run -d \
  --security-opt no-new-privileges:true \
  nginx

# Seccomp 配置文件
docker run -d \
  --security-opt seccomp=/path/to/seccomp-profile.json \
  nginx

# AppArmor 配置文件
docker run -d \
  --security-opt apparmor=/path/to/apparmor-profile \
  nginx

# 禁用特定安全功能 (不推薦)
docker run -d \
  --security-opt seccomp=unconfined \
  --security-opt apparmor=unconfined \
  nginx
```

### 6. 網絡隔離

```bash
# 創建專用網絡
docker network create --driver bridge isolated-net

# 運行容器到專用網絡
docker run -d \
  --network=isolated-net \
  --name=app1 \
  nginx

# 禁用外部網絡訪問
docker run -d \
  --network=none \
  nginx

# 限制端口暴露
docker run -d \
  -p 127.0.0.1:8080:80 \
  nginx
```

## 🖼️ 鏡像安全

### 1. 選擇可信鏡像

```bash
# ✅ 推薦：官方鏡像
docker pull nginx:latest
docker pull node:22-bookworm

# ✅ 推薦：驗證簽名
docker trust inspect --pretty nginx

# ❌ 避免：未知來源鏡像
docker pull unknown-user/unknown-image
```

### 2. 掃描鏡像漏洞

```bash
# 使用 Docker Scan
docker scan nginx

# 使用 Trivy
trivy image nginx:latest

# 使用 Clair
clair-scanner -w example-alpine.yaml alpine:3.14
```

### 3. 最小化基礎鏡像

```dockerfile
# ❌ 避免：完整鏡像
FROM ubuntu:24.04

# ✅ 推薦：精簡鏡像
FROM ubuntu:24.04-slim

# ✅ 推薦：極簡鏡像
FROM alpine:3.18

# ✅ 推薦：Distroless (Google)
FROM gcr.io/distroless/static-debian11
```

### 4. 多階段構建

```dockerfile
# 構建階段
FROM node:22 AS builder
WORKDIR /app
COPY package*.json ./
RUN npm ci --only=production
COPY . .

# 運行階段 (只包含必要文件)
FROM node:22-slim
WORKDIR /app
COPY --from=builder /app/node_modules ./node_modules
COPY --from=builder /app/dist ./dist
USER node
CMD ["node", "dist/index.js"]
```

## 🔍 安全審計

### 1. Docker Bench for Security

```bash
# 克隆倉庫
git clone https://github.com/docker/docker-bench-security.git
cd docker-bench-security

# 運行審計
sudo sh docker-bench-security.sh

# 查看結果
cat log/docker-bench-security.log
```

### 2. 檢查容器配置

```bash
# 檢查特權容器
docker ps --format '{{.Names}}' | while read name; do
  if [ "$(docker inspect --format '{{.HostConfig.Privileged}}' $name)" = "true" ]; then
    echo "⚠️ 特權容器：$name"
  fi
done

# 檢查 root 運行的容器
docker ps --format '{{.Names}}' | while read name; do
  if [ "$(docker inspect --format '{{.Config.User}}' $name)" = "" ]; then
    echo "⚠️ Root 運行：$name"
  fi
done

# 檢查能力
docker inspect --format '{{.HostConfig.CapAdd}}' <container>
```

### 3. 監控異常行為

```bash
# 監控容器事件
docker events --filter 'type=container'

# 監控文件系統變化
docker diff <container>

# 監控進程
docker top <container>
```

## 🚨 安全事件響應

### 1. 隔離受感染容器

```bash
# 斷開網絡
docker network disconnect bridge <container>

# 停止容器
docker stop <container>

# 保存證據
docker export <container> > container-<id>-$(date +%Y%m%d).tar
docker logs <container> > container-<id>-logs.txt
```

### 2. 調查取證

```bash
# 檢查容器配置
docker inspect <container> > container-inspect.json

# 檢查文件系統變化
docker diff <container> > container-diff.txt

# 檢查進程
docker top <container> > container-top.txt
```

### 3. 恢復與修復

```bash
# 刪除受感染容器
docker rm -f <container>

# 刪除受感染鏡像
docker rmi <image>

# 重新部署安全版本
docker run [安全選項] <safe-image>
```
