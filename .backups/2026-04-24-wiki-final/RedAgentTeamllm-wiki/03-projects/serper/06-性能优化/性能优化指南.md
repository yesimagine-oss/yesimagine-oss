## 1️⃣ 缓存策略

### 问题

重复查询相同关键词导致成本增加

### 解决方案

```python
import requests
import hashlib
import json
import time
from typing import Optional, Dict

class CachedSerperClient:
    def __init__(self, api_key, cache_ttl=3600):
        self.api_key = api_key
        self.cache_ttl = cache_ttl  # 缓存有效期（秒）
        self.cache: Dict[str, dict] = {}
        self.cache_timestamps: Dict[str, float] = {}
        
        self.headers = {
            "X-API-KEY": api_key,
            "Content-Type": "application/json"
        }
    
    def _get_cache_key(self, query: str, **params) -> str:
        """生成缓存键"""
        key_data = f"{query}:{json.dumps(params, sort_keys=True)}"
        return hashlib.md5(key_data.encode()).hexdigest()
    
    def _is_cache_valid(self, cache_key: str) -> bool:
        """检查缓存是否有效"""
        if cache_key not in self.cache:
            return False
        
        age = time.time() - self.cache_timestamps[cache_key]
        return age < self.cache_ttl
    
    def search(self, query: str, use_cache=True, **params) -> dict:
        """搜索（带缓存）"""
        cache_key = self._get_cache_key(query, **params)
        
        # 检查缓存
        if use_cache and self._is_cache_valid(cache_key):
            print(f"Cache hit for: {query}")
            return self.cache[cache_key]
        
        # API 请求
        print(f"API request for: {query}")
        response = requests.post(
            "https://google.serper.dev/search",
            headers=self.headers,
            json={"q": query, **params}
        )
        response.raise_for_status()
        result = response.json()
        
        # 更新缓存
        self.cache[cache_key] = result
        self.cache_timestamps[cache_key] = time.time()
        
        return result
    
    def get_cache_stats(self) -> dict:
        """获取缓存统计"""
        return {
            "total_keys": len(self.cache),
            "memory_usage": f"{len(json.dumps(self.cache)) / 1024:.2f} KB"
        }
    
    def clear_cache(self):
        """清空缓存"""
        self.cache.clear()
        self.cache_timestamps.clear()

# 使用示例
client = CachedSerperClient(
    api_key="01529847d4aa3cf47b86ca87d28519110db06390",
    cache_ttl=3600  # 1 小时缓存
)

# 第一次请求（API 调用）
result1 = client.search("AI agent")

# 第二次请求（缓存命中）
result2 = client.search("AI agent")

# 查看缓存统计
print(client.get_cache_stats())
```

## 2️⃣ 并发控制

### 问题

单线程请求效率低

### 解决方案

```python
import requests
import asyncio
import aiohttp
from typing import List, Dict

class AsyncSerperClient:
    def __init__(self, api_key, max_concurrency=10):
        self.api_key = api_key
        self.max_concurrency = max_concurrency
        self.headers = {
            "X-API-KEY": api_key,
            "Content-Type": "application/json"
        }
    
    async def _search_single(self, session: aiohttp.ClientSession, query: str, **params) -> dict:
        """单个搜索请求"""
        url = "https://google.serper.dev/search"
        
        async with session.post(url, headers=self.headers, json={"q": query, **params}) as response:
            return await response.json()
    
    async def search_batch(self, queries: List[str], **params) -> List[dict]:
        """批量搜索"""
        semaphore = asyncio.Semaphore(self.max_concurrency)
        
        async def limited_search(session, query):
            async with semaphore:
                return await self._search_single(session, query, **params)
        
        async with aiohttp.ClientSession() as session:
            tasks = [limited_search(session, query) for query in queries]
            results = await asyncio.gather(*tasks)
        
        return results

# 使用示例
async def main():
    client = AsyncSerperClient(
        api_key="01529847d4aa3cf47b86ca87d28519110db06390",
        max_concurrency=10
    )
    
    queries = [f"keyword {i}" for i in range(100)]
    
    import time
    start = time.time()
    results = await client.search_batch(queries)
    end = time.time()
    
    print(f"Searched {len(queries)} queries in {end - start:.2f} seconds")
    print(f"Average: {(end - start) / len(queries) * 1000:.2f} ms/query")

# 运行
asyncio.run(main())
```

class OptimizedSerperClient:
    def __init__(self, api_key):
        self.api_key = api_key
        self.headers = {
            "X-API-KEY": api_key,
            "Content-Type": "application/json"
        }
    
    def search_optimized(self, query: str, need_all_fields=True) -> dict:
        """优化的搜索请求"""
        params = {
            "q": query,
            "num": 10 if need_all_fields else 5,  # 减少结果数量
            "page": 1
        }
        
        # 如果不需要完整信息，减少字段
        if not need_all_fields:
            params['autocorrect'] = False
        
        response = requests.post(
            "https://google.serper.dev/search",
            headers=self.headers,
            json=params
        )
        return response.json()
    
    def extract_only_titles(self, results: dict) -> List[str]:
        """只提取标题（减少数据处理）"""
        return [r['title'] for r in results.get('organic', [])]
    
    def extract_only_links(self, results: dict) -> List[str]:
        """只提取链接"""
        return [r['link'] for r in results.get('organic', [])]

# 使用示例
client = OptimizedSerperClient("01529847d4aa3cf47b86ca87d28519110db06390")

# 场景 1: 只需要标题
results = client.search_optimized("AI agent", need_all_fields=False)

# 场景 2: 需要完整信息
results = client.search_optimized("AI agent", need_all_fields=True)

## 4️⃣ 错误处理优化

### 问题

错误处理不完善导致程序崩溃

### 解决方案

```python
import time
from typing import Optional

class RobustSerperClient:
    def __init__(self, api_key, max_retries=3):
        self.api_key = api_key
        self.max_retries = max_retries
        self.headers = {
            "X-API-KEY": api_key,
            "Content-Type": "application/json"
        }
    
    def search_with_retry(self, query: str, **params) -> Optional[dict]:
        """带重试的搜索"""
        for attempt in range(self.max_retries):
            try:
                response = requests.post(
                    "https://google.serper.dev/search",
                    headers=self.headers,
                    json={"q": query, **params},
                    timeout=30
                )
                
                if response.status_code == 200:
                    return response.json()
                
                elif response.status_code == 429:
                    # 速率限制，指数退避
                    wait_time = 2 ** attempt
                    print(f"Rate limited. Waiting {wait_time}s...")
                    time.sleep(wait_time)
                
                elif response.status_code >= 500:
                    # 服务器错误，重试
                    wait_time = 2 ** attempt
                    print(f"Server error. Retrying in {wait_time}s...")
                    time.sleep(wait_time)
                
                else:
                    # 其他错误，不重试
                    print(f"Error {response.status_code}: {response.text}")
                    return None
            
            except requests.exceptions.Timeout:
                print(f"Timeout on attempt {attempt + 1}")
                if attempt == self.max_retries - 1:
                    return None
                time.sleep(2 ** attempt)
            
            except requests.exceptions.RequestException as e:
                print(f"Request failed: {e}")
                return None
        
        return None
```

### 问题

无法及时发现 API 使用异常

### 解决方案

```python
import time
from collections import deque
from datetime import datetime

class MonitoredSerperClient:
    def __init__(self, api_key, alert_threshold=100):
        self.api_key = api_key
        self.alert_threshold = alert_threshold  # 错误次数阈值
        self.headers = {
            "X-API-KEY": api_key,
            "Content-Type": "application/json"
        }
        
        # 监控指标
        self.request_times = deque(maxlen=1000)
        self.error_times = deque(maxlen=100)
        self.credits_used = 0
    
    def search_with_monitoring(self, query: str, **params) -> dict:
        """带监控的搜索"""
        start_time = time.time()
        
        try:
            response = requests.post(
                "https://google.serper.dev/search",
                headers=self.headers,
                json={"q": query, **params},
                timeout=30
            )
            
            # 记录请求
            self.request_times.append({
                'timestamp': time.time(),
                'latency': time.time() - start_time,
                'status': response.status_code
            })
            
            # 记录错误
            if response.status_code != 200:
                self.error_times.append({
                    'timestamp': time.time(),
                    'error': response.status_code
                })
                
                # 检查是否需要告警
                self._check_alerts()
            
            # 记录 Credits 使用
            if response.status_code == 200:
                self.credits_used += 1
            
            response.raise_for_status()
            return response.json()
        
        except Exception as e:
            self.error_times.append({
                'timestamp': time.time(),
                'error': str(e)
            })
            self._check_alerts()
            raise
    
    def _check_alerts(self):
        """检查是否需要告警"""
        # 检查错误率
        now = time.time()
        recent_errors = [e for e in self.error_times if now - e['timestamp'] < 300]  # 5 分钟内
        
        if len(recent_errors) >= self.alert_threshold:
            self._send_alert(f"High error rate: {len(recent_errors)} errors in 5 minutes")
    
    def _send_alert(self, message: str):
        """发送告警"""
        print(f"🚨 ALERT: {message}")
        # 可以集成邮件、Slack、钉钉等告警方式
    
    def get_stats(self) -> dict:
        """获取统计信息"""
        now = time.time()
        
        # 计算 P95 延迟
        latencies = [r['latency'] for r in self.request_times]
        p95_latency = sorted(latencies)[int(len(latencies) * 0.95)] if latencies else 0
        
        # 计算错误率
        recent_requests = [r for r in self.request_times if now - r['timestamp'] < 300]
        error_count = len([r for r in recent_requests if r['status'] != 200])
        error_rate = error_count / len(recent_requests) if recent_requests else 0
        
        return {
            'total_requests': len(self.request_times),
            'p95_latency_ms': p95_latency * 1000,
            'error_rate': f"{error_rate * 100:.2f}%",
            'credits_used': self.credits_used,
            'recent_errors': len([e for e in self.error_times if now - e['timestamp'] < 300])
        }

# 使用示例
client = MonitoredSerperClient(
    api_key="01529847d4aa3cf47b86ca87d28519110db06390",
    alert_threshold=10
)

# 搜索
result = client.search_with_monitoring("AI agent")

# 查看统计
stats = client.get_stats()
print(stats)
```
