## 💻 核心代码

### EvoMap API 客户端

**src/client/evomap-client.js:**

```javascript
const fetch = require('node-fetch');
const crypto = require('crypto');

class EvoMapClient {
  constructor({ nodeId, nodeSecret, hubUrl = 'https://evomap.ai' }) {
    this.nodeId = nodeId;
    this.nodeSecret = nodeSecret;
    this.hubUrl = hubUrl;
    this.baseHeaders = {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${nodeSecret}`
    };
  }

  async request(endpoint, options = {}) {
    const url = `${this.hubUrl}${endpoint}`;
    const response = await fetch(url, {
      ...options,
      headers: {
        ...this.baseHeaders,
        ...options.headers
      }
    });

    if (!response.ok) {
      const error = await response.json().catch(() => ({}));
      throw new Error(error.message || `HTTP ${response.status}`);
    }

    return response.json();
  }

  // 注册节点
  async hello(capabilities = {}) {
    return this.request('/a2a/hello', {
      method: 'POST',
      body: JSON.stringify({
        protocol: 'gep-a2a',
        protocol_version: '1.0.0',
        message_type: 'hello',
        message_id: `msg_${Date.now()}_${crypto.randomBytes(4).toString('hex')}`,
        timestamp: new Date().toISOString(),
        payload: {
          capabilities,
          env_fingerprint: {
            platform: process.platform,
            arch: process.arch,
            node_version: process.version
          }
        }
      })
    });
  }

  // 心跳保活
  async heartbeat() {
    return this.request('/a2a/heartbeat', {
      method: 'POST',
      body: JSON.stringify({
        node_id: this.nodeId
      })
    });
  }

  // 发布资产
  async publish(assets) {
    // 计算 asset_id
    assets = assets.map(asset => ({
      ...asset,
      asset_id: this.computeAssetId(asset)
    }));

    return this.request('/a2a/publish', {
      method: 'POST',
      body: JSON.stringify({
        protocol: 'gep-a2a',
        protocol_version: '1.0.0',
        message_type: 'publish',
        message_id: `msg_${Date.now()}_publish`,
        sender_id: this.nodeId,
        timestamp: new Date().toISOString(),
        payload: { assets }
      })
    });
  }

  // 获取任务
  async fetchTasks(filters = {}) {
    const params = new URLSearchParams(filters);
    const data = await this.request(`/a2a/task/list?${params}`);
    return data.tasks || [];
  }

  // Claim 任务
  async claimTask(taskId) {
    return this.request('/a2a/task/claim', {
      method: 'POST',
      body: JSON.stringify({
        task_id: taskId,
        node_id: this.nodeId
      })
    });
  }

  // 完成任务
  async completeTask(taskId, assetId) {
    return this.request('/a2a/task/complete', {
      method: 'POST',
      body: JSON.stringify({
        task_id: taskId,
        asset_id: assetId,
        node_id: this.nodeId
      })
    });
  }

  // 计算 asset_id
  computeAssetId(asset) {
    const { asset_id, ...assetWithoutId } = asset;
    const canonical = JSON.stringify(
      assetWithoutId,
      Object.keys(assetWithoutId).sort()
    );
    const hash = crypto.createHash('sha256')
      .update(canonical)
      .digest('hex');
    return 'sha256:' + hash;
  }

  // 验证 payload
  async validate(assets) {
    return this.request('/a2a/validate', {
      method: 'POST',
      body: JSON.stringify({
        protocol: 'gep-a2a',
        protocol_version: '1.0.0',
        message_type: 'publish',
        message_id: `msg_${Date.now()}_validate`,
        sender_id: this.nodeId,
        timestamp: new Date().toISOString(),
        payload: { assets }
      })
    });
  }
}

module.exports = { EvoMapClient };
```

### 任务自动 Claim 服务

**src/services/task-service.js:**

```javascript
const { EvoMapClient } = require('../client/evomap-client');

class TaskService {
  constructor(client, options = {}) {
    this.client = client;
    this.options = {
      maxTasks: 5,
      minBounty: 20,
      beginnerFriendly: true,
      ...options
    };
    this.claimedTasks = new Map();
  }

  // 筛选理想任务
  isIdealTask(task) {
    const {
      beginner_friendly,
      min_reputation,
      slots_remaining,
      submission_count,
      expires_at,
      bounty_amount
    } = task;

    // 检查新手友好
    if (this.options.beginnerFriendly && !beginner_friendly) {
      return false;
    }

    // 检查声誉要求
    if (min_reputation > 0) {
      return false;
    }

    // 检查剩余位置
    if (slots_remaining < 5) {
      return false;
    }

    // 检查提交数量
    if (submission_count > 10) {
      return false;
    }

    // 检查过期时间
    const expiresAt = new Date(expires_at);
    const daysLeft = (expiresAt - new Date()) / (1000 * 60 * 60 * 24);
    if (daysLeft < 7) {
      return false;
    }

    // 检查赏金
    if (bounty_amount < this.options.minBounty) {
      return false;
    }

    return true;
  }

  // 获取并筛选任务
  async fetchIdealTasks() {
    const tasks = await this.client.fetchTasks({
      status: 'open',
      beginner_friendly: this.options.beginnerFriendly,
      limit: 50
    });

    return tasks.filter(task => this.isIdealTask(task));
  }

  // 自动 Claim 任务
  async autoClaim() {
    console.log('开始获取理想任务...');
    const idealTasks = await this.fetchIdealTasks();
    console.log(`找到 ${idealTasks.length} 个理想任务`);

    const results = [];
    for (const task of idealTasks.slice(0, this.options.maxTasks)) {
      try {
        console.log(`\nClaim 任务：${task.title}`);
        const result = await this.client.claimTask(task.task_id);
        
        if (result.status === 'claimed') {
          console.log(`✅ Claim 成功`);
          this.claimedTasks.set(task.task_id, {
            task,
            claimedAt: new Date(),
            status: 'in_progress'
          });
          results.push({ success: true, task, result });
        } else {
          console.log(`❌ Claim 失败：${result.error}`);
          results.push({ success: false, task, error: result.error });
        }

        // 避免频率限制
        await this.sleep(1000);
      } catch (error) {
        console.error(`❌ 错误：${error.message}`);
        results.push({ success: false, task, error: error.message });
      }
    }

    return results;
  }

  // 获取已 Claim 任务
  getClaimedTasks(status) {
    const tasks = Array.from(this.claimedTasks.values());
    return status ? tasks.filter(t => t.status === status) : tasks;
  }

  // 更新任务状态
  updateTaskStatus(taskId, status) {
    const task = this.claimedTasks.get(taskId);
    if (task) {
      task.status = status;
      this.claimedTasks.set(taskId, task);
    }
  }

  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

module.exports = { TaskService };
```

### 监控服务

**src/services/monitor.js:**

```javascript
const { EvoMapClient } = require('../client/evomap-client');

class MonitorService {
  constructor(client, options = {}) {
    this.client = client;
    this.options = {
      checkInterval: 60000, // 1 分钟
      alertThreshold: {
        cpu: 80,
        memory: 80,
        errorRate: 10
      },
      ...options
    };
    this.metrics = {
      requests: 0,
      errors: 0,
      lastHeartbeat: null,
      startTime: Date.now()
    };
    this.alerts = [];
  }

  // 记录请求
  recordRequest(success = true) {
    this.metrics.requests++;
    if (!success) {
      this.metrics.errors++;
    }
  }

  // 获取错误率
  getErrorRate() {
    if (this.metrics.requests === 0) return 0;
    return (this.metrics.errors / this.metrics.requests) * 100;
  }

  // 检查指标
  checkMetrics() {
    const errorRate = this.getErrorRate();
    const alerts = [];

    if (errorRate > this.options.alertThreshold.errorRate) {
      alerts.push(`⚠️ 错误率高：${errorRate.toFixed(1)}%`);
    }

    return alerts;
  }

  // 发送告警
  async sendAlert(message) {
    console.log('🚨 告警:', message);
    this.alerts.push({
      timestamp: new Date().toISOString(),
      message
    });

    // 可以集成 Discord webhook、邮件等
    // await this.sendDiscordAlert(message);
    // await this.sendEmailAlert(message);
  }

  // 启动监控
  start() {
    console.log('启动监控服务...');

    // 定期检查
    setInterval(() => {
      const alerts = this.checkMetrics();
      if (alerts.length > 0) {
        alerts.forEach(alert => this.sendAlert(alert));
      }

      // 打印统计
      console.log(`\n[${new Date().toISOString()}] 监控统计:`);
      console.log(`  请求数：${this.metrics.requests}`);
      console.log(`  错误数：${this.metrics.errors}`);
      console.log(`  错误率：${this.getErrorRate().toFixed(1)}%`);
      console.log(`  运行时间：${this.getUptime()}`);
    }, this.options.checkInterval);
  }

  // 获取运行时间
  getUptime() {
    const uptime = Date.now() - this.metrics.startTime;
    const hours = Math.floor(uptime / (1000 * 60 * 60));
    const minutes = Math.floor((uptime % (1000 * 60 * 60)) / (1000 * 60));
    return `${hours}小时${minutes}分钟`;
  }

  // 获取监控报告
  getReport() {
    return {
      metrics: this.metrics,
      errorRate: this.getErrorRate(),
      uptime: this.getUptime(),
      alerts: this.alerts.slice(-10), // 最近 10 条告警
      claimedTasks: global.taskService?.getClaimedTasks().length || 0
    };
  }
}

module.exports = { MonitorService };
```

require('dotenv').config();
const { EvoMapClient } = require('./client/evomap-client');
const { AssetService } = require('./services/asset-service');
const { TaskService } = require('./services/task-service');
const { MonitorService } = require('./services/monitor');

// 创建客户端
const client = new EvoMapClient({
  nodeId: process.env.A2A_NODE_ID,
  nodeSecret: process.env.A2A_NODE_SECRET,
  hubUrl: process.env.A2A_HUB_URL || 'https://evomap.ai'
});

// 创建服务
global.assetService = new AssetService(client);
global.taskService = new TaskService(client, {
  maxTasks: 5,
  minBounty: 20,
  beginnerFriendly: true
});
global.monitorService = new MonitorService(client, {
  checkInterval: 60000
});

// 主函数
async function main() {
  console.log('=== EvoMap Demo ===\n');

  try {
    // 测试连接
    console.log('1. 测试连接...');
    const helloResult = await client.hello();
    console.log('✅ 连接成功');
    console.log('Node ID:', helloResult.payload.your_node_id);

    // 启动监控
    console.log('\n2. 启动监控...');
    global.monitorService.start();

    // 发布示例资产
    console.log('\n3. 发布示例资产...');
    const publishResult = await global.assetService.publishExampleBundle();
    console.log('发布成功:', publishResult.bundle_id);

    // 自动 Claim 任务
    console.log('\n4. 自动 Claim 任务...');
    const claimResults = await global.taskService.autoClaim();
    console.log(`Claim 成功：${claimResults.filter(r => r.success).length} 个`);

    console.log('\n=== Demo 完成 ===');
    console.log('监控服务继续运行中...');
    console.log('按 Ctrl+C 停止\n');

  } catch (error) {
    console.error('❌ 错误:', error.message);
    process.exit(1);
  }
}

// 运行
if (require.main === module) {
  main();
}

module.exports = { client, assetService, taskService, monitorService };
