## 🔧 健康追踪脚本

```javascript
// ~/.openclaw/workspace/skills/health-tracker/index.js
const fs = require('fs');
const path = require('path');

module.exports = {
  meta: {
    name: 'health-tracker',
    version: '1.0.0',
    description: '健康数据追踪技能'
  },

  config: {
    enabled: true,
    dataPath: '~/.openclaw/health-data',
    goals: {
      weight: { min: 65, max: 70, unit: 'kg' },
      sleep: { min: 7, max: 9, unit: 'hours' },
      exercise: { min: 30, unit: 'minutes' },
      steps: { min: 8000, unit: 'steps' },
      water: { min: 2000, unit: 'ml' },
      mood: { min: 4, max: 5, unit: 'score' }
    }
  },

  async execute(context, params) {
    const action = params?.action || 'log';
    
    switch(action) {
      case 'log':
        return await this.logData(params);
      case 'report':
        return await this.generateReport(params.period);
      case 'trend':
        return await this.showTrend(params.metric);
      case 'tips':
        return await this.getHealthTips();
      default:
        return await this.getTodayStatus();
    }
  },

  async logData(params) {
    const { metric, value, date = new Date().toISOString().split('T')[0] } = params;
    const dataFile = path.join(this.config.dataPath, `${metric}.json`);
    
    // 读取现有数据
    let data = [];
    if (fs.existsSync(dataFile)) {
      data = JSON.parse(fs.readFileSync(dataFile, 'utf8'));
    }
    
    // 添加新记录
    data.push({ date, value, timestamp: new Date().toISOString() });
    
    // 保存
    fs.writeFileSync(dataFile, JSON.stringify(data, null, 2), 'utf8');
    
    // 检查是否达标
    const goal = this.config.goals[metric];
    const status = this.checkGoal(metric, value, goal);
    
    return {
      content: `✅ ${metric} 已记录\n\n数值：${value} ${goal?.unit || ''}\n状态：${status}`
    };
  },

  async generateReport(period = 'week') {
    const metrics = Object.keys(this.config.goals);
    const report = { period, date: new Date().toISOString(), metrics: {} };
    
    for (const metric of metrics) {
      const dataFile = path.join(this.config.dataPath, `${metric}.json`);
      if (!fs.existsSync(dataFile)) continue;
      
      const data = JSON.parse(fs.readFileSync(dataFile, 'utf8'));
      const filtered = this.filterByPeriod(data, period);
      
      report.metrics[metric] = {
        average: this.calculateAverage(filtered.map(d => d.value)),
        min: Math.min(...filtered.map(d => d.value)),
        max: Math.max(...filtered.map(d => d.value)),
        records: filtered.length,
        goalAchieved: this.calculateGoalAchievement(filtered, this.config.goals[metric])
      };
    }
    
    return { content: this.formatReport(report) };
  },

  async showTrend(metric) {
    const dataFile = path.join(this.config.dataPath, `${metric}.json`);
    if (!fs.existsSync(dataFile)) {
      return { content: `❌ 未找到 ${metric} 数据` };
    }
    
    const data = JSON.parse(fs.readFileSync(dataFile, 'utf8'));
    const last30 = data.slice(-30);
    
    let trend = `📈 ${metric} 趋势 (最近${last30.length}天)\n\n`;
    
    // 简单的 ASCII 图表
    const values = last30.map(d => d.value);
    const max = Math.max(...values);
    const min = Math.min(...values);
    const range = max - min || 1;
    
    for (const d of last30) {
      const barLength = Math.round(((d.value - min) / range) * 20);
      const bar = '█'.repeat(barLength) + '░'.repeat(20 - barLength);
      trend += `${d.date.slice(5)}: ${bar} ${d.value}\n`;
    }
    
    return { content: trend };
  },

  async getHealthTips() {
    const tips = [
      '💧 每天喝足够的水，保持身体水分平衡',
      '😴 保证 7-8 小时睡眠，有助于身体恢复',
      '🏃 每天运动 30 分钟，增强心肺功能',
      '🥗 均衡饮食，多吃蔬菜水果',
      '🧘 适当放松，减轻压力',
      '☀️ 多晒太阳，补充维生素 D',
      '📱 减少屏幕时间，保护视力'
    ];
    
    const randomTip = tips[Math.floor(Math.random() * tips.length)];
    
    return { content: `💡 健康小贴士\n\n${randomTip}` };
  },

  async getTodayStatus() {
    const today = new Date().toISOString().split('T')[0];
    let status = `📊 今日健康状态 - ${today}\n\n`;
    
    for (const [metric, goal] of Object.entries(this.config.goals)) {
      const dataFile = path.join(this.config.dataPath, `${metric}.json`);
      if (!fs.existsSync(dataFile)) {
        status += `${metric}: ⚪ 无数据\n`;
        continue;
      }
      
      const data = JSON.parse(fs.readFileSync(dataFile, 'utf8'));
      const todayData = data.find(d => d.date === today);
      
      if (!todayData) {
        status += `${metric}: ⚪ 未记录\n`;
      } else {
        const achieved = this.checkGoal(metric, todayData.value, goal);
        status += `${metric}: ${achieved ? '✅' : '❌'} ${todayData.value} ${goal.unit}\n`;
      }
    }
    
    return { content: status };
  },

  checkGoal(metric, value, goal) {
    if (!goal) return '未知';
    if (goal.min !== undefined && value < goal.min) return '未达标';
    if (goal.max !== undefined && value > goal.max) return '超标';
    return '✅ 达标';
  },

  filterByPeriod(data, period) {
    const now = new Date();
    const days = period === 'week' ? 7 : period === 'month' ? 30 : 365;
    const cutoff = new Date(now.getTime() - days * 24 * 60 * 60 * 1000);
    
    return data.filter(d => new Date(d.date) >= cutoff);
  },

  calculateAverage(values) {
    if (!values.length) return 0;
    return (values.reduce((a, b) => a + b, 0) / values.length).toFixed(2);
  },

  calculateGoalAchievement(data, goal) {
    if (!data.length) return 0;
    const achieved = data.filter(d => this.checkGoal('', d.value, goal) === '✅ 达标').length;
    return Math.round((achieved / data.length) * 100);
  },

  formatReport(report) {
    let formatted = `📊 健康报告 (${report.period})\n\n`;
    formatted += `生成时间：${new Date(report.date).toLocaleString('zh-CN')}\n\n`;
    
    for (const [metric, stats] of Object.entries(report.metrics)) {
      formatted += `## ${metric}\n`;
      formatted += `平均：${stats.average} | 最低：${stats.min} | 最高：${stats.max}\n`;
      formatted += `目标达成率：${stats.goalAchieved}%\n\n`;
    }
    
    formatted += `💡 建议：${this.generateAdvice(report.metrics)}`;
    
    return formatted;
  },

  generateAdvice(metrics) {
    const advice = [];
    
    if (metrics.sleep?.average < 7) {
      advice.push('睡眠时间不足，建议早点休息');
    }
    if (metrics.exercise?.average < 30) {
      advice.push('运动量不够，建议增加活动');
    }
    if (metrics.water?.average < 2000) {
      advice.push('饮水不足，记得多喝水');
    }
    
    return advice.length ? advice.join('；') : '继续保持良好习惯！';
  }
};
```
