## 📋 安装监控插件

```bash

wget http://cms-agent.oss-cn-beijing.aliyuncs.com/release/cms_agent_latest.tgz

# 2. 解压
tar -xzf cms_agent_latest.tgz
cd CmsAgent

# 3. 安装
sudo bash install.sh

# 4. 启动
sudo systemctl start CmsAgent
sudo systemctl enable CmsAgent

# 5. 检查状态
```

## 🤖 OpenClaw 集成

### 监控脚本

```bash
#!/bin/bash
# ~/scripts/server-monitor.sh

# 获取 CPU 使用率
CPU=$(top -bn1 | grep "Cpu(s)" | awk '{print $2}' | cut -d'%' -f1)

# 获取内存使用率
MEM=$(free | grep Mem | awk '{printf("%.2f"), $3/$2 * 100.0}')

# 获取磁盘使用率
DISK=$(df -h / | tail -1 | awk '{print $5}' | sed 's/%//')

# 告警阈值
CPU_THRESHOLD=80
MEM_THRESHOLD=85
DISK_THRESHOLD=80

# 检查并发送告警
if (( $(echo "$CPU > $CPU_THRESHOLD" | bc -l) )); then
  openclaw chat "⚠️ CPU 告警：当前使用率 ${CPU}%"
fi

if (( $(echo "$MEM > $MEM_THRESHOLD" | bc -l) )); then
  openclaw chat "⚠️ 内存告警：当前使用率 ${MEM}%"
fi

if (( $(echo "$DISK > $DISK_THRESHOLD" | bc -l) )); then
  openclaw chat "⚠️ 磁盘告警：当前使用率 ${DISK}%"
fi

# 记录日志
echo "$(date): CPU=${CPU}%, MEM=${MEM}%, DISK=${DISK}%" >> ~/monitor.log
```

### 定时执行

```json

        "command": "执行 ~/scripts/server-monitor.sh",
```
