### 郵件自動發送腳本 (示例)

```python
#!/usr/bin/env python3
# evomap_daily_summary.py

import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
import requests
import json
from datetime import datetime

def get_evo_data():
    """獲取 EvoMap 數據"""
    # API 調用
    response = requests.get(
        'https://evomap.ai/a2a/stats',
        headers={'Authorization': 'Bearer YOUR_TOKEN'}
    )
    return response.json()

def generate_report(data):
    """生成報告"""
    report = f"""
    主題：【EvoMap 日報】{datetime.now().strftime('%Y-%m-%d')}
    
    💰 今日收入：¥{data['today_income']}
    🏆 聲譽：{data['reputation']} (+{data['reputation_change']})
    📝 產出：文章 {data['articles']} 篇 | 視頻 {data['videos']} 個
    ✅ 任務：完成 {data['tasks_completed']} 個
    📊 粉絲：{data['followers']} (+{data['followers_change']})
    """
    return report

def send_email(report):
    """發送郵件"""
    msg = MIMEMultipart()
    msg['From'] = 'evomap@yourdomain.com'
    msg['To'] = 'yesimagine@gmail.com'
    msg['Subject'] = f"【EvoMap 日報】{datetime.now().strftime('%Y-%m-%d')}"
    
    msg.attach(MIMEText(report, 'plain', 'utf-8'))
    
    server = smtplib.SMTP('smtp.gmail.com', 587)
    server.starttls()
    server.login('your_email@gmail.com', 'your_password')
    server.send_message(msg)
    server.quit()

if __name__ == '__main__':
    data = get_evo_data()
    report = generate_report(data)
    send_email(report)
    print("日報已發送")
```
