## 🔄 自动化检查脚本

### 版本检查脚本

```python
#!/usr/bin/env python3
# monitoring/version_checker.py
"""Serper API 版本更新检查脚本"""

import requests
import json
from datetime import datetime
import os

class SerperVersionChecker:
    def __init__(self):
        self.api_base = "https://google.serper.dev"
        self.status_api = "https://status.serper.dev/api/v2"
        self.webhook_url = os.getenv("VERSION_CHECK_WEBHOOK")
    
    def check_api_version(self):
        """检查 API 版本信息"""
        try:
            # 通过 API 响应头获取版本信息
            response = requests.get(
                f"{self.api_base}/search",
                headers={"X-API-KEY": os.getenv("SERPER_API_KEY")},
                timeout=10
            )
            version = response.headers.get("X-API-Version", "unknown")
            return version
        except Exception as e:
            return f"error: {e}"
    
    def check_status_page(self):
        """检查状态页面公告"""
        try:
            response = requests.get(
                f"{self.status_api}/incidents/unresolved.json",
                timeout=10
            )
            data = response.json()
            return data.get("incidents", [])
        except Exception as e:
            return []
    
    def check_github_releases(self):
        """检查 GitHub Releases"""
        try:
            response = requests.get(
                "https://api.github.com/repos/langchain-ai/langchain/releases",
                timeout=10
            )
            data = response.json()
            # 过滤 Serper 相关更新
            serper_releases = [
                r for r in data[:10]
                if "serper" in r.get("name", "").lower() or
                   "serper" in r.get("body", "").lower()
            ]
            return serper_releases
        except Exception as e:
            return []
    
    def send_notification(self, title, message):
        """发送通知"""
        if not self.webhook_url:
            print(f"📢 {title}: {message}")
            return
        
        payload = {
            "title": title,
            "message": message,
            "timestamp": datetime.utcnow().isoformat()
        }
        
        try:
            requests.post(self.webhook_url, json=payload, timeout=10)
        except Exception as e:
            print(f"Failed to send notification: {e}")
    
    def run_check(self):
        """执行完整检查"""
        print(f"🔍 开始检查 Serper 更新 - {datetime.now()}")
        
        # 检查 API 版本
        version = self.check_api_version()
        print(f"  API 版本：{version}")
        
        # 检查状态页面
        incidents = self.check_status_page()
        if incidents:
            for incident in incidents:
                self.send_notification(
                    "🚨 服务状态更新",
                    f"{incident['name']}: {incident['status']}"
                )
        
        # 检查 GitHub Releases
        releases = self.check_github_releases()
        if releases:
            for release in releases:
                self.send_notification(
                    "📦 新版本发布",
                    f"{release['name']}: {release['html_url']}"
                )
        
        print("✅ 检查完成")

if __name__ == "__main__":
    checker = SerperVersionChecker()
    checker.run_check()
```
