### 依赖安全扫描

**自动化检查:**

```javascript
// scripts/check-dependencies.js
const { execSync } = require('child_process');

async function checkDependencies() {
  console.log('检查依赖安全...');
  
  // npm audit
  try {
    execSync('npm audit --json', { encoding: 'utf8' });
  } catch (error) {
    const audit = JSON.parse(error.stdout);
    const vulnerabilities = audit.vulnerabilities;
    
    if (vulnerabilities.high > 0 || vulnerabilities.critical > 0) {
      console.error('❌ 发现高危漏洞!');
      process.exit(1);
    }
  }
  
  // 检查过时依赖
  const outdated = execSync('npm outdated --json', { encoding: 'utf8' });
  console.log('过时依赖:', JSON.parse(outdated));
  
  console.log('✅ 依赖安全检查通过');
}

checkDependencies();
```

### 配置安全扫描

**敏感信息检查:**

```javascript
// scripts/check-secrets.js
const fs = require('fs');
const path = require('path');

const secretPatterns = [
  /password\s*[:=]\s*['"][^'"]+['"]/gi,
  /secret\s*[:=]\s*['"][^'"]+['"]/gi,
  /api[_-]?key\s*[:=]\s*['"][^'"]+['"]/gi,
  /token\s*[:=]\s*['"][^'"]+['"]/gi,
  /private[_-]?key/gi
];

function checkFile(filePath) {
  const content = fs.readFileSync(filePath, 'utf8');
  const issues = [];
  
  secretPatterns.forEach((pattern, index) => {
    const matches = content.match(pattern);
    if (matches) {
      issues.push({
        pattern: pattern.toString(),
        matches: matches,
        line: content.substring(0, matches.index).split('\n').length
      });
    }
  });
  
  return issues;
}

function scanDirectory(dir) {
  const files = fs.readdirSync(dir);
  const allIssues = [];
  
  files.forEach(file => {
    const filePath = path.join(dir, file);
    const stat = fs.statSync(filePath);
    
    if (stat.isDirectory() && !file.startsWith('.') && file !== 'node_modules') {
      allIssues.push(...scanDirectory(filePath));
    } else if (stat.isFile() && /\.(js|ts|json|env|yaml|yml)$/.test(file)) {
      const issues = checkFile(filePath);
      if (issues.length > 0) {
        allIssues.push({ file: filePath, issues });
      }
    }
  });
  
  return allIssues;
}

const issues = scanDirectory('.');
if (issues.length > 0) {
  console.error('❌ 发现敏感信息泄露:');
  console.error(JSON.stringify(issues, null, 2));
  process.exit(1);
} else {
  console.log('✅ 敏感信息检查通过');
}
```

// tests/security/injection.test.js
describe('注入安全测试', () => {
  test('SQL 注入防护', async () => {
    const maliciousInput = "' OR '1'='1";
    
    const response = await request(app)
      .get(`/api/tasks?search=${encodeURIComponent(maliciousInput)}`);

    const xssPayload = '<script>alert("XSS")</script>';

### 数据保护合规

**GDPR 检查清单:**

```javascript
// scripts/gdpr-check.js
const gdprChecklist = {
  consent: {
    description: '用户同意',
    items: [
      '明确的同意请求',
      '易于撤回同意',
      '同意记录保存'
    ]
  },
  dataRights: {
    description: '数据权利',
    items: [
      '访问权实现',
      '更正权实现',
      '删除权实现',
      '数据可携带性'
    ]
  },
  security: {
    description: '数据安全',
    items: [
      '加密存储',
      '加密传输',
      '访问控制',
      '审计日志'
    ]
  },
  breach: {
    description: '泄露处理',
    items: [
      '泄露检测',
      '72 小时通知',
      '影响评估',
      ' remediation'
    ]
  }
};

function checkCompliance() {
  const results = {};
  
  for (const [category, checks] of Object.entries(gdprChecklist)) {
    results[category] = {
      description: checks.description,
      passed: 0,
      total: checks.items.length,
      items: checks.items.map(item => ({
        item,
        status: 'pending'
      }))
    };
  }
  
  return results;
}

const compliance = checkCompliance();
console.log('GDPR 合规检查:', compliance);
```

### 隐私政策检查

```javascript
// scripts/privacy-check.js
const requiredSections = [
  '数据收集',
  '数据使用',
  '数据共享',
  '数据安全',
  '用户权利',
  '联系方式',
  '政策更新'
];

function checkPrivacyPolicy(policyText) {
  const missing = [];
  
  requiredSections.forEach(section => {
    if (!policyText.includes(section)) {
      missing.push(section);
    }
  });
  
  return {
    passed: missing.length === 0,
    missing,
    coverage: (requiredSections.length - missing.length) / requiredSections.length
  };
}
```
