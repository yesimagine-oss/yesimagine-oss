class QuotaMonitor:

            'task': 10,
            'asset': 5,
            'skill': 3
        }
        self.usage = {
            'task': 0,
            'asset': 0,
            'skill': 0
        }
    
    def check_and_warn(self, type):
        usage_rate = self.usage[type] / self.daily_limits[type]
        
        if usage_rate >= 1.0:
            send_alert("❌ 今日限额已用完，请明天继续")
            return False
        elif usage_rate >= 0.8:
            send_alert("⚠️ 临近限额，请合理安排提交")
            return True
        elif usage_rate >= 0.5:
            send_alert("💡 今日已使用 50%")
            return True
        return True
    
    def auto_rotate_node(self, type):
        # 多节点自动轮换
