```javascript
// ~/.openclaw/workspace/skills/multi-level-cache/index.js
const NodeCache = require('node-cache');
const Redis = require('ioredis');

// L1: 内存缓存 (5 分钟)
const l1Cache = new NodeCache({ stdTTL: 300 });

// L2: Redis 缓存 (30 分钟)
const l2Cache = new Redis();

module.exports = {
  async execute(context, params) {
    const cacheKey = this.generateKey(params.message);
    
    // 尝试 L1
    const l1Result = l1Cache.get(cacheKey);
    if (l1Result) {
      return { content: l1Result, cache: 'L1' };
    }
    
    // 尝试 L2
    const l2Result = await l2Cache.get(cacheKey);
    if (l2Result) {
      l1Cache.set(cacheKey, l2Result);
      return { content: l2Result, cache: 'L2' };
    }
    
    // 调用 AI
    const response = await this.callAI(params.message);
    
    // 存入缓存
    l1Cache.set(cacheKey, response);
    await l2Cache.set(cacheKey, response, 1800);
    
    return { content: response, cache: 'miss' };
  }
};
```

```javascript
const batchProcessor = {

  add(message) {
    return new Promise((resolve) => {
      this.queue.push({ message, resolve });
      
      if (!this.timer) {
        this.timer = setTimeout(() => this.process(), 100);
      }
    });

    batch.forEach((item, i) => item.resolve(results[i]));

  }
```

#### 1. 会话自动清理

```javascript
// 添加会话过期机制
class SessionManager {
  constructor() {
    this.sessions = new Map();
    this.ttl = 7 * 24 * 60 * 60 * 1000; // 7 天
    
    // 定期清理
    setInterval(() => this.cleanup(), 60 * 60 * 1000);
  }
  
  cleanup() {
    const now = Date.now();
    for (const [id, session] of this.sessions) {
      if (now - session.lastAccess > this.ttl) {
        this.sessions.delete(id);
      }
    }
  }
}
```
