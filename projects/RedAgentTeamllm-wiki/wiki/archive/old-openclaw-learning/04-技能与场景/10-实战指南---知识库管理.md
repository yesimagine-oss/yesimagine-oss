## 📝 知识库管理脚本

```javascript
// ~/.openclaw/workspace/skills/knowledge-manager/index.js
const fs = require('fs');
const path = require('path');

module.exports = {
  meta: {
    name: 'knowledge-manager',
    version: '1.0.0',
    description: '个人知识库管理技能'
  },

  config: {
    enabled: true,
    basePath: '~/knowledge-base',
    categories: ['tech', 'work', 'life', 'ideas'],
    autoTag: true
  },

  async execute(context, params) {
    const action = params?.action || 'list';
    
    switch(action) {
      case 'add':
        return await this.addNote(params);
      case 'organize':
        return await this.organizeInbox();
      case 'search':
        return await this.searchNotes(params.query);
      case 'review':
        return await this.getReviewItems();
      case 'map':
        return await this.generateKnowledgeMap();
      default:
        return await this.listNotes();
    }
  },

  async addNote(params) {
    const { title, content, category, tags } = params;
    const date = new Date().toISOString().split('T')[0];
    const filename = `${date}-${this.slugify(title)}.md`;
    
    const note = `---
title: ${title}
date: ${new Date().toISOString()}
category: ${category || 'inbox'}
tags: ${(tags || []).join(', ')}
---

# ${title}

${content}

---
_最后更新：${new Date().toLocaleString('zh-CN')}_
`;

    const filepath = path.join(
      this.config.basePath,
      category || 'inbox',
      filename
    );
    
    fs.writeFileSync(filepath, note, 'utf8');
    
    return {
      content: `✅ 笔记已添加\n\n标题：${title}\n分类：${category || 'inbox'}\n文件：${filename}`
    };
  },

  async organizeInbox() {
    const inboxPath = path.join(this.config.basePath, 'inbox');
    const files = fs.readdirSync(inboxPath);
    
    let organized = 0;
    for (const file of files) {
      if (!file.endsWith('.md')) continue;
      
      const content = fs.readFileSync(path.join(inboxPath, file), 'utf8');
      
      // AI 自动分类
      const category = await this.autoClassify(content);
      
      if (category !== 'inbox') {
        const newPath = path.join(this.config.basePath, category, file);
        fs.renameSync(path.join(inboxPath, file), newPath);
        organized++;
      }
    }
    
    return {
      content: `📁 整理完成\n\n整理文件：${organized} 个\n剩余待整理：${files.length - organized} 个`
    };
  },

  async searchNotes(query) {
    const results = [];
    
    for (const category of this.config.categories) {
      const catPath = path.join(this.config.basePath, category);
      if (!fs.existsSync(catPath)) continue;
      
      const files = fs.readdirSync(catPath);
      for (const file of files) {
        if (!file.endsWith('.md')) continue;
        
        const content = fs.readFileSync(path.join(catPath, file), 'utf8');
        if (content.toLowerCase().includes(query.toLowerCase())) {
          results.push({ category, file });
        }
      }
    }
    
    let summary = `🔍 搜索结果："${query}"\n\n`;
    if (results.length === 0) {
      summary += '未找到相关笔记';
    } else {
      results.slice(0, 10).forEach(r => {
        summary += `• [${r.category}] ${r.file}\n`;
      });
      if (results.length > 10) {
        summary += `\n... 还有 ${results.length - 10} 条结果`;
      }
    }
    
    return { content: summary };
  },

  async generateKnowledgeMap() {
    const stats = {};
    let totalNotes = 0;
    
    for (const category of this.config.categories) {
      const catPath = path.join(this.config.basePath, category);
      if (!fs.existsSync(catPath)) continue;
      
      const files = fs.readdirSync(catPath).filter(f => f.endsWith('.md'));
      stats[category] = files.length;
      totalNotes += files.length;
    }
    
    let map = `🗺️ 知识地图\n\n`;
    map += `总笔记数：${totalNotes}\n\n`;
    
    for (const [cat, count] of Object.entries(stats)) {
      const bar = '█'.repeat(Math.min(count, 20));
      map += `${cat}: ${bar} (${count})\n`;
    }
    
    // 最近更新的笔记
    map += `\n## 最近更新的笔记\n\n`;
    // ... 实现最近笔记列表
    
    return { content: map };
  },

  async getReviewItems() {
    // 实现定期回顾逻辑
    // 随机选择 5 篇旧笔记进行复习
    return { content: '📖 今日复习内容...\n\n(实现中)' };
  },

  slugify(text) {
    return text.toLowerCase()
      .replace(/[^\w\s-]/g, '')
      .replace(/\s+/g, '-')
      .replace(/-+/g, '-')
      .trim();
  },

  async autoClassify(content) {
    // 使用 AI 自动分类
    const prompt = `请判断以下内容属于哪个分类：tech, work, life, ideas

内容：${content.substring(0, 500)}...

只返回分类名称。`;
    
    // const result = await callAI(prompt);
    // return result.trim();
    return 'inbox'; // 临时返回
  }
};
```
