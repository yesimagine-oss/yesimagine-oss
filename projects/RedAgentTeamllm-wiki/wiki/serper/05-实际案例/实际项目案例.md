## 案例 1: AI 助手搜索增强

### 场景描述

**问题:** AI 助手缺乏实时信息获取能力

**解决方案:** 集成 Serper API 提供实时搜索

### 实现代码

```python
from langchain.utilities import SerperAPIWrapper
from langchain.agents import initialize_agent, Tool
from langchain.llms import OpenAI

# 初始化 Serper
serper = SerperAPIWrapper(
    serper_api_key="01529847d4aa3cf47b86ca87d28519110db06390"
)

# 创建搜索工具
search_tool = Tool(
    name="Search",
    func=serper.run,
    description="Useful for when you need to answer questions about current events or access up-to-date information."
)

# 初始化 Agent
llm = OpenAI(temperature=0)
agent = initialize_agent(
    tools=[search_tool],
    llm=llm,
    agent="zero-shot-react-description",
    verbose=True
)

# 使用示例
response = agent.run("What are the latest developments in AI agent technology?")
print(response)
```

## 案例 2: 竞品监控系统

### 场景描述

**问题:** 需要实时监控竞品动态

**解决方案:** 定时搜索竞品相关新闻和提及

### 实现代码

```python
import requests
import schedule
import time
from datetime import datetime

class CompetitorMonitor:
    def __init__(self, api_key, competitors):
        self.api_key = api_key
        self.competitors = competitors
        self.headers = {
            "X-API-KEY": api_key,
            "Content-Type": "application/json"
        }
    
    def search_competitor(self, competitor):
        """搜索竞品相关信息"""
        response = requests.post(
            "https://google.serper.dev/news",
            headers=self.headers,
            json={
                "q": competitor,
                "tbs": "qdr:d",  # 过去 1 天
                "num": 10
            }
        )
        return response.json()
    
    def monitor_all(self):
        """监控所有竞品"""
        results = {}
        for competitor in self.competitors:
            print(f"Monitoring {competitor}...")
            news = self.search_competitor(competitor)
            results[competitor] = news.get('news', [])
        
        # 保存结果
        self.save_results(results)
        return results
    
    def save_results(self, results):
        """保存监控结果"""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        filename = f"competitor_monitor_{timestamp}.json"
        
        import json
        with open(filename, 'w', encoding='utf-8') as f:
            json.dump(results, f, ensure_ascii=False, indent=2)
        
        print(f"Results saved to {filename}")
    
    def start(self, interval_hours=24):
        """启动监控"""
        print(f"Starting competitor monitor (interval: {interval_hours}h)")
        
        # 立即执行一次
        self.monitor_all()
        
        # 定时执行
        schedule.every(interval_hours).hours.do(self.monitor_all)
        
        while True:
            schedule.run_pending()
            time.sleep(60)

# 使用示例
if __name__ == "__main__":
    competitors = ["Competitor A", "Competitor B", "Competitor C"]
    
    monitor = CompetitorMonitor(
        api_key="01529847d4aa3cf47b86ca87d28519110db06390",
        competitors=competitors
    )
    
    monitor.start(interval_hours=24)  # 每天监控一次
```

- ✅ 每天自动收集竞品新闻
- ✅ 及时发现竞品动态
- ✅ 支持历史数据对比

## 案例 3: 市场调研工具

### 场景描述

**问题:** 需要快速收集行业市场数据

**解决方案:** 多维度搜索 + 数据分析

### 实现代码

```python
import requests
import pandas as pd
import matplotlib.pyplot as plt

class MarketResearch:
    def __init__(self, api_key):
        self.api_key = api_key
        self.headers = {
            "X-API-KEY": api_key,
            "Content-Type": "application/json"
        }
    
    def search_trends(self, topic):
        """搜索市场趋势"""
        response = requests.post(
            "https://google.serper.dev/search",
            headers=self.headers,
            json={
                "q": f"{topic} market trends 2026",
                "num": 20
            }
        )
        return response.json()
    
    def search_news(self, topic):
        """搜索行业新闻"""
        response = requests.post(
            "https://google.serper.dev/news",
            headers=self.headers,
            json={
                "q": f"{topic} industry news",
                "tbs": "qdr:w",  # 过去 1 周
                "num": 20
            }
        )
        return response.json()
    
    def analyze_sentiment(self, results):
        """简单情感分析"""
        positive_keywords = ['growth', 'increase', 'positive', 'success', 'opportunity']
        negative_keywords = ['decline', 'decrease', 'negative', 'failure', 'challenge']
        
        sentiments = {'positive': 0, 'negative': 0, 'neutral': 0}
        
        for result in results.get('organic', []):
            text = (result.get('title', '') + ' ' + result.get('snippet', '')).lower()
            
            if any(word in text for word in positive_keywords):
                sentiments['positive'] += 1
            elif any(word in text for word in negative_keywords):
                sentiments['negative'] += 1
            else:
                sentiments['neutral'] += 1
        
        return sentiments
    
    def generate_report(self, topic):
        """生成市场研究报告"""
        print(f"Researching {topic}...")
        
        # 收集数据
        trends = self.search_trends(topic)
        news = self.search_news(topic)
        
        # 分析情感
        sentiment = self.analyze_sentiment(trends)
        
        # 生成报告
        report = {
            'topic': topic,
            'total_results': len(trends.get('organic', [])),
            'news_count': len(news.get('news', [])),
            'sentiment': sentiment,
            'trends': trends.get('organic', [])[:10],
            'news': news.get('news', [])[:10]
        }
        
        # 可视化
        self.visualize_sentiment(sentiment)
        
        return report
    
    def visualize_sentiment(self, sentiment):
        """情感分析可视化"""
        labels = sentiment.keys()
        values = sentiment.values()
        
        plt.figure(figsize=(10, 6))
        plt.bar(labels, values, color=['green', 'red', 'gray'])
        plt.title('Market Sentiment Analysis')
        plt.ylabel('Number of Articles')
        plt.savefig('market_sentiment.png')
        print("Sentiment chart saved to market_sentiment.png")

# 使用示例
if __name__ == "__main__":
    research = MarketResearch("01529847d4aa3cf47b86ca87d28519110db06390")
    
    report = research.generate_report("AI agent")
    
    import json
    print(json.dumps(report, indent=2, ensure_ascii=False))
```

### 效果

- ✅ 自动生成市场研究报告
- ✅ 情感分析可视化
- ✅ 支持多个主题对比
- ✅ 节省人工调研时间 80%

## 案例 4: SEO 分析工具

### 场景描述

**问题:** 需要跟踪关键词排名

**解决方案:** 定期搜索关键词并记录排名

### 实现代码

```python
import requests
import sqlite3
from datetime import datetime

class SEOTracker:
    def __init__(self, api_key, db_path='seo_tracker.db'):
        self.api_key = api_key
        self.headers = {
            "X-API-KEY": api_key,
            "Content-Type": "application/json"
        }
        self.db_path = db_path
        self.init_db()
    
    def init_db(self):
        """初始化数据库"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS rankings (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                keyword TEXT NOT NULL,
                url TEXT NOT NULL,
                position INTEGER NOT NULL,
                tracked_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
            )
        ''')
        
        conn.commit()
        conn.close()
    
    def track_keyword(self, keyword, target_url):
        """跟踪关键词排名"""
        response = requests.post(
            "https://google.serper.dev/search",
            headers=self.headers,
            json={
                "q": keyword,
                "num": 100  # 获取前 100 个结果
            }
        )
        
        results = response.json()
        
        # 查找目标 URL 的排名
        position = None
        for i, result in enumerate(results.get('organic', []), 1):
            if target_url in result.get('link', ''):
                position = i
                break
        
        # 保存结果
        if position:
            self.save_ranking(keyword, target_url, position)
        
        return position
    
    def save_ranking(self, keyword, url, position):
        """保存排名记录"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute(
            "INSERT INTO rankings (keyword, url, position) VALUES (?, ?, ?)",
            (keyword, url, position)
        )
        
        conn.commit()
        conn.close()
    
    def get_ranking_history(self, keyword, url, days=30):
        """获取排名历史"""
        conn = sqlite3.connect(self.db_path)
        
        query = '''
            SELECT position, tracked_at
            FROM rankings
            WHERE keyword = ? AND url = ?
            AND tracked_at >= datetime('now', '-' || ? || ' days')
            ORDER BY tracked_at DESC
        '''
        
        df = pd.read_sql_query(query, conn, params=(keyword, url, days))
        conn.close()
        
        return df
    
    def track_all_keywords(self, keywords_dict):
        """跟踪所有关键词"""
        results = {}
        for keyword, url in keywords_dict.items():
            position = self.track_keyword(keyword, url)
            results[keyword] = position
            print(f"{keyword}: #{position if position else 'Not found'}")
        return results

# 使用示例
if __name__ == "__main__":
    tracker = SEOTracker("01529847d4aa3cf47b86ca87d28519110db06390")
    
    # 跟踪关键词
    keywords = {
        "AI agent platform": "https://mywebsite.com",
        "AI automation tool": "https://mywebsite.com/products",
        "machine learning API": "https://mywebsite.com/api"
    }
    
    results = tracker.track_all_keywords(keywords)
    
    # 获取排名历史
    history = tracker.get_ranking_history("AI agent platform", "https://mywebsite.com")
    print(history)
```

- ✅ 自动跟踪关键词排名
- ✅ 历史记录可查询
- ✅ 支持批量跟踪
