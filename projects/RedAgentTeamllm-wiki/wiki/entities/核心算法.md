## 1. 信号匹配算法

### 算法实现

```javascript
/**
 * 匹配模式到信号列表
 * @param {string} pattern - 匹配模式
 * @param {string[]} signals - 信号列表
 * @returns {boolean} - 是否匹配
 */
function matchPatternToSignals(pattern, signals) {
  if (!pattern || !signals || signals.length === 0) return false;
  
  const p = String(pattern);
  const sig = signals.map(s => String(s));

  // 1. 正则表达式匹配：/body/flags
  const regexLike = p.length >= 2 && p.startsWith('/') && p.lastIndexOf('/') > 0;
  if (regexLike) {
    const lastSlash = p.lastIndexOf('/');
    const body = p.slice(1, lastSlash);
    const flags = p.slice(lastSlash + 1);
    try {
      const re = new RegExp(body, flags || 'i');
      return sig.some(s => re.test(s));
    } catch (e) {
      // 正则失败，降级到子字符串匹配
    }
  }

  // 2. 多语言别名：en_term|zh_term|ja_term
  if (p.includes('|') && !p.startsWith('/')) {
    const branches = p.split('|')
      .map(b => b.trim().toLowerCase())
      .filter(Boolean);
    return branches.some(needle => 
      sig.some(s => s.toLowerCase().includes(needle))
    );
  }

  // 3. 子字符串匹配（不区分大小写）
  const needle = p.toLowerCase();
  return sig.some(s => s.toLowerCase().includes(needle));
}
```

### 匹配示例

```javascript
// 正则匹配
matchPatternToSignals('/react_.*/i', ['react_rerender', 'vue_update'])
// → true (匹配 react_rerender)

// 多语言匹配
matchPatternToSignals('performance|性能|パフォーマンス', ['前端性能优化'])
// → true (匹配中文"性能")

// 子字符串匹配
matchPatternToSignals('error', ['RuntimeError', 'Success'])
// → true (匹配 RuntimeError)
```

### 时间复杂度

- **正则匹配:** O(n × m)，n=信号数，m=模式长度
- **多语言匹配:** O(n × k × l)，k=语言数，l=平均长度
- **子字符串匹配:** O(n × l)

**优化策略:**
- 缓存已匹配结果
- 优先使用精确匹配
- 限制信号数量（建议 5-10 个）

## 2. Gene 评分算法

### 算法实现

```javascript
/**
 * 为 Gene 评分
 * @param {Object} gene - Gene 对象
 * @param {string[]} signals - 当前信号列表
 * @returns {number} - 评分 (0+)
 */
function scoreGene(gene, signals) {
  if (!gene || gene.type !== 'Gene') return 0;
  
  const patterns = Array.isArray(gene.signals_match) 
    ? gene.signals_match 
    : [];
    
  if (patterns.length === 0) return 0;
  
  let score = 0;
  for (const pat of patterns) {
    if (matchPatternToSignals(pat, signals)) {
      score += 1;
    }
  }
  
  // 特殊处理：distilled 基因有加成
  const DISTILLED_PREFIX = 'gene_distilled_';
  if (score > 0 && gene.id && 
      String(gene.id).startsWith(DISTILLED_PREFIX)) {
    score *= 0.8;  // 轻微降级，优先使用非蒸馏基因
  }
  
  return score;
}
```

### 评分示例

```javascript
const gene = {
  type: 'Gene',
  id: 'gene_react_optimization',
  signals_match: [
    'react_rerender',
    'react_memo',
    'useMemo',
    'performance'
  ]
};

const signals = ['react_rerender', 'performance_optimization'];

const score = scoreGene(gene, signals);
// → 2 (匹配 react_rerender 和 performance)
```

### 选择策略

```javascript
function selectGene(genes, signals, opts) {
  // 1. 评分所有 Genes

    .map(g => ({ gene: g, score: scoreGene(g, signals) }))

    .sort((a, b) => b.score - a.score);

  if (useDrift && scored.length > 1) {

    const top3 = scored.slice(0, 3);
    return weightedRandom(top3);
  }
  
  // 3. 返回最佳
  return scored[0]?.gene || null;
}

## 3. 漂变强度算法

### 理论基础

基于种群遗传学：
- **遗传漂变 (Genetic Drift):** 小种群中随机性更强
- **有效种群大小 (Ne):** 影响漂变强度
- **公式:** intensity = 1 / √Ne

### 算法实现

```javascript
/**
 * 计算漂变强度
 * @param {Object} opts - 选项
 * @returns {number} - 漂变强度 (0-1)
 */
function computeDriftIntensity(opts) {
  const driftEnabled = !!(opts && opts.driftEnabled);
  
  // 有效种群大小
  const effectivePopulationSize = opts && Number.isFinite(Number(opts.effectivePopulationSize))
    ? Number(opts.effectivePopulationSize)
    : null;
  
  // 基因池大小（后备）
  const genePoolSize = opts && Number.isFinite(Number(opts.genePoolSize))
    ? Number(opts.genePoolSize)
    : null;
  
  const ne = effectivePopulationSize || genePoolSize || null;
  
  if (driftEnabled) {
    // 显式启用漂变：中高强度
    return ne && ne > 1 
      ? Math.min(1, 1 / Math.sqrt(ne) + 0.3) 
      : 0.7;
  }
  
  if (ne != null && ne > 0) {
    // 种群规模依赖：小种群=高漂变
    // Ne=1: 1.0 (纯漂变), Ne=25: 0.2, Ne=100: 0.1
    return Math.min(1, 1 / Math.sqrt(ne));
  }
  
  return 0; // 无漂变，纯选择
}
```

### 漂变强度对照表

| Ne (种群大小) | 漂变强度 | 说明 |
|--------------|---------|------|
| 1 | 1.0 | 纯漂变（完全随机） |
| 4 | 0.5 | 高漂变 |
| 10 | 0.32 | 中高漂变 |
| 25 | 0.2 | 中等漂变 |
| 100 | 0.1 | 低漂变 |
| 1000 | 0.03 | 极低漂变 |
| ∞ | 0 | 纯选择（无随机性） |

### 应用场景

**高漂变 (0.5-1.0):**
- 新系统，Gene 池小
- 探索阶段，需要多样性
- 避免局部最优

**低漂变 (0-0.2):**
- 成熟系统，Gene 池大
- 稳定阶段，需要一致性
- 已有高质量 Genes

function computeGDI(asset) {
  let score = 0;
  
  // 1. 结构完整性 (25 分)
  score += evaluateStructuralIntegrity(asset) * 25;
  
  // 2. 语义质量 (25 分)
  score += evaluateSemanticQuality(asset) * 25;
  
  // 3. 信号特异性 (20 分)
  score += evaluateSignalSpecificity(asset) * 20;
  
  // 4. 策略质量 (20 分)
  score += evaluateStrategyQuality(asset) * 20;
  
  // 5. 验证强度 (10 分)
  score += evaluateValidationStrength(asset) * 10;
  
  return score;  // 0-100
}

### 各维度评估

#### 结构完整性

```javascript
function evaluateStructuralIntegrity(asset) {
  const required = ['type', 'id', 'asset_id'];
  const optional = ['summary', 'description', 'tags'];
  
  let score = 1.0;
  
  // 缺少必填字段
  for (const field of required) {
    if (!asset[field]) score -= 0.2;
  }
  
  // 类型错误
  if (typeof asset.id !== 'string') score -= 0.1;
  if (typeof asset.asset_id !== 'string') score -= 0.1;
  
  // asset_id 验证
  if (!asset.asset_id.startsWith('sha256:')) score -= 0.2;
  
  return Math.max(0, score);
}
```

#### 语义质量

```javascript
function evaluateSemanticQuality(asset) {
  let score = 1.0;
  
  // summary 长度
  if (asset.summary) {
    const len = asset.summary.length;
    if (len < 50) score -= 0.3;       // 太短
    else if (len > 200) score -= 0.1;  // 太长
  } else {
    score -= 0.5;
  }
  
  // 代码示例
  if (asset.content && asset.content.includes('```')) {
    score += 0.1;  // 有代码示例加分
  }
  
  // 注释质量
  if (asset.content && asset.content.includes('//')) {
    score += 0.05;  // 有注释加分
  }
  
  return Math.min(1.0, Math.max(0, score));
}
```

## 📊 算法性能

### 优化技巧

#### 1. 缓存策略

```javascript
const cache = new Map();

function cachedScoreGene(geneId, signals) {
  const key = `${geneId}:${signals.join(',')}`;
  
  if (cache.has(key)) {
    return cache.get(key);
  }
  
  const score = scoreGene(genes.find(g => g.id === geneId), signals);
  cache.set(key, score);
  
  // 限制缓存大小
  if (cache.size > 1000) {
    const firstKey = cache.keys().next().value;
    cache.delete(firstKey);
  }
  
  return score;
}
```

#### 2. 预计算

```javascript
// 预计算 signal 索引
const signalIndex = new Map();

function buildSignalIndex(genes) {
  genes.forEach(gene => {
    gene.signals_match.forEach(signal => {
      if (!signalIndex.has(signal)) {
        signalIndex.set(signal, []);
      }
      signalIndex.get(signal).push(gene.id);
    });
  });
}

// 快速查找
function findMatchingGenes(signals) {
  const geneIds = new Set();
  signals.forEach(signal => {
    const matches = signalIndex.get(signal) || [];
    matches.forEach(id => geneIds.add(id));
  });
  return Array.from(geneIds);
}
```

#### 3. 并行计算

```javascript
async function parallelScoreGenes(genes, signals) {
  const chunks = chunkArray(genes, 100);  // 每批 100 个
  
  const results = await Promise.all(
    chunks.map(chunk => 
      Promise.all(chunk.map(gene => scoreGene(gene, signals)))
    )
  );
  
  return results.flat();
}
