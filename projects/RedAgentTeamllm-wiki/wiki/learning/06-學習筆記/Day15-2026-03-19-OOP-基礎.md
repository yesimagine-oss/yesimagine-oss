## 1️⃣ 類與實例

### 1.1 類的定義

```python
class MyClass:
    """這是一個簡單的類示例"""
    
    # 類屬性 (所有實例共享)
    class_attribute = "我是類屬性"
    
    def __init__(self, name, value):
        """初始化方法 (構造函數)"""
        # 實例屬性 (每個實例獨立)
        self.name = name
        self.value = value
    
    def instance_method(self):
        """實例方法"""
        return f"我是 {self.name}, 值是 {self.value}"
    
    @classmethod
    def class_method(cls):
        """類方法 - 可以訪問類屬性"""
        return f"類屬性：{cls.class_attribute}"
    
    @staticmethod
    def static_method():
        """靜態方法 - 不訪問類或實例屬性"""
        return "我是靜態方法"

# 創建實例
obj1 = MyClass("對象 1", 100)
obj2 = MyClass("對象 2", 200)

# 訪問屬性和方法
print(obj1.instance_method())  # 我是對象 1, 值是 100
print(obj2.instance_method())  # 我是對象 2, 值是 200
print(MyClass.class_attribute)  # 我是類屬性
print(obj1.class_method())      # 類屬性：我是類屬性
print(MyClass.static_method())  # 我是靜態方法
```

### 1.2 實例屬性 vs 類屬性

```python
class Student:
    # 類屬性 - 所有學生共享
    school = "Python 學校"
    student_count = 0
    
    def __init__(self, name, age):
        # 實例屬性 - 每個學生獨立
        self.name = name
        self.age = age
        # 修改類屬性
        Student.student_count += 1
    
    def introduce(self):
        return f"我是 {self.name}, 今年 {self.age} 歲，來自 {self.school}"

# 創建學生
s1 = Student("小明", 18)
s2 = Student("小紅", 19)

print(s1.introduce())  # 我是小明，今年 18 歲，來自 Python 學校
print(s2.introduce())  # 我是小紅，今年 19 歲，來自 Python 學校
print(f"學生總數：{Student.student_count}")  # 學生總數：2

# 修改類屬性會影響所有實例
Student.school = "Python 大學"
print(s1.school)  # Python 大學
print(s2.school)  # Python 大學
```

## 2️⃣ 繼承與多態

### 2.1 單繼承

```python
class Animal:
    """動物基類"""
    
    def __init__(self, name):
        self.name = name
    
    def speak(self):
        return "動物發出聲音"
    
    def move(self):
        return "動物移動"

class Dog(Animal):
    """狗類 - 繼承自動物類"""
    
    def speak(self):
        """重寫父類方法"""
        return f"{self.name} 說：汪汪汪！"
    
    def fetch(self):
        """狗特有的方法"""
        return f"{self.name} 去撿球了"

class Cat(Animal):
    """貓類 - 繼承自動物類"""
    
    def speak(self):
        """重寫父類方法"""
        return f"{self.name} 說：喵喵喵！"
    
    def climb(self):
        """貓特有的方法"""
        return f"{self.name} 爬樹了"

# 使用
dog = Dog("小黑")
cat = Cat("小白")

print(dog.speak())  # 小黑說：汪汪汪！
print(dog.move())   # 動物移動 (繼承自父類)
print(dog.fetch())  # 小黑去撿球了

print(cat.speak())  # 小白說：喵喵喵！
print(cat.move())   # 動物移動 (繼承自父類)
print(cat.climb())  # 小白爬樹了
```

### 2.2 多態 (Polymorphism)

```python
class Animal:
    def __init__(self, name):
        self.name = name
    
    def speak(self):
        raise NotImplementedError("子類必須實現此方法")

class Dog(Animal):
    def speak(self):
        return f"{self.name}: 汪汪！"

class Cat(Animal):
    def speak(self):
        return f"{self.name}: 喵喵！"

class Duck(Animal):
    def speak(self):
        return f"{self.name}: 嘎嘎！"

# 多態示例 - 同一接口，不同實現
animals = [Dog("小黑"), Cat("小白"), Duck("小黃")]

for animal in animals:
    print(animal.speak())
# 輸出:
# 小黑：汪汪！
# 小白：喵喵！
# 小黃：嘎嘎！
```

### 2.3 多重繼承

```python
class Flyable:
    def fly(self):
        return "我會飛"

class Swimmable:
    def swim(self):
        return "我會游泳"

class Duck(Flyable, Swimmable):
    def __init__(self, name):
        self.name = name
    
    def quack(self):
        return f"{self.name}: 嘎嘎！"

duck = Duck("唐老鴨")
print(duck.fly())    # 我會飛 (來自 Flyable)
print(duck.swim())   # 我會游泳 (來自 Swimmable)
print(duck.quack())  # 唐老鴨：嘎嘎！
```

## 3️⃣ 封裝與抽象

### 3.1 封裝 - 私有屬性

```python
class BankAccount:
    def __init__(self, owner, balance):
        self.owner = owner
        self.__balance = balance  # 私有屬性 (雙下劃線)
    
    def deposit(self, amount):
        """存款"""
        if amount > 0:
            self.__balance += amount
            return f"存入 {amount}, 餘額：{self.__balance}"
        return "金額必須大於 0"
    
    def withdraw(self, amount):
        """取款"""
        if 0 < amount <= self.__balance:
            self.__balance -= amount
            return f"取出 {amount}, 餘額：{self.__balance}"
        return "餘額不足"
    
    def get_balance(self):
        """查詢餘額 (公開接口)"""
        return f"{self.owner} 的餘額：{self.__balance}"

# 使用
account = BankAccount("小明", 1000)
print(account.get_balance())  # 小明的餘額：1000
print(account.deposit(500))   # 存入 500, 餘額：1500
print(account.withdraw(200))  # 取出 200, 餘額：1300

# 無法直接訪問私有屬性
# print(account.__balance)  # ❌ 會報錯：AttributeError

# 但可以通过名稱重整訪問 (不推薦)
print(account._BankAccount__balance)  # ⚠️ 不推薦這樣做
```

### 3.2 抽象基類

```python
from abc import ABC, abstractmethod

class Shape(ABC):
    """抽象基類 - 不能直接實例化"""
    
    @abstractmethod
    def area(self):
        """計算面積 - 子類必須實現"""
        pass
    
    @abstractmethod
    def perimeter(self):
        """計算周長 - 子類必須實現"""
        pass

class Rectangle(Shape):
    def __init__(self, width, height):
        self.width = width
        self.height = height
    
    def area(self):
        return self.width * self.height
    
    def perimeter(self):
        return 2 * (self.width + self.height)

class Circle(Shape):
    def __init__(self, radius):
        self.radius = radius
    
    def area(self):
        return 3.14159 * self.radius ** 2
    
    def perimeter(self):
        return 2 * 3.14159 * self.radius

# 使用
rect = Rectangle(5, 3)
circle = Circle(4)

print(f"矩形面積：{rect.area()}")        # 15
print(f"矩形周長：{rect.perimeter()}")    # 16
print(f"圓形面積：{circle.area():.2f}")    # 50.27
print(f"圓形周長：{circle.perimeter():.2f}") # 25.13

# 不能實例化抽象類
# shape = Shape()  # ❌ 會報錯：TypeError
```

## 4️⃣ 特殊方法 (Magic Methods)

### 4.1 常用特殊方法

```python
class Vector:
    def __init__(self, x, y):
        self.x = x
        self.y = y
    
    def __str__(self):
        """字符串表示 (print 時調用)"""
        return f"Vector({self.x}, {self.y})"
    
    def __repr__(self):
        """開發者字符串表示"""
        return f"Vector(x={self.x}, y={self.y})"
    
    def __add__(self, other):
        """加法運算"""
        return Vector(self.x + other.x, self.y + other.y)
    
    def __sub__(self, other):
        """減法運算"""
        return Vector(self.x - other.x, self.y - other.y)
    
    def __mul__(self, scalar):
        """數乘運算"""
        return Vector(self.x * scalar, self.y * scalar)
    
    def __eq__(self, other):
        """等於運算"""
        return self.x == other.x and self.y == other.y
    
    def __len__(self):
        """長度"""
        return int((self.x ** 2 + self.y ** 2) ** 0.5)
    
    def __getitem__(self, index):
        """索引訪問"""
        if index == 0:
            return self.x
        elif index == 1:
            return self.y
        raise IndexError("索引超出範圍")

# 使用
v1 = Vector(3, 4)
v2 = Vector(1, 2)

print(v1)           # Vector(3, 4) - 調用 __str__
print(repr(v1))     # Vector(x=3, y=4) - 調用 __repr__
print(v1 + v2)      # Vector(4, 6) - 調用 __add__
print(v1 - v2)      # Vector(2, 2) - 調用 __sub__
print(v1 * 2)       # Vector(6, 8) - 調用 __mul__
print(v1 == v2)     # False - 調用 __eq__
print(len(v1))      # 5 - 調用 __len__
print(v1[0])        # 3 - 調用 __getitem__
```

### 4.2 上下文管理器特殊方法

```python
class FileManager:
    def __init__(self, filename, mode):
        self.filename = filename
        self.mode = mode
        self.file = None
    
    def __enter__(self):
        """進入上下文時調用"""
        print(f"打開文件：{self.filename}")
        self.file = open(self.filename, self.mode)
        return self.file
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        """退出上下文時調用"""
        print(f"關閉文件：{self.filename}")
        self.file.close()
        # 返回 True 可以吞掉異常，False 或 None 會讓異常繼續傳播
        return False

# 使用
with FileManager("test.txt", "w") as f:
    f.write("Hello, World!")
# 自動關閉文件
```

### 練習 1: 創建一個圖書管理系統

```python
class Book:
    def __init__(self, title, author, isbn):
        self.title = title
        self.author = author
        self.isbn = isbn
        self.is_borrowed = False
    
    def __str__(self):
        status = "已借出" if self.is_borrowed else "可借閱"
        return f"《{self.title}》- {self.author} [{status}]"
    
    def borrow(self):
        if not self.is_borrowed:
            self.is_borrowed = True
            return True
        return False
    
    def return_book(self):
        self.is_borrowed = False

class Library:
    def __init__(self, name):
        self.name = name
        self.books = []
    
    def add_book(self, book):
        self.books.append(book)
        return f"已添加：{book.title}"
    
    def find_book(self, title):
        for book in self.books:
            if book.title == title:
                return book
        return None
    
    def list_books(self):
        return [str(book) for book in self.books]

# 測試
library = Library("市立圖書館")
book1 = Book("Python 編程", "Guido", "978-0001")
book2 = Book("數據結構", "Knuth", "978-0002")

library.add_book(book1)
library.add_book(book2)

print(library.list_books())
```

### 練習 2: 實現一個計數器裝飾器

```python
from functools import wraps

def count_calls(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        wrapper.call_count += 1
        print(f"第 {wrapper.call_count} 次調用 {func.__name__}")
        return func(*args, **kwargs)
    wrapper.call_count = 0
    return wrapper

@count_calls
def greet(name):
    print(f"你好，{name}！")

greet("小明")
greet("小紅")
greet("小剛")
print(f"總共調用：{greet.call_count} 次")
```
