### 1.1 完整的裝飾器庫

```python
"""
decorators.py - Python 裝飾器工具庫
"""

from functools import wraps, lru_cache
import time
import logging
import traceback
from typing import Callable, Any, Optional

# 配置日誌
logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


# ==================== 基礎裝飾器 ====================

def log_calls(func: Callable) -> Callable:
    """日誌裝飾器 - 記錄函數調用"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        logger.info(f"調用 {func.__name__}, 參數：{args}, {kwargs}")
        try:
            result = func(*args, **kwargs)
            logger.info(f"{func.__name__} 返回：{result}")
            return result
        except Exception as e:
            logger.error(f"{func.__name__} 異常：{e}")
            raise
    return wrapper


def timing_decorator(func: Callable) -> Callable:
    """計時裝飾器 - 記錄函數執行時間"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.time()
        result = func(*args, **kwargs)
        end = time.time()
        elapsed = end - start
        logger.info(f"{func.__name__} 耗時：{elapsed:.4f}秒")
        return result
    return wrapper


def retry(max_attempts: int = 3, delay: float = 1.0, 
          exceptions: tuple = (Exception,)):
    """重試裝飾器"""
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            last_exception = None
            for attempt in range(max_attempts):
                try:
                    return func(*args, **kwargs)
                except exceptions as e:
                    last_exception = e
                    if attempt < max_attempts - 1:
                        logger.warning(
                            f"重試 {attempt + 1}/{max_attempts}: {e}"
                        )
                        time.sleep(delay)
            raise last_exception
        return wrapper
    return decorator


# ==================== 性能裝飾器 ====================

def lru_cache_decorator(maxsize: int = 128):
    """緩存裝飾器 - 包裝 lru_cache"""
    def decorator(func: Callable) -> Callable:
        return lru_cache(maxsize=maxsize)(func)
    return decorator


def memoize(func: Callable) -> Callable:
    """記憶化裝飾器 - 簡單緩存"""
    cache = {}
    
    @wraps(func)
    def wrapper(*args):
        if args in cache:
            return cache[args]
        result = func(*args)
        cache[args] = result
        return result
    
    return wrapper


# ==================== 驗證裝飾器 ====================

def validate_types(**expected_types):
    """類型驗證裝飾器"""
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            import inspect
            sig = inspect.signature(func)
            bound = sig.bind(*args, **kwargs)
            bound.apply_defaults()
            
            for param_name, expected_type in expected_types.items():
                if param_name in bound.arguments:
                    value = bound.arguments[param_name]
                    if not isinstance(value, expected_type):
                        raise TypeError(
                            f"{func.__name__}: {param_name} 必須是 "
                            f"{expected_type.__name__}, 實際是 "
                            f"{type(value).__name__}"
                        )
            return func(*args, **kwargs)
        return wrapper
    return decorator


def validate_range(**ranges):
    """數值範圍驗證裝飾器"""
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            import inspect
            sig = inspect.signature(func)
            bound = sig.bind(*args, **kwargs)
            bound.apply_defaults()
            
            for param_name, (min_val, max_val) in ranges.items():
                if param_name in bound.arguments:
                    value = bound.arguments[param_name]
                    if not (min_val <= value <= max_val):
                        raise ValueError(
                            f"{func.__name__}: {param_name} 必須在 "
                            f"[{min_val}, {max_val}] 範圍內，實際是 {value}"
                        )
            return func(*args, **kwargs)
        return wrapper
    return decorator


# ==================== 流程控制裝飾器 ====================

def timeout(seconds: float):
    """超時裝飾器"""
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            import signal
            
            def timeout_handler(signum, frame):
                raise TimeoutError(f"{func.__name__} 超時 ({seconds}秒)")
            
            old_handler = signal.signal(signal.SIGALRM, timeout_handler)
            signal.alarm(int(seconds))
            
            try:
                return func(*args, **kwargs)
            finally:
                signal.alarm(0)
                signal.signal(signal.SIGALRM, old_handler)
        return wrapper
    return decorator


def rate_limit(calls: int, period: float):
    """限速裝飾器"""
    def decorator(func: Callable) -> Callable:
        call_times = []
        
        @wraps(func)
        def wrapper(*args, **kwargs):
            nonlocal call_times
            now = time.time()
            
            # 清除過期的調用記錄
            call_times = [t for t in call_times if now - t < period]
            
            if len(call_times) >= calls:
                wait_time = period - (now - call_times[0])
                logger.warning(f"限速：等待 {wait_time:.2f}秒")
                time.sleep(wait_time)
            
            call_times.append(now)
            return func(*args, **kwargs)
        return wrapper
    return decorator


# ==================== 異常處理裝飾器 ====================

def suppress_exceptions(*exceptions, default=None):
    """抑制異常裝飾器"""
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            try:
                return func(*args, **kwargs)
            except exceptions as e:
                logger.warning(f"{func.__name__} 抑制異常：{e}")
                return default
        return wrapper
    return decorator


def handle_exceptions(handler: Callable[[Exception], Any]):
    """自定義異常處理裝飾器"""
    def decorator(func: Callable) -> Callable:
        @wraps(func)
        def wrapper(*args, **kwargs):
            try:
                return func(*args, **kwargs)
            except Exception as e:
                logger.error(f"{func.__name__} 異常：{e}")
                traceback.print_exc()
                return handler(e)
        return wrapper
    return decorator


# ==================== 類裝飾器 ====================

def singleton(cls):
    """單例裝飾器"""
    instances = {}
    
    @wraps(cls)
    def get_instance(*args, **kwargs):
        if cls not in instances:
            instances[cls] = cls(*args, **kwargs)
        return instances[cls]
    
    return get_instance


def dataclass_repr(cls):
    """自动生成 __repr__"""
    def __repr__(self):
        attrs = ', '.join(f"{k}={v!r}" for k, v in self.__dict__.items())
        return f"{cls.__name__}({attrs})"
    
    cls.__repr__ = __repr__
    return cls


# ==================== 調試裝飾器 ====================

def debug(func: Callable) -> Callable:
    """調試裝飾器 - 打印參數和返回值"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        args_repr = [repr(a) for a in args]
        kwargs_repr = [f"{k}={v!r}" for k, v in kwargs.items()]
        signature = ", ".join(args_repr + kwargs_repr)
        
        print(f"調用 {func.__name__}({signature})")
        result = func(*args, **kwargs)
        print(f"返回 {result!r}")
        
        return result
    return wrapper


def profile(func: Callable) -> Callable:
    """性能分析裝飾器"""
    @wraps(func)
    def wrapper(*args, **kwargs):
        import cProfile
        import pstats
        import io
        
        pr = cProfile.Profile()
        pr.enable()
        
        result = func(*args, **kwargs)
        
        pr.disable()
        s = io.StringIO()
        ps = pstats.Stats(pr, stream=s).sort_stats('cumulative')
        ps.print_stats(10)
        
        print(f"=== {func.__name__} 性能分析 ===")
        print(s.getvalue())
        
        return result
    return wrapper


# ==================== 導出 ====================

__all__ = [
    'log_calls',
    'timing_decorator',
    'retry',
    'lru_cache_decorator',
    'memoize',
    'validate_types',
    'validate_range',
    'timeout',
    'rate_limit',
    'suppress_exceptions',
    'handle_exceptions',
    'singleton',
    'dataclass_repr',
    'debug',
    'profile',
]
```

```python
from decorators import (
    log_calls, timing_decorator, retry, validate_types, 
    suppress_exceptions, memoize

# 多個裝飾器組合
@log_calls
@timing_decorator
@retry(max_attempts=3, delay=0.5)
@validate_types(name=str, age=int)
@suppress_exceptions(ValueError, default=None)
def create_user(name, age):
    """創建用戶"""
    if age < 0 or age > 150:
        raise ValueError("年齡無效")
    return {"name": name, "age": age}

user = create_user("小明", 18)

invalid = create_user("小紅", -5)  # 不會報錯，返回 None
```

### 2.2 裝飾器類

```python
from functools import wraps
import time

class DecoratorPipeline:
    """裝飾器管道"""
    def __init__(self):
        self.decorators = []
    
    def add(self, decorator):
        """添加裝飾器"""
        self.decorators.append(decorator)
        return self
    
    def decorate(self, func):
        """應用所有裝飾器"""
        for decorator in reversed(self.decorators):
            func = decorator(func)
        return func
    
    def __call__(self, func):
        return self.decorate(func)

# 使用
pipeline = (DecoratorPipeline()
    .add(log_calls)
    .add(timing_decorator)
    .add(retry(max_attempts=3)))

@pipeline
def process_data(data):
    return data * 2

result = process_data(5)
```

```python
def conditional_decorator(condition, decorator):
    """條件裝飾器"""
    def wrapper(func):
        if condition:
            return decorator(func)
        return func
    return wrapper

# 使用
DEBUG = True

@conditional_decorator(DEBUG, log_calls)
@conditional_decorator(DEBUG, timing_decorator)
def my_function(x):
    return x * 2

# 生產環境
DEBUG = False

@conditional_decorator(DEBUG, log_calls)  # 不會應用
```

### 3.1 API 認證裝飾器

```python
from functools import wraps
import jwt
import hashlib

def require_auth(secret_key):
    """API 認證裝飾器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            # 從參數中獲取 token
            token = kwargs.get('token') or \
                    kwargs.get('headers', {}).get('Authorization')
            
            if not token:
                raise PermissionError("缺少認證令牌")
            
            try:
                # 驗證 token
                payload = jwt.decode(token, secret_key, algorithms=['HS256'])
                kwargs['user'] = payload
                return func(*args, **kwargs)
            except jwt.ExpiredSignatureError:
                raise PermissionError("令牌已過期")
            except jwt.InvalidTokenError:
                raise PermissionError("無效令牌")
        return wrapper
    return decorator


def require_permission(permission):
    """權限檢查裝飾器"""
    def decorator(func):
        @wraps(func)
        @require_auth('secret-key')
        def wrapper(*args, **kwargs):
            user = kwargs.get('user', {})
            permissions = user.get('permissions', [])
            
            if permission not in permissions:
                raise PermissionError(
                    f"權限不足：需要 {permission}"
                )
            
            return func(*args, **kwargs)
        return wrapper
    return decorator


# 使用示例
@require_permission('admin')
def delete_user(user_id, **kwargs):
    """刪除用戶 - 需要管理員權限"""
    return {"status": "deleted", "user_id": user_id}

@require_permission('read')
def get_user(user_id, **kwargs):
    """獲取用戶 - 需要讀取權限"""
    return {"user_id": user_id, "name": "小明"}
```

### 3.2 API 限流裝飾器

```python
from functools import wraps
import time
from collections import defaultdict

class RateLimiter:
    """API 限流器"""
    def __init__(self, max_calls, period):
        self.max_calls = max_calls
        self.period = period
        self.calls = defaultdict(list)
    
    def __call__(self, func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            # 使用 IP 或用戶 ID 作為 key
            key = kwargs.get('ip', 'default')
            
            now = time.time()
            
            # 清除過期記錄
            self.calls[key] = [
                t for t in self.calls[key] 
                if now - t < self.period
            ]
            
            if len(self.calls[key]) >= self.max_calls:
                wait_time = self.period - (now - self.calls[key][0])
                raise Exception(f"請求過於頻繁，請等待 {wait_time:.1f}秒")
            
            self.calls[key].append(now)
            return func(*args, **kwargs)
        return wrapper

# 使用
limiter = RateLimiter(max_calls=100, period=60)  # 每分鐘 100 次

@limiter
def api_endpoint(data, **kwargs):
    return {"status": "success", "data": data}
```

### 3.3 API 緩存裝飾器

```python
from functools import wraps
import hashlib
import json
import time

def cache_response(ttl=300):
    """API 響應緩存裝飾器"""
    cache = {}
    
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            # 生成緩存 key
            key_data = {
                'func': func.__name__,
                'args': args,
                'kwargs': {k: v for k, v in kwargs.items() 
                          if k not in ['token', 'headers']}
            }
            key = hashlib.md5(
                json.dumps(key_data, sort_keys=True).encode()
            ).hexdigest()
            
            # 檢查緩存
            now = time.time()
            if key in cache:
                cached_time, cached_result = cache[key]
                if now - cached_time < ttl:
                    return cached_result
            
            # 執行函數
            result = func(*args, **kwargs)
            
            # 存儲緩存
            cache[key] = (now, result)
            
            return result
        return wrapper
    return decorator

# 使用
@cache_response(ttl=60)  # 緩存 60 秒
def get_weather(city, **kwargs):
    """獲取天氣 - 緩存結果"""
    # 模擬 API 調用
    time.sleep(0.1)
    return {"city": city, "temperature": 25}
```

```python
from functools import wraps
from typing import List, Dict, Any

def validate_schema(schema: Dict[str, Any]):
    """JSON Schema 風格的驗證裝飾器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            import inspect
            
            # 獲取函數參數
            sig = inspect.signature(func)
            bound = sig.bind(*args, **kwargs)
            bound.apply_defaults()
            
            # 驗證每個參數
            for param_name, rules in schema.items():
                if param_name not in bound.arguments:
                    continue
                
                value = bound.arguments[param_name]
                
                # 類型檢查
                if 'type' in rules:
                    expected_type = rules['type']
                    if not isinstance(value, expected_type):
                        raise TypeError(
                            f"{param_name} 必須是 {expected_type.__name__}"
                        )
                
                # 範圍檢查
                if 'min' in rules and value < rules['min']:
                    raise ValueError(
                        f"{param_name} 不能小於 {rules['min']}"
                    )
                if 'max' in rules and value > rules['max']:
                    raise ValueError(
                        f"{param_name} 不能大於 {rules['max']}"
                    )
                
                # 長度檢查
                if 'min_length' in rules and len(value) < rules['min_length']:
                    raise ValueError(
                        f"{param_name} 長度不能小於 {rules['min_length']}"
                    )
                if 'max_length' in rules and len(value) > rules['max_length']:
                    raise ValueError(
                        f"{param_name} 長度不能大於 {rules['max_length']}"
                    )
                
                # 枚舉檢查
                if 'enum' in rules and value not in rules['enum']:
                    raise ValueError(
                        f"{param_name} 必須是 {rules['enum']} 之一"
                    )
            
            return func(*args, **kwargs)
        return wrapper
    return decorator

# 使用
@validate_schema({
    'username': {
        'type': str,
        'min_length': 3,
        'max_length': 20
    },
    'age': {
        'type': int,
        'min': 0,
        'max': 150
    },
    'email': {
        'type': str
    },
    'role': {
        'type': str,
        'enum': ['user', 'admin', 'guest']
    }
})
def create_user(username, age, email, role='user'):
    return {
        "username": username,
        "age": age,
        "email": email,
        "role": role
    }

# 測試
user = create_user("xiaoming", 18, "xiao@example.com", role='user')
print(user)

# 會引發異常
# create_user("ab", 18, "xiao@example.com")  # username 太短
# create_user("xiaoming", 200, "xiao@example.com")  # age 超出範圍
# create_user("xiaoming", 18, "xiao@example.com", role='super')  # role 無效
```

### 4.2 數據轉換裝飾器

```python
from functools import wraps

def transform_input(**transformers):
    """輸入數據轉換裝飾器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            import inspect
            sig = inspect.signature(func)
            bound = sig.bind(*args, **kwargs)
            bound.apply_defaults()
            
            # 應用轉換
            for param_name, transformer in transformers.items():
                if param_name in bound.arguments:
                    bound.arguments[param_name] = transformer(
                        bound.arguments[param_name]
                    )
            
            return func(*bound.args, **bound.kwargs)
        return wrapper
    return decorator


def transform_output(transformer):
    """輸出數據轉換裝飾器"""
    def decorator(func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            result = func(*args, **kwargs)
            return transformer(result)
        return wrapper
    return decorator

# 使用
@transform_input(
    name=str.strip,
    email=str.lower,
    age=int
)
@transform_output(lambda x: {**x, 'created': True})
def create_user(name, email, age):
    return {
        "name": name,
        "email": email,
        "age": age
    }

user = create_user("  XiaoMing  ", "XIAO@EXAMPLE.COM", "18")
print(user)
# {'name': 'XiaoMing', 'email': 'xiao@example.com', 'age': 18, 'created': True}
```

```python
def decorator_chain(*decorators):
    def wrapper(func):
        for decorator in reversed(decorators):
            func = decorator(func)
        return func
    return wrapper

# 測試
@decorator_chain(log_calls, timing_decorator, memoize)
def fibonacci(n):
    if n < 2:
        return n
    return fibonacci(n - 1) + fibonacci(n - 2)
```

```python
import asyncio
from functools import wraps

def async_timeout(seconds):
    """異步超時裝飾器"""
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            return await asyncio.wait_for(
                func(*args, **kwargs), 
                timeout=seconds
            )
        return wrapper
    return decorator

# 測試
@async_timeout(5)
async def slow_operation():
    await asyncio.sleep(10)
    return "done"
```

### 練習 3: 實現 cache_with_key 裝飾器

```python
def cache_with_key(key_func):
    """自定義緩存鍵的裝飾器"""
    def decorator(func):
        cache = {}
        
        @wraps(func)
        def wrapper(*args, **kwargs):
            key = key_func(*args, **kwargs)
            if key in cache:
                return cache[key]
            result = func(*args, **kwargs)
            cache[key] = result
            return result
        return wrapper
    return decorator

# 測試
@cache_with_key(lambda x, y: f"{x}:{y}")
def expensive_operation(x, y):
    time.sleep(1)
    return x + y
```
