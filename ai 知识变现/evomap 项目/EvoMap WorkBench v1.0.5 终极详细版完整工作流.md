import time
import random
import logging

logger = logging.getLogger('EvoMapWorkBench')

def test_with_retry(endpoint, max_retries=5):
    """智能重试测速"""
    for attempt in range(max_retries):
        try:
            start = time.time()
            response = requests.get(endpoint, timeout=10)
            latency = int((time.time() - start) * 1000)
            return latency

# evomap_workbench.py
class RateLimiter:
    """令牌桶限流器"""
    def __init__(self, max_calls=6, period=60):
        self.max_calls = max_calls  # 最大调用次数
        self.period = period  # 时间窗口（秒）
        self.calls = []  # 调用时间记录
    
    def wait_if_needed(self):
        """等待直到可以调用"""
        now = time.time()
        # 移除过期调用记录
        self.calls = [t for t in self.calls if now - t < self.period]
        # 检查是否超限
        if len(self.calls) >= self.max_calls:
            wait = self.period - (now - self.calls[0])
            logger.warning(f"限流，等待{wait}秒...")
            time.sleep(wait)
        # 记录本次调用
        self.calls.append(time.time())

def fetch_tasks(client, rate_limiter, max_retries=5):
    """获取任务列表（带限流保护）"""
    for attempt in range(max_retries):
        try:
            rate_limiter.wait_if_needed()
            response = client.fetch_tasks()
            return response['tasks']
        except RateLimitError as e:
            wait = (5 * (2 ** attempt)) + random.uniform(0, 1)
            logger.warning(f"429 限流，等待{wait}秒后重试...")
            time.sleep(wait)
    raise Exception("多次重试后仍限流")
```

# deduplication.py
class DeduplicationManager:
    def __init__(self, cache_size=1000):
        self.cache = {}  # LRU 缓存
        self.cache_size = cache_size
        self.db = SQLiteConnection('deduplication.db')
    
    def check_duplicate(self, task_id, content_hash):
        """检查是否重复"""
        cache_key = f"{task_id}:{content_hash}"
        
        # 先查缓存（97.4% 命中率）
        if cache_key in self.cache:
            return True  # 缓存命中，跳过
        
        # 缓存未命中，查数据库
        result = self.db.query(task_id, content_hash)
        
        # 写入缓存
        self.cache[cache_key] = result
        if len(self.cache) > self.cache_size:
            # LRU 清理：移除最旧的 10%
            oldest_keys = list(self.cache.keys())[:int(self.cache_size * 0.1)]
            for key in oldest_keys:
                del self.cache[key]
        
        return result
```

# rate_limiter.py
class RateLimiter:
    def __init__(self, max_calls=6, period=60):
        self.max_calls = max_calls
        self.period = period
        self.calls = []
    
    def wait_if_needed(self):
        """等待直到可以调用"""
        now = time.time()
        self.calls = [t for t in self.calls if now - t < self.period]
        if len(self.calls) >= self.max_calls:
            wait = self.period - (now - self.calls[0])
            logger.warning(f"限流，等待{wait}秒...")
            time.sleep(wait)
        self.calls.append(time.time())

def fetch_with_retry(client, rate_limiter, max_retries=5):
    """带重试的获取"""
    for attempt in range(max_retries):
        try:
            rate_limiter.wait_if_needed()
            return client.fetch_tasks()
        except RateLimitError as e:
            wait = (5 * (2 ** attempt)) + random.uniform(0, 1)
            logger.warning(f"429 限流，等待{wait}秒后重试...")
            time.sleep(wait)
    raise Exception("多次重试后仍限流")
