## 📚 異步編程演進

### 1. 回調函數 (Callback)

```javascript
// 回調地獄示例
fs.readFile('file1.txt', 'utf-8', (err, data1) => {
  if (err) throw err;
  fs.readFile('file2.txt', 'utf-8', (err, data2) => {
    if (err) throw err;
    fs.readFile('file3.txt', 'utf-8', (err, data3) => {
      if (err) throw err;
      console.log(data1, data2, data3);
    });
  });
});
```

### 2. Promise

```javascript
// Promise 鏈式調用
fs.promises.readFile('file1.txt', 'utf-8')
  .then(data1 => fs.promises.readFile('file2.txt', 'utf-8'))
  .then(data2 => fs.promises.readFile('file3.txt', 'utf-8'))
  .then(data3 => console.log(data3))
  .catch(err => console.error(err));
```

### 3. async/await (推薦)

```javascript
// async/await 同步風格
async function readFiles() {
  try {
    const data1 = await fs.readFile('file1.txt', 'utf-8');
    const data2 = await fs.readFile('file2.txt', 'utf-8');
    const data3 = await fs.readFile('file3.txt', 'utf-8');
    console.log(data1, data2, data3);
  } catch (err) {
    console.error(err);
  }
}
```

## 🔄 事件循環 (Event Loop)

```
┌───────────────────────────┐
┌─────►│        Timers         │
│      └─────────────┬─────────┘
│      ┌─────────────▼──────────┐
│      │     I/O Callbacks      │
│      └─────────────┬──────────┘
│      ┌─────────────▼──────────┐
│      │   Idle, Prepare        │
│      └─────────────┬──────────┘
│      ┌─────────────▼──────────┐
│      │         Poll            │◄─────┐
│      └─────────────┬──────────┘      │
│      ┌─────────────▼──────────┐      │
│      │        Check            │      │
│      └─────────────┬──────────┘      │
│      ┌─────────────▼──────────┐      │
└──────┤    Close Callbacks     │──────┘
       └────────────────────────┘
```

### 各階段說明

| 階段 | 說明 | 示例 |
|------|------|------|
| **Timers** | setTimeout/setInterval 回調 | setTimeout() |
| **I/O Callbacks** | I/O 操作回調 | fs.read() |
| **Idle/Prepare** | 系統內部使用 | - |
| **Poll** | 輪詢新事件 | 大部分 I/O |
| **Check** | setImmediate 回調 | setImmediate() |
| **Close Callbacks** | 關閉回調 | socket.on('close') |

## ⚡ Promise 進階

### Promise.all (並行執行)

```javascript
async function readAllFiles() {
  const [data1, data2, data3] = await Promise.all([
    fs.readFile('file1.txt', 'utf-8'),
    fs.readFile('file2.txt', 'utf-8'),
    fs.readFile('file3.txt', 'utf-8')
  ]);
  console.log(data1, data2, data3);
}
```

### Promise.race (競賽)

```javascript
async function fetchWithTimeout(url, timeout) {
  const timeoutPromise = new Promise((_, reject) => {
    setTimeout(() => reject(new Error('超時')), timeout);
  });
  
  const fetchPromise = fetch(url);
  
  return Promise.race([fetchPromise, timeoutPromise]);
}
```

### Promise.allSettled (全部完成)

```javascript
const promises = [
  fetch('/api/users'),
  fetch('/api/posts'),
  fetch('/api/comments')
];

const results = await Promise.allSettled(promises);
results.forEach((result, index) => {
  if (result.status === 'fulfilled') {
    console.log(`請求 ${index} 成功:`, result.value);
  } else {
    console.log(`請求 ${index} 失敗:`, result.reason);
  }
});
```

## 🎯 async/await 最佳實踐

### 錯誤處理

```javascript
// ❌ 錯誤：未捕獲異常
async function getUser(id) {
  const user = await db.findById(id);
  return user;
}

// ✅ 正確：try-catch
async function getUser(id) {
  try {
    const user = await db.findById(id);
    return user;
  } catch (err) {
    console.error('獲取用戶失敗:', err);
    throw err;
  }
}

// ✅ 正確：上層統一處理
app.get('/users/:id', async (req, res, next) => {
  try {
    const user = await db.findById(req.params.id);
    res.json(user);
  } catch (err) {
    next(err);
  }
});
```

### 並行 vs 串行

```javascript
// ❌ 串行 (慢)
async function getUsers(ids) {
  const users = [];
  for (const id of ids) {
    const user = await db.findById(id); // 串行
    users.push(user);
  }
  return users;
}

// ✅ 並行 (快)
async function getUsers(ids) {
  const promises = ids.map(id => db.findById(id));
  return await Promise.all(promises); // 並行
}

// ✅ 並行 (限制並發數)
async function getUsers(ids) {
  const users = [];
  for (let i = 0; i < ids.length; i += 10) {
    const batch = ids.slice(i, i + 10);
    const batchUsers = await Promise.all(
      batch.map(id => db.findById(id))
    );
    users.push(...batchUsers);
  }
  return users;
}
```

## 🔧 異步工具庫

### async.js

```javascript
const async = require('async');

// 串行執行
async.series([
  callback => { /* 任務 1 */ callback(null, 'result1'); },
  callback => { /* 任務 2 */ callback(null, 'result2'); }
], (err, results) => {
  console.log(results); // ['result1', 'result2']
});

// 並行執行
async.parallel([
  callback => { /* 任務 1 */ callback(null, 'result1'); },
  callback => { /* 任務 2 */ callback(null, 'result2'); }
], (err, results) => {
  console.log(results);
});

// 限制並發數
async.parallelLimit(tasks, 5, callback);

// 水庫模式 (動態控制並發)
const q = async.queue((task, callback) => {
  // 處理任務
  callback();
}, 5); // 並發數 5

q.push({ id: 1 });
q.push({ id: 2 });
```

### p-limit

```javascript
const pLimit = require('p-limit');
const limit = pLimit(5); // 限制並發數為 5

const promises = ids.map(id =>
  limit(() => db.findById(id))
);

const users = await Promise.all(promises);
```

## 🐛 常見陷阱

### 1. 忘記 await

```javascript
// ❌ 錯誤
async function getData() {
  const data = db.find(); // 忘記 await
  return data; // 返回 Promise
}

// ✅ 正確
async function getData() {
  const data = await db.find();
  return data; // 返回實際數據
}
```

### 2. 在循環中串行 await

```javascript
// ❌ 慢
for (const id of ids) {
  const user = await db.findById(id);
}

// ✅ 快
await Promise.all(ids.map(id => db.findById(id)));
```

### 3. 未處理的 Promise 拒絕

```javascript
// ❌ 危險
fetch('/api/data'); // 未處理錯誤

// ✅ 安全
fetch('/api/data').catch(err => console.error(err));

// ✅ 安全
try {
  await fetch('/api/data');
} catch (err) {
  console.error(err);
}
```
