## 📊 MongoDB 集成

### 安裝與連接

```bash
npm install mongoose
```

```javascript
const mongoose = require('mongoose');

mongoose.connect('mongodb://localhost:27017/mydb', {
  useNewUrlParser: true,
  useUnifiedTopology: true
}).then(() => {
  console.log('MongoDB 連接成功');
}).catch(err => {
  console.error('MongoDB 連接失敗:', err);
});
```

### Schema 定義

```javascript
const userSchema = new mongoose.Schema({
  name: {
    type: String,
    required: [true, '姓名必填'],
    trim: true
  },
  email: {
    type: String,
    required: [true, '郵箱必填'],
    unique: true,
    lowercase: true,
    match: [/^\w+([\.-]?\w+)*@\w+([\.-]?\w+)*(\.\w{2,3})+$/, '請填寫有效的郵箱']
  },
  age: {
    type: Number,
    min: [0, '年齡不能小於 0'],
    max: [150, '年齡不能大於 150']
  },
  role: {
    type: String,
    enum: ['user', 'admin', 'vip'],
    default: 'user'
  },
  createdAt: {
    type: Date,
    default: Date.now
  }
}, {
  timestamps: true, // 自動添加 createdAt 和 updatedAt
  toJSON: { virtuals: true },
  toObject: { virtuals: true }
});

// 虛擬字段
userSchema.virtual('posts', {
  ref: 'Post',
  localField: '_id',
  foreignField: 'author'
});

const User = mongoose.model('User', userSchema);
```

## 🐬 MySQL 集成

### 安裝與連接

```bash
npm install mysql2
```

```javascript
const mysql = require('mysql2/promise');

const pool = mysql.createPool({
  host: 'localhost',
  user: 'root',
  password: 'password',
  database: 'mydb',
  waitForConnections: true,
  connectionLimit: 10,
  queueLimit: 0
});

module.exports = pool;
```

### CRUD 操作

```javascript
const pool = require('./db');

// 查詢
const [rows] = await pool.query('SELECT * FROM users WHERE age >= ?', [18]);

// 插入
const [result] = await pool.query(
  'INSERT INTO users (name, email, age) VALUES (?, ?, ?)',
  ['小明', 'xiao@example.com', 18]
);
console.log('插入 ID:', result.insertId);

// 更新
const [result] = await pool.query(
  'UPDATE users SET age = ? WHERE id = ?',
  [19, 1]
);
console.log('影響行數:', result.affectedRows);

// 刪除
const [result] = await pool.query('DELETE FROM users WHERE id = ?', [1]);
console.log('刪除行數:', result.affectedRows);

// 事務
async function transferMoney(fromId, toId, amount) {
  const connection = await pool.getConnection();
  try {
    await connection.beginTransaction();
    
    await connection.query(
      'UPDATE accounts SET balance = balance - ? WHERE id = ?',
      [amount, fromId]
    );
    
    await connection.query(
      'UPDATE accounts SET balance = balance + ? WHERE id = ?',
      [amount, toId]
    );
    
    await connection.commit();
  } catch (err) {
    await connection.rollback();
    throw err;
  } finally {
    connection.release();
  }
}
```

## 🔴 Redis 集成

### 安裝與連接

```bash
npm install ioredis
```

```javascript
const Redis = require('ioredis');

const redis = new Redis({
  host: 'localhost',
  port: 6379,
  password: 'password', // 可選
  db: 0 // 可選，默認數據庫
});

// 連接事件
redis.on('connect', () => {
  console.log('Redis 連接成功');
});

redis.on('error', (err) => {
  console.error('Redis 錯誤:', err);
});

module.exports = redis;
```

### 基本操作

```javascript
const redis = require('./redis');

// String
await redis.set('key', 'value');
await redis.setex('key', 3600, 'value'); // 1 小時過期
const value = await redis.get('key');

// Hash
await redis.hset('user:1', 'name', '小明');
await redis.hset('user:1', 'email', 'xiao@example.com');
const user = await redis.hgetall('user:1');

// List
await redis.lpush('queue', 'task1', 'task2');
const task = await redis.rpop('queue');

// Set
await redis.sadd('tags', 'tag1', 'tag2');
const tags = await redis.smembers('tags');

// Sorted Set
await redis.zadd('leaderboard', 100, 'player1');
await redis.zadd('leaderboard', 200, 'player2');
const topPlayers = await redis.zrevrange('leaderboard', 0, 9, 'WITHSCORES');
```

### 緩存示例

```javascript
async function getUser(id) {
  const cacheKey = `user:${id}`;
  
  // 嘗試從緩存獲取
  const cached = await redis.get(cacheKey);
  if (cached) {
    return JSON.parse(cached);
  }
  
  // 從數據庫獲取
  const user = await User.findById(id);
  if (!user) {
    return null;
  }
  
  // 寫入緩存 (1 小時過期)
  await redis.setex(cacheKey, 3600, JSON.stringify(user));
  
  return user;
}

// 緩存更新
async function updateUser(id, data) {
  const user = await User.findByIdAndUpdate(id, data, { new: true });
  
  // 更新緩存
  const cacheKey = `user:${id}`;
  await redis.setex(cacheKey, 3600, JSON.stringify(user));
  
  return user;
}

// 緩存刪除
async function deleteUser(id) {
  await User.findByIdAndDelete(id);
  
  // 刪除緩存
  const cacheKey = `user:${id}`;
  await redis.del(cacheKey);
}
```
