### 进阶示例：带错误处理

```python
import requests
from typing import Dict, List, Optional
import time

class SerperClient:
    def __init__(self, api_key: str):
        self.api_key = api_key
        self.base_url = "https://google.serper.dev"
        self.session = requests.Session()
        self.session.headers.update({
            'X-API-KEY': api_key,
            'Content-Type': 'application/json'
        })
    
    def search(self, query: str, **params) -> Dict:
        """
        执行 Web 搜索
        
        Args:
            query: 搜索关键词
            **params: 额外参数 (gl, hl, page, num 等)
        
        Returns:
            搜索结果字典
        """
        return self._request('/search', {'q': query, **params})
    
    def search_images(self, query: str, **params) -> Dict:
        """执行图片搜索"""
        return self._request('/images', {'q': query, **params})
    
    def search_news(self, query: str, **params) -> Dict:
        """执行新闻搜索"""
        return self._request('/news', {'q': query, **params})
    
    def search_places(self, query: str, location: str, **params) -> Dict:
        """执行地点搜索"""
        return self._request('/places', {'q': query, 'location': location, **params})
    
    def search_scholar(self, query: str, **params) -> Dict:
        """执行学术搜索"""
        return self._request('/scholar', {'q': query, **params})
    
    def _request(self, endpoint: str, payload: Dict, max_retries: int = 3) -> Dict:
        """发送请求（带重试）"""
        url = f"{self.base_url}{endpoint}"
        
        for attempt in range(max_retries):
            try:
                response = self.session.post(url, json=payload)
                response.raise_for_status()
                return response.json()
            
            except requests.exceptions.RequestException as e:
                if attempt == max_retries - 1:
                    raise
                
                # 指数退避
                wait_time = (2 ** attempt)
                print(f"请求失败，{wait_time}秒后重试...")
                time.sleep(wait_time)
        
        return {}
    
    def extract_organic_results(self, response: Dict) -> List[Dict]:
        """提取有机搜索结果"""
        return response.get('organic', [])
    
    def extract_top_10(self, response: Dict) -> List[Dict]:
        """提取前 10 个结果"""
        results = self.extract_organic_results(response)
        return results[:10]

# 使用示例
if __name__ == "__main__":
    client = SerperClient("01529847d4aa3cf47b86ca87d28519110db06390")
    
    # Web 搜索
    results = client.search("AI agent platform", num=10)
    top_results = client.extract_top_10(results)
    
    print(f"找到 {len(top_results)} 个结果:\n")
    for i, result in enumerate(top_results, 1):
        print(f"{i}. {result.get('title', 'N/A')}")
        print(f"   {result.get('link', 'N/A')}")
        print(f"   {result.get('snippet', 'N/A')[:100]}...\n")
```

### 进阶示例：带错误处理

```javascript
const axios = require('axios');

class SerperClient {
  constructor(apiKey) {
    this.apiKey = apiKey;
    this.baseURL = 'https://google.serper.dev';
    this.client = axios.create({
      baseURL: this.baseURL,
      headers: {
        'X-API-KEY': apiKey,
        'Content-Type': 'application/json'
      }
    });
  }

  async search(query, params = {}) {
    return this._request('/search', { q: query, ...params });
  }

  async searchImages(query, params = {}) {
    return this._request('/images', { q: query, ...params });
  }

  async searchNews(query, params = {}) {
    return this._request('/news', { q: query, ...params });
  }

  async searchPlaces(query, location, params = {}) {
    return this._request('/places', { q: query, location, ...params });
  }

  async searchScholar(query, params = {}) {
    return this._request('/scholar', { q: query, ...params });
  }

  async _request(endpoint, payload, maxRetries = 3) {
    for (let attempt = 0; attempt < maxRetries; attempt++) {
      try {
        const response = await this.client.post(endpoint, payload);
        return response.data;
      } catch (error) {
        if (attempt === maxRetries - 1) {
          throw error;
        }
        
        // 指数退避
        const waitTime = Math.pow(2, attempt) * 1000;
        console.log(`请求失败，${waitTime}ms 后重试...`);
        await new Promise(resolve => setTimeout(resolve, waitTime));
      }
    }
  }

  extractOrganicResults(response) {
    return response.organic || [];
  }

  extractTop10(response) {
    return this.extractOrganicResults(response).slice(0, 10);
  }
}

// 使用示例
(async () => {
  const client = new SerperClient('01529847d4aa3cf47b86ca87d28519110db06390');
  
  const results = await client.search('AI agent platform', { num: 10 });
  const topResults = client.extractTop10(results);
  
  console.log(`找到 ${topResults.length} 个结果:\n`);
  topResults.forEach((result, i) => {
    console.log(`${i + 1}. ${result.title}`);
    console.log(`   ${result.link}`);
    console.log(`   ${result.snippet?.substring(0, 100)}...\n`);
  });
})();
```

## 3️⃣ Bash 示例

### 基础搜索

```bash
#!/bin/bash

# 配置
API_KEY="01529847d4aa3cf47b86ca87d28519110db06390"
BASE_URL="https://google.serper.dev"

# 搜索函数
serper_search() {
  local query="$1"
  
  curl --request POST \
    --url "${BASE_URL}/search" \
    --header "X-API-KEY: ${API_KEY}" \
    --header 'Content-Type: application/json' \
    --data "{\"q\": \"${query}\"}" \
    2>/dev/null | jq .
}

# 使用示例
serper_search "EvoMap AI agent"
```

### 进阶脚本

```bash
#!/bin/bash

# Serper API 搜索工具

API_KEY="01529847d4aa3cf47b86ca87d28519110db06390"
BASE_URL="https://google.serper.dev"

# 颜色
RED='\033[0;31m'
GREEN='\033[0;32m'
BLUE='\033[0;34m'
NC='\033[0m'

# 搜索函数
search() {
  local query="$1"
  local type="${2:-search}"
  
  echo -e "${BLUE}搜索：${query}${NC}"
  echo ""
  
  local response=$(curl --request POST \
    --url "${BASE_URL}/${type}" \
    --header "X-API-KEY: ${API_KEY}" \
    --header 'Content-Type: application/json' \
    --silent \
    --data "{\"q\": \"${query}\"}")
  
  # 提取并显示结果
  echo "$response" | jq -r '.organic[]? | "\(.title)\n\(.link)\n\(.snippet)\n"'
}

# 显示帮助
show_help() {
  echo "Serper API 搜索工具"
  echo ""
  echo "用法："
  echo "  $0 search <query>    - Web 搜索"
  echo "  $0 news <query>      - 新闻搜索"
  echo "  $0 images <query>    - 图片搜索"
  echo "  $0 scholar <query>   - 学术搜索"
  echo ""
  echo "示例："
  echo "  $0 search 'AI agent'"
  echo "  $0 news 'AI technology'"
}

# 主程序
case "$1" in
  search|news|images|scholar)
    if [ -z "$2" ]; then
      echo -e "${RED}错误：请提供搜索关键词${NC}"
      show_help
      exit 1
    fi
    search "$2" "$1"
    ;;
  help|--help|-h)
    show_help
    ;;
  *)
    echo -e "${RED}错误：未知命令${NC}"
    show_help
    exit 1
    ;;
esac
```
