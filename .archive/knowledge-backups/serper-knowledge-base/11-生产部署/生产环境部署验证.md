## 📊 性能测试

### 基准测试脚本

```python
#!/usr/bin/env python3
# testing/benchmark.py
"""Serper API 性能基准测试"""

import requests
import time
import statistics
from concurrent.futures import ThreadPoolExecutor

API_KEY = "01529847d4aa3cf47b86ca87d28519110db06390"
BASE_URL = "https://google.serper.dev"

def single_request(endpoint="search", query="test"):
    """单次请求"""
    start = time.time()
    response = requests.post(
        f"{BASE_URL}/{endpoint}",
        headers={"X-API-KEY": API_KEY},
        json={"q": query},
        timeout=30
    )
    latency = time.time() - start
    return {
        "status": response.status_code,
        "latency": latency,
        "success": response.status_code == 200
    }

def benchmark(concurrency=10, requests_per_client=100):
    """性能基准测试"""
    print(f"🚀 开始基准测试 - 并发：{concurrency}, 请求数：{requests_per_client}")
    
    results = []
    
    with ThreadPoolExecutor(max_workers=concurrency) as executor:
        futures = [
            executor.submit(single_request)
            for _ in range(requests_per_client)
        ]
        
        for future in futures:
            results.append(future.result())
    
    # 计算统计
    successful = [r for r in results if r["success"]]
    latencies = [r["latency"] for r in successful]
    
    print(f"\n📊 测试结果:")
    print(f"  总请求数：{len(results)}")
    print(f"  成功数：{len(successful)}")
    print(f"  成功率：{len(successful)/len(results)*100:.2f}%")
    print(f"  平均延迟：{statistics.mean(latencies)*1000:.2f}ms")
    print(f"  P50 延迟：{statistics.median(latencies)*1000:.2f}ms")
    print(f"  P95 延迟：{sorted(latencies)[int(len(latencies)*0.95)]*1000:.2f}ms")
    print(f"  P99 延迟：{sorted(latencies)[int(len(latencies)*0.99)]*1000:.2f}ms")
    
    return results

if __name__ == "__main__":
    benchmark(concurrency=10, requests_per_client=100)
```

## 🔄 容错机制

### 重试策略

```python
# retry_config.py
from tenacity import retry, stop_after_attempt, wait_exponential

@retry(
    stop=stop_after_attempt(3),
    wait=wait_exponential(multiplier=1, min=1, max=10),
    reraise=True
)
def serper_search_with_retry(query):
    """带重试的搜索"""
    response = requests.post(
        "https://google.serper.dev/search",
        headers={"X-API-KEY": API_KEY},
        json={"q": query},
        timeout=30
    )
    response.raise_for_status()
    return response.json()
```

### 降级策略

```python
# fallback_config.py
def search_with_fallback(query):
    """带降级策略的搜索"""
    # 主服务
    try:
        return serper_search_with_retry(query)
    except Exception as e:
        log_error(f"Serper failed: {e}")
        
        # 降级到缓存
        cached = get_from_cache(query)
        if cached:
            return cached
        
        # 降级到备用服务
        try:
            return bing_search(query)
        except Exception as e2:
            log_error(f"Fallback failed: {e2}")
            return {"error": "Search unavailable", "fallback": True}
```
