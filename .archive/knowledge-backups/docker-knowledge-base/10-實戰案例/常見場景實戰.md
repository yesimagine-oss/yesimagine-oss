## 🌐 場景 1: Web 應用部署

### 需求
- Nginx 反向代理
- Node.js 應用
- Redis 緩存
- MySQL 數據庫

### Docker Compose 配置

```yaml
version: '3.8'

services:
  nginx:
    image: nginx:alpine
    container_name: nginx
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx.conf:/etc/nginx/nginx.conf:ro
      - ./ssl:/etc/nginx/ssl:ro
    depends_on:
      - app
    networks:
      - webnet
    restart: unless-stopped

  app:
    build:
      context: ./app
      dockerfile: Dockerfile
    container_name: node-app
    environment:
      - NODE_ENV=production
      - DB_HOST=mysql
      - REDIS_HOST=redis
    depends_on:
      - mysql
      - redis
    networks:
      - webnet
      - dbnet
    restart: unless-stopped
    deploy:
      resources:
        limits:
          cpus: '2'
          memory: 1G

  mysql:
    image: mysql:8.0
    container_name: mysql
    environment:
      - MYSQL_ROOT_PASSWORD=${MYSQL_ROOT_PASSWORD}
      - MYSQL_DATABASE=app
      - MYSQL_USER=appuser
      - MYSQL_PASSWORD=${MYSQL_PASSWORD}
    volumes:
      - mysql-data:/var/lib/mysql
      - ./mysql/init.sql:/docker-entrypoint-initdb.d/init.sql:ro
    networks:
      - dbnet
    restart: unless-stopped
    deploy:
      resources:
        limits:
          cpus: '2'
          memory: 2G

  redis:
    image: redis:7-alpine
    container_name: redis
    command: redis-server --appendonly yes
    volumes:
      - redis-data:/data
    networks:
      - dbnet
    restart: unless-stopped

volumes:
  mysql-data:
  redis-data:

networks:
  webnet:
  dbnet:
```

### 部署命令

```bash
# 啟動
docker-compose up -d

# 查看狀態
docker-compose ps

# 查看日誌
docker-compose logs -f

# 重啟
docker-compose restart

# 停止
docker-compose down

# 完全清理
docker-compose down -v
```

## 🔄 場景 2: CI/CD 流水線

### 需求
- GitLab CI 集成
- Docker 構建
- 自動化測試
- 自動部署

### GitLab CI 配置

```yaml
# .gitlab-ci.yml
stages:
  - build
  - test
  - deploy

variables:
  DOCKER_IMAGE: $CI_REGISTRY_IMAGE:$CI_COMMIT_SHA
  DOCKER_TLS_CERTDIR: "/certs"

build:
  stage: build
  image: docker:24-dind
  services:
    - docker:24-dind
  script:
    - docker login -u $CI_REGISTRY_USER -p $CI_REGISTRY_PASSWORD $CI_REGISTRY
    - docker build -t $DOCKER_IMAGE .
    - docker push $DOCKER_IMAGE
    - docker tag $DOCKER_IMAGE $CI_REGISTRY_IMAGE:latest
    - docker push $CI_REGISTRY_IMAGE:latest

test:
  stage: test
  image: docker:24
  services:
    - docker:24-dind
  script:
    - docker run --rm $DOCKER_IMAGE npm test

deploy:
  stage: deploy
  image: alpine:latest
  script:
    - apk add --no-cache openssh-client
    - eval $(ssh-agent -s)
    - echo "$SSH_PRIVATE_KEY" | ssh-add -
    - ssh -o StrictHostKeyChecking=no user@server "
        cd /app &&
        docker pull $DOCKER_IMAGE &&
        docker stop app || true &&
        docker rm app || true &&
        docker run -d --name app $DOCKER_IMAGE
      "
  only:
    - main
```

## 📊 場景 3: 監控系統

### 需求
- Prometheus 監控
- Grafana 可視化
- Node Exporter 主機監控
- cAdvisor 容器監控

### Docker Compose 配置

```yaml
version: '3.8'

services:
  prometheus:
    image: prom/prometheus:latest
    container_name: prometheus
    ports:
      - "9090:9090"
    volumes:
      - ./prometheus.yml:/etc/prometheus/prometheus.yml:ro
      - prometheus-data:/prometheus
    command:
      - '--config.file=/etc/prometheus/prometheus.yml'
      - '--storage.tsdb.path=/prometheus'
    networks:
      - monitor
    restart: unless-stopped

  grafana:
    image: grafana/grafana:latest
    container_name: grafana
    ports:
      - "3000:3000"
    environment:
      - GF_SECURITY_ADMIN_PASSWORD=${GRAFANA_PASSWORD}
    volumes:
      - grafana-data:/var/lib/grafana
    depends_on:
      - prometheus
    networks:
      - monitor
    restart: unless-stopped

  node-exporter:
    image: prom/node-exporter:latest
    container_name: node-exporter
    ports:
      - "9100:9100"
    volumes:
      - /proc:/host/proc:ro
      - /sys:/host/sys:ro
      - /:/rootfs:ro
    command:
      - '--path.procfs=/host/proc'
      - '--path.sysfs=/host/sys'
    networks:
      - monitor
    restart: unless-stopped

  cadvisor:
    image: gcr.io/cadvisor/cadvisor:latest
    container_name: cadvisor
    ports:
      - "8080:8080"
    volumes:
      - /:/rootfs:ro
      - /var/run:/var/run:ro
      - /sys:/sys:ro
      - /var/lib/docker/:/var/lib/docker:ro
    networks:
      - monitor
    restart: unless-stopped

volumes:
  prometheus-data:
  grafana-data:

networks:
  monitor:
```

### Prometheus 配置

```yaml
# prometheus.yml
global:
  scrape_interval: 15s

scrape_configs:
  - job_name: 'prometheus'
    static_configs:
      - targets: ['localhost:9090']

  - job_name: 'node'
    static_configs:
      - targets: ['node-exporter:9100']

  - job_name: 'cadvisor'
    static_configs:
      - targets: ['cadvisor:8080']
```

## 🗄️ 場景 4: 數據庫集群

### 需求
- MySQL 主從複製
- 自動故障轉移
- 數據備份

### Docker Compose 配置

```yaml
version: '3.8'

services:
  mysql-master:
    image: mysql:8.0
    container_name: mysql-master
    environment:
      - MYSQL_ROOT_PASSWORD=${MYSQL_ROOT_PASSWORD}
      - MYSQL_REPLICATION_USER=repl
      - MYSQL_REPLICATION_PASSWORD=${REPL_PASSWORD}
    volumes:
      - master-data:/var/lib/mysql
      - ./mysql/master.cnf:/etc/mysql/conf.d/master.cnf:ro
    ports:
      - "3306:3306"
    networks:
      - dbnet
    restart: unless-stopped

  mysql-slave:
    image: mysql:8.0
    container_name: mysql-slave
    environment:
      - MYSQL_ROOT_PASSWORD=${MYSQL_ROOT_PASSWORD}
    volumes:
      - slave-data:/var/lib/mysql
      - ./mysql/slave.cnf:/etc/mysql/conf.d/slave.cnf:ro
    depends_on:
      - mysql-master
    networks:
      - dbnet
    restart: unless-stopped
    command: >
      bash -c "
        sleep 30 &&
        mysql -h mysql-master -u root -p${MYSQL_ROOT_PASSWORD} -e
        'CHANGE MASTER TO MASTER_HOST=\"mysql-master\",
         MASTER_USER=\"repl\",
         MASTER_PASSWORD=\"${REPL_PASSWORD}\",
         MASTER_AUTO_POSITION=1;
         START SLAVE;'
      "

  backup:
    image: mysql:8.0
    container_name: mysql-backup
    environment:
      - MYSQL_ROOT_PASSWORD=${MYSQL_ROOT_PASSWORD}
    volumes:
      - ./backups:/backups
    depends_on:
      - mysql-master
    networks:
      - dbnet
    command: >
      bash -c "
        while true; do
          mysqldump -h mysql-master -u root -p${MYSQL_ROOT_PASSWORD}
          --all-databases > /backups/backup-$(date +%Y%m%d_%H%M%S).sql;
          sleep 86400;
        done
      "

volumes:
  master-data:
  slave-data:

networks:
  dbnet:
```

## 🔐 場景 5: 安全加固部署

### 需求
- 非 root 用戶運行
- 只讀文件系統
- 資源限制
- 網絡隔離

### Docker Compose 配置

```yaml
version: '3.8'

services:
  secure-app:
    build: .
    container_name: secure-app
    user: "1000:1000"
    read_only: true
    tmpfs:
      - /tmp:rw,noexec,nosuid,size=100m
      - /var/run:rw,noexec,nosuid,size=50m
    cap_drop:
      - ALL
    cap_add:
      - NET_BIND_SERVICE
    security_opt:
      - no-new-privileges:true
    networks:
      - internal
    deploy:
      resources:
        limits:
          cpus: '1'
          memory: 512M
          pids: 100
    restart: unless-stopped

  proxy:
    image: nginx:alpine
    container_name: secure-proxy
    ports:
      - "80:80"
    volumes:
      - ./proxy.conf:/etc/nginx/nginx.conf:ro
    depends_on:
      - secure-app
    networks:
      - public
      - internal
    deploy:
      resources:
        limits:
          cpus: '0.5'
          memory: 128M
    restart: unless-stopped

networks:
  public:
  internal:
    internal: true
```

## 📦 場景 6: 微服務架構

### 需求
- 服務發現
- 負載均衡
- API 網關
- 服務網格

### Docker Compose 配置

```yaml
version: '3.8'

services:
  gateway:
    image: kong:latest
    container_name: api-gateway
    ports:
      - "80:8000"
      - "443:8443"
    environment:
      - KONG_DATABASE=off
      - KONG_DECLARATIVE_CONFIG=/kong/declarative/kong.yml
    volumes:
      - ./kong:/kong/declarative:ro
    networks:
      - mesh
    restart: unless-stopped

  service-a:
    build: ./services/a
    container_name: service-a
    environment:
      - SERVICE_NAME=service-a
      - DB_HOST=postgres
    depends_on:
      - postgres
    networks:
      - mesh
    deploy:
      replicas: 3
    restart: unless-stopped

  service-b:
    build: ./services/b
    container_name: service-b
    environment:
      - SERVICE_NAME=service-b
      - REDIS_HOST=redis
    depends_on:
      - redis
    networks:
      - mesh
    deploy:
      replicas: 2
    restart: unless-stopped

  postgres:
    image: postgres:15-alpine
    container_name: postgres
    environment:
      - POSTGRES_PASSWORD=${DB_PASSWORD}
    volumes:
      - postgres-data:/var/lib/postgresql/data
    networks:
      - mesh
    restart: unless-stopped

  redis:
    image: redis:7-alpine
    container_name: redis
    volumes:
      - redis-data:/data
    networks:
      - mesh
    restart: unless-stopped

volumes:
  postgres-data:
  redis-data:

networks:
  mesh:
```
