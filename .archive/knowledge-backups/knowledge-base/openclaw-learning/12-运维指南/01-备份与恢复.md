## 🔧 备份方法

### 方法 1: 手动备份

```bash
# 1. 创建备份目录
mkdir -p ~/backups/openclaw

# 2. 备份配置
cp -r ~/.openclaw/config ~/backups/openclaw/config-$(date +%Y%m%d)

# 3. 备份工作区
tar -czf ~/backups/openclaw/workspace-$(date +%Y%m%d).tar.gz \
  ~/.openclaw/workspace/

# 4. 验证备份
ls -lh ~/backups/openclaw/
```

### 方法 2: 使用 OpenClaw 命令

```bash
# 导出配置
openclaw config --export > ~/backups/openclaw/config-$(date +%Y%m%d).json

# 导出技能列表
openclaw skills list > ~/backups/openclaw/skills-$(date +%Y%m%d).txt
```

### 方法 3: 自动化备份脚本

```bash
#!/bin/bash
# ~/scripts/openclaw-backup.sh

BACKUP_DIR=~/backups/openclaw
DATE=$(date +%Y%m%d_%H%M%S)
RETENTION_DAYS=30

# 创建备份目录
mkdir -p $BACKUP_DIR

# 备份配置
echo "📋 备份配置文件..."
cp -r ~/.openclaw/config $BACKUP_DIR/config-$DATE

# 备份工作区
echo "📁 备份工作区..."
tar -czf $BACKUP_DIR/workspace-$DATE.tar.gz \
  --exclude='*.log' \
  --exclude='cache/*' \
  ~/.openclaw/workspace/

# 备份环境变量（如有）
if [ -f ~/.openclaw/.env ]; then
  cp ~/.openclaw/.env $BACKUP_DIR/env-$DATE
fi

# 生成备份清单
cat > $BACKUP_DIR/manifest-$DATE.txt << EOF
OpenClaw 备份清单
==================
备份时间：$(date)
配置版本：$(openclaw --version)
工作区大小：$(du -sh ~/.openclaw/workspace | cut -f1)

备份文件:
$(ls -lh $BACKUP_DIR/*-$DATE*)
EOF

# 清理旧备份
echo "🧹 清理 $RETENTION_DAYS 天前的备份..."
find $BACKUP_DIR -name "*.tar.gz" -mtime +$RETENTION_DAYS -delete
find $BACKUP_DIR -name "config-*" -mtime +$RETENTION_DAYS -delete

# 发送通知
echo "✅ 备份完成"
echo "备份位置：$BACKUP_DIR"
echo "备份大小：$(du -sh $BACKUP_DIR | cut -f1)"

# 可选：发送到远程存储
# rsync -av $BACKUP_DIR/ user@remote:/backups/openclaw/
```

## 🔄 恢复方法

### 恢复配置

```bash
# 1. 停止服务
openclaw stop

# 2. 备份当前配置（以防万一）
cp -r ~/.openclaw/config ~/.openclaw/config.bak.$(date +%Y%m%d)

# 3. 恢复配置
cp -r ~/backups/openclaw/config-20260312/* ~/.openclaw/config/

# 4. 验证配置
openclaw config --validate

# 5. 重启服务
openclaw start
```

### 恢复工作区

```bash
# 1. 停止服务
openclaw stop

# 2. 备份当前工作区
tar -czf ~/workspace.bak.$(date +%Y%m%d).tar.gz ~/.openclaw/workspace/

# 3. 恢复工作区
tar -xzf ~/backups/openclaw/workspace-20260312.tar.gz -C ~/

# 4. 修复权限
chmod -R 755 ~/.openclaw/workspace/
chown -R $USER:$USER ~/.openclaw/workspace/

# 5. 验证
openclaw status

# 6. 重启服务
openclaw start
```

### 恢复单个文件

```bash
# 从备份中提取单个文件
tar -xzf workspace-20260312.tar.gz \
  --strip-components=3 \
  home/admin/.openclaw/workspace/MEMORY.md

# 或直接复制
cp ~/backups/openclaw/config-20260312/models.json ~/.openclaw/config/
```

## 📋 备份验证

### 验证脚本

```bash
#!/bin/bash
# ~/scripts/openclaw-verify-backup.sh

BACKUP_DIR=~/backups/openclaw
LATEST_BACKUP=$(ls -t $BACKUP_DIR/workspace-*.tar.gz | head -1)

echo "🔍 验证备份完整性..."

# 1. 检查备份文件是否存在
if [ ! -f "$LATEST_BACKUP" ]; then
  echo "❌ 未找到备份文件"
  exit 1
fi

echo "✅ 备份文件存在：$LATEST_BACKUP"

# 2. 验证 tar 文件完整性
if tar -tzf $LATEST_BACKUP > /dev/null 2>&1; then
  echo "✅ 压缩包完整性检查通过"
else
  echo "❌ 压缩包损坏"
  exit 1
fi

# 3. 检查关键文件
echo "📋 检查关键文件..."
tar -tzf $LATEST_BACKUP | grep -q "workspace/SOUL.md" && echo "  ✅ SOUL.md"
tar -tzf $LATEST_BACKUP | grep -q "workspace/MEMORY.md" && echo "  ✅ MEMORY.md"
tar -tzf $LATEST_BACKUP | grep -q "workspace/USER.md" && echo "  ✅ USER.md"

# 4. 检查备份大小
BACKUP_SIZE=$(du -h $LATEST_BACKUP | cut -f1)
echo "📊 备份大小：$BACKUP_SIZE"

# 5. 检查备份年龄
BACKUP_DATE=$(stat -c %y $LATEST_BACKUP | cut -d' ' -f1)
echo "📅 备份日期：$BACKUP_DATE"

echo "✅ 备份验证完成"
```

## ☁️ 远程备份

### 备份到云存储

```bash
# 使用 rclone 备份到云存储

# 1. 安装 rclone
curl https://rclone.org/install.sh | sudo bash

# 2. 配置云存储
rclone config

# 3. 同步备份
rclone sync ~/backups/openclaw remote:backups/openclaw

# 4. 配置定时同步
# 添加到 crontab
0 5 * * * rclone sync ~/backups/openclaw remote:backups/openclaw
```

### 备份到 Git

```bash
# 1. 初始化 Git 仓库
cd ~/.openclaw/workspace
git init

# 2. 创建 .gitignore
cat > .gitignore << EOF
*.log
cache/
node_modules/
.env
EOF

# 3. 提交配置
git add SOUL.md USER.md TOOLS.md skills/
git commit -m "Backup OpenClaw workspace"

# 4. 推送到远程
git remote add origin git@github.com:user/openclaw-backup.git
git push -u origin main
```
