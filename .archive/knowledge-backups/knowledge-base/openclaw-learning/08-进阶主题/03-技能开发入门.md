## 🔧 使用工具的示例

### 调用 web_search

```javascript
async execute(context, params) {
  const { tools } = context;
  
  // 调用搜索工具
  const searchResult = await tools.web_search({
    query: '最新 AI 新闻',
    count: 5
  });
  
  // 处理结果
  const summary = searchResult.results
    .slice(0, 3)
    .map(r => `• ${r.title}`)
    .join('\n');
  
  return {
    content: `📰 搜索结果：\n\n${summary}`
  };
}
```

## 🎯 完整示例：天气查询技能

```javascript
/**
 * 技能名称：simple-weather
 * 描述：简化版天气查询
 */

const https = require('https');

module.exports = {
  meta: {
    name: 'simple-weather',
    version: '1.0.0',
    description: '简化版天气查询技能'
  },

  config: {
    enabled: true,
    triggerPatterns: [
      /天气/,
      /下雨/,
      /温度/
    ],
    defaultLocation: '北京'
  },

  async execute(context, params) {
    const { message, config } = context;
    
    // 提取城市名
    const city = this.extractCity(message.content) || config.defaultLocation;
    
    // 调用天气 API
    const weather = await this.fetchWeather(city);
    
    // 生成回复
    const reply = this.formatWeather(weather);
    
    return { content: reply };
  },

  // 提取城市名
  extractCity(text) {
    const cities = ['北京', '上海', '广州', '深圳', '杭州'];
    for (const city of cities) {
      if (text.includes(city)) return city;
    }
    return null;
  },

  // 获取天气数据
  async fetchWeather(city) {
    return new Promise((resolve, reject) => {
      const url = `https://wttr.in/${city}?format=j1`;
      https.get(url, (res) => {
        let data = '';
        res.on('data', chunk => data += chunk);
        res.on('end', () => {
          try {
            const json = JSON.parse(data);
            resolve({
              temp: json.current_condition[0].temp_C,
              weather: json.current_condition[0].weatherDesc[0].value,
              humidity: json.current_condition[0].humidity
            });
          } catch (e) {
            reject(e);
          }
        });
      }).on('error', reject);
    });
  },

  // 格式化输出
  formatWeather(data) {
    return `🌤️ 天气信息\n\n` +
           `温度：${data.temp}°C\n` +
           `天气：${data.weather}\n` +
           `湿度：${data.humidity}%`;
  }
};
```

## 📚 技能开发最佳实践

### 1. 错误处理

```javascript
async execute(context, params) {
  try {
    // 主要逻辑
    return await this.doSomething();
  } catch (error) {
    console.error('[my-skill] 错误:', error);
    return {
      content: `❌ 出错了：${error.message}`,
      type: 'error'
    };
  }
}
```

### 2. 超时控制

```javascript
async execute(context, params) {
  const timeout = 10000; // 10 秒
  
  const result = await Promise.race([
    this.doSomething(),
    new Promise((_, reject) => 
      setTimeout(() => reject(new Error('超时')), timeout)
    )
  ]);
  
  return result;
}
```

### 3. 配置验证

```javascript
async init(config) {
  if (!config.apiKey) {
    throw new Error('缺少 apiKey 配置');
  }
  if (config.timeout < 1000) {
    throw new Error('timeout 不能小于 1000ms');
  }
}
```
