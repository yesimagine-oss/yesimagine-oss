### 步骤 1: 启用 Calendar API

## 🔧 日历管理脚本

```javascript
// ~/.openclaw/workspace/skills/calendar-manager/index.js
const {google} = require('googleapis');

module.exports = {
  meta: {
    name: 'calendar-manager',
    version: '1.0.0'
  },

  config: {
    enabled: true,
    credentialsPath: '~/.openclaw/calendar-credentials.json',
    calendarId: 'primary',
    timezone: 'Asia/Shanghai'
  },

  async execute(context, params) {
    const action = params?.action || 'summary';
    
    switch(action) {
      case 'summary':
        return await this.getDailySummary();
      case 'next-event':
        return await this.getNextEvent();
      case 'create':
        return await this.createEvent(params);
      default:
        return {content: '未知操作'};
    }
  },

  async getDailySummary() {
    const calendar = await this.getCalendar();
    const now = new Date();
    const endOfDay = new Date(now);
    endOfDay.setHours(23, 59, 59);

    const events = await calendar.events.list({
      calendarId: this.config.calendarId,
      timeMin: now.toISOString(),
      timeMax: endOfDay.toISOString(),
      singleEvents: true,
      orderBy: 'startTime'
    });

    if (!events.data.items.length) {
      return {content: '📅 今日没有日程安排'};
    }

    let summary = '📅 今日日程\n\n';
    events.data.items.forEach((event, i) => {
      const start = new Date(event.start.dateTime || event.start.date);
      const time = start.toLocaleTimeString('zh-CN', {hour: '2-digit', minute:'2-digit'});
      summary += `${i + 1}. ${time} - ${event.summary}\n`;
      if (event.location) summary += `   📍 ${event.location}\n`;
      summary += '\n';
    });

    return {content: summary};
  },

  async getNextEvent() {
    const calendar = await this.getCalendar();
    const now = new Date();
    const nextHour = new Date(now.getTime() + 60 * 60 * 1000);

    const events = await calendar.events.list({
      calendarId: this.config.calendarId,
      timeMin: now.toISOString(),
      timeMax: nextHour.toISOString(),
      singleEvents: true,
      orderBy: 'startTime',
      maxResults: 1
    });

    if (!events.data.items.length) {
      return {content: '✅ 接下来 1 小时没有会议'};
    }

    const event = events.data.items[0];
    const start = new Date(event.start.dateTime);
    const minutesUntil = Math.round((start - now) / 60000);

    return {
      content: `⏰ 会议提醒\n\n` +
               `主题：${event.summary}\n` +
               `时间：${minutesUntil} 分钟后\n` +
               `链接：${event.hangoutLink || '无'}`
    };
  },

  async createEvent(params) {
    const calendar = await this.getCalendar();
    
    const event = {
      summary: params.title,
      description: params.description,
      start: {
        dateTime: params.startTime,
        timeZone: this.config.timezone
      },
      end: {
        dateTime: params.endTime,
        timeZone: this.config.timezone
      }
    };

    const result = await calendar.events.insert({
      calendarId: this.config.calendarId,
      requestBody: event
    });

    return {
      content: `✅ 日程已创建\n\n` +
               `主题：${params.title}\n` +
               `时间：${params.startTime}\n` +
               `链接：${result.data.hangoutLink || '无'}`
    };
  },

  async getCalendar() {
    const auth = new google.auth.GoogleAuth({
      keyFile: this.config.credentialsPath,
      scopes: ['https://www.googleapis.com/auth/calendar']
    });
    
    return google.calendar({version: 'v3', auth});
  }
};
```

## 🔨 进阶功能

### 1. 冲突检测

```javascript
async checkConflicts() {
  const calendar = await this.getCalendar();
  const events = await calendar.events.list({...});
  
  // 检测时间冲突
  for (let i = 0; i < events.length - 1; i++) {
    if (events[i].end > events[i+1].start) {
      return {conflict: true, events: [events[i], events[i+1]]};
    }
  }
  return {conflict: false};
}
```
