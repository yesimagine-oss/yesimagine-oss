### 步骤 2: 创建邮件检查脚本

```javascript
// ~/.openclaw/workspace/skills/email-checker/index.js
const Imap = require('imap');

module.exports = {
  meta: {
    name: 'email-checker',
    version: '1.0.0'
  },

  config: {
    enabled: true,
    imapConfig: {
      user: 'your-email@gmail.com',
      password: 'your-app-password',
      host: 'imap.gmail.com',
      port: 993,
      tls: true
    }
  },

  async execute(context) {
    const imap = new Imap(this.config.imapConfig);
    
    return new Promise((resolve, reject) => {
      imap.connect();
      
      imap.on('ready', () => {
        imap.openBox('INBOX', false, (err, box) => {
          if (err) reject(err);
          
          imap.search(['UNSEEN', ['SINCE', 'Today']], (err, results) => {
            if (err) reject(err);
            
            if (!results || !results.length) {
              resolve({ content: '📧 没有新邮件' });
              imap.end();
              return;
            }
            
            const fetch = imap.fetch(results, {
              bodies: 'HEADER.FIELDS (FROM SUBJECT DATE)',
              limit: 5
            });
            
            let emails = [];
            fetch.on('message', (msg) => {
              msg.on('body', (stream) => {
                let buffer = '';
                stream.on('data', (chunk) => buffer += chunk);
                stream.on('end', () => {
                  emails.push(Imap.parseHeader(buffer));
                });
              });
            });
            
            fetch.on('end', () => {
              const summary = this.formatEmails(emails);
              resolve({ content: summary });
              imap.end();
            });
          });
        });
      });
      
      imap.on('error', reject);
    });
  },

  formatEmails(emails) {
    let summary = '📧 新邮件提醒\n\n';
    emails.slice(0, 5).forEach((email, i) => {
      summary += `${i + 1}. ${email.subject?.[0] || '(无主题)'}\n`;
      summary += `   来自：${email.from?.[0] || '未知'}\n\n`;
    });
    return summary;
  }
};
```

## 🔨 进阶功能

### 1. 智能分类

```javascript
classifyEmail(email) {
  const subject = email.subject?.[0] || '';
  const from = email.from?.[0] || '';
  
  if (from.includes('boss') || subject.includes('紧急')) {
    return 'urgent';
  }
  if (subject.includes('会议') || subject.includes('邀请')) {
    return 'meeting';
  }
  if (from.includes('github') || from.includes('gitlab')) {
    return 'notification';
  }
  return 'normal';
}
```

## ⚠️ 常见问题

### Q1: IMAP 连接失败
**检查**:
- 用户名密码是否正确
- 是否启用 IMAP
- 防火墙是否阻止 993 端口

### Q2: 认证失败
**解决**:
- Gmail 需要使用应用专用密码
- 开启两步验证
- 检查 OAuth 配置

### Q3: 中文乱码
**解决**:
```javascript
// 添加编码处理
const subject = Imap.decodeHeader(email.subject[0]);
```
