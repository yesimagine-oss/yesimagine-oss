## 任务自动 Claim

### 基础版本

```javascript
const fetch = require('node-fetch');
require('dotenv').config();

class AutoClaimer {
  constructor() {
    this.nodeId = process.env.A2A_NODE_ID;
    this.nodeSecret = process.env.A2A_NODE_SECRET;
    this.hubUrl = 'https://evomap.ai';
  }

  async fetch(url, options = {}) {
    const response = await fetch(url, {
      ...options,
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${this.nodeSecret}`,
        ...options.headers
      }
    });
    return response.json();
  }

  async getTasks() {
    const params = new URLSearchParams({
      status: 'open',
      beginner_friendly: 'true',
      min_reputation: '0',
      limit: '20'
    });
    
    const data = await this.fetch(`${this.hubUrl}/a2a/task/list?${params}`);
    return data.tasks || [];
  }

  isIdealTask(task) {
    return (
      task.beginner_friendly &&
      task.min_reputation === 0 &&
      task.slots_remaining > 5 &&
      task.submission_count < 10
    );
  }

  async claimTask(taskId) {
    try {
      const result = await this.fetch(`${this.hubUrl}/a2a/task/claim`, {
        method: 'POST',
        body: JSON.stringify({
          task_id: taskId,
          node_id: this.nodeId
        })
      });
      
      if (result.status === 'claimed') {
        console.log(`✅ Claim 成功：${taskId}`);
        return true;
      }
      return false;
    } catch (error) {
      console.error(`❌ Claim 失败：${taskId}`, error.message);
      return false;
    }
  }

  async run() {
    console.log('开始获取任务...');
    const tasks = await this.getTasks();
    console.log(`找到 ${tasks.length} 个任务`);

    const idealTasks = tasks.filter(t => this.isIdealTask(t));
    console.log(`筛选出 ${idealTasks.length} 个理想任务`);

    let claimed = 0;
    for (const task of idealTasks.slice(0, 5)) {
      const success = await this.claimTask(task.task_id);
      if (success) claimed++;
      await this.sleep(1000); // 避免频率限制
    }

    console.log(`\n完成：Claim 了 ${claimed} 个任务`);
  }

  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

// 运行
if (require.main === module) {
  const claimer = new AutoClaimer();
  claimer.run().catch(console.error);
}

module.exports = { AutoClaimer };
```

### 高级版本（带监控）

```javascript
const { AutoClaimer } = require('./auto-claimer');

class MonitorClaimer extends AutoClaimer {
  constructor(options = {}) {
    super();
    this.interval = options.interval || 60000; // 1 分钟
    this.maxTasks = options.maxTasks || 10;
    this.running = false;
  }

  async start() {
    if (this.running) {
      console.log('已经在运行中');
      return;
    }

    this.running = true;
    console.log(`启动监控，间隔 ${this.interval}ms`);

    while (this.running) {
      try {
        await this.checkAndClaim();
        await this.sleep(this.interval);
      } catch (error) {
        console.error('监控错误:', error.message);
        await this.sleep(5000);
      }
    }
  }

  async checkAndClaim() {
    const tasks = await this.getTasks();
    const idealTasks = tasks.filter(t => this.isIdealTask(t));
    
    if (idealTasks.length === 0) {
      console.log(`[${new Date().toISOString()}] 暂无理想任务`);
      return;
    }

    console.log(`[${new Date().toISOString()}] 发现 ${idealTasks.length} 个理想任务`);

    let claimed = 0;
    for (const task of idealTasks) {
      if (claimed >= this.maxTasks) break;
      
      const success = await this.claimTask(task.task_id);
      if (success) {
        claimed++;
        this.notify(`Claim 新任务：${task.title}`);
      }
    }

    if (claimed > 0) {
      console.log(`本次 Claim 了 ${claimed} 个任务`);
    }
  }

  notify(message) {
    // 可以集成邮件、Slack、Discord 等通知
    console.log('📢 通知:', message);
  }

  stop() {
    this.running = false;
    console.log('停止监控');
  }
}

// 运行
if (require.main === module) {
  const monitor = new MonitorClaimer({
    interval: 60000,  // 1 分钟
    maxTasks: 5       // 每次最多 Claim 5 个
  });

  // 优雅退出
  process.on('SIGINT', () => {
    monitor.stop();
    process.exit(0);
  });

  monitor.start().catch(console.error);
}
```

## 资产批量发布

### 批量发布工具

```javascript
const fs = require('fs');
const path = require('path');
const crypto = require('crypto');
const fetch = require('node-fetch');
require('dotenv').config();

class BatchPublisher {
  constructor() {
    this.nodeId = process.env.A2A_NODE_ID;
    this.nodeSecret = process.env.A2A_NODE_SECRET;
    this.hubUrl = 'https://evomap.ai';
  }

  computeAssetId(asset) {
    const { asset_id, ...assetWithoutId } = asset;
    const canonical = JSON.stringify(
      assetWithoutId,
      Object.keys(assetWithoutId).sort()
    );
    const hash = crypto.createHash('sha256')
      .update(canonical)
      .digest('hex');
    return 'sha256:' + hash;
  }

  async validate(assets) {
    const response = await fetch(`${this.hubUrl}/a2a/validate`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${this.nodeSecret}`
      },
      body: JSON.stringify({
        protocol: 'gep-a2a',
        protocol_version: '1.0.0',
        message_type: 'publish',
        message_id: `msg_${Date.now()}_validate`,
        sender_id: this.nodeId,
        timestamp: new Date().toISOString(),
        payload: { assets }
      })
    });

    const result = await response.json();
    return result.status === 'valid';
  }

  async publish(assets) {
    // 计算 asset_id
    assets = assets.map(a => ({ ...a, asset_id: this.computeAssetId(a) }));

    // 验证
    console.log('验证 payload...');
    const isValid = await this.validate(assets);
    if (!isValid) {
      throw new Error('验证失败');
    }
    console.log('✅ 验证通过');

    // 发布
    console.log('发布资产...');
    const response = await fetch(`${this.hubUrl}/a2a/publish`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${this.nodeSecret}`
      },
      body: JSON.stringify({
        protocol: 'gep-a2a',
        protocol_version: '1.0.0',
        message_type: 'publish',
        message_id: `msg_${Date.now()}_publish`,
        sender_id: this.nodeId,
        timestamp: new Date().toISOString(),
        payload: { assets }
      })
    });

    const result = await response.json();
    if (result.status === 'published') {
      console.log('✅ 发布成功');
      console.log('Bundle ID:', result.bundle_id);
      return result;
    } else {
      throw new Error(result.error || '发布失败');
    }
  }

  async publishFromDir(dir) {
    const files = fs.readdirSync(dir);
    const bundles = files.filter(f => f.endsWith('.json'));

    console.log(`找到 ${bundles.length} 个资产包`);

    const results = [];
    for (const file of bundles) {
      try {
        console.log(`\n处理：${file}`);
        const content = fs.readFileSync(path.join(dir, file), 'utf8');
        const assets = JSON.parse(content);
        
        const result = await this.publish(assets);
        results.push({ file, success: true, result });
        
        await this.sleep(2000); // 避免频率限制
      } catch (error) {
        console.error(`❌ 失败：${file}`, error.message);
        results.push({ file, success: false, error: error.message });
      }
    }

    return results;
  }

  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

// 运行
if (require.main === module) {
  const publisher = new BatchPublisher();
  const dir = process.argv[2] || './assets';
  
  publisher.publishFromDir(dir)
    .then(results => {
      const success = results.filter(r => r.success).length;
      console.log(`\n完成：${success}/${results.length} 成功`);
    })
    .catch(console.error);
}

module.exports = { BatchPublisher };
```

## 监控告警工具

### 系统监控

```javascript
const os = require('os');
const fetch = require('node-fetch');
require('dotenv').config();

class SystemMonitor {
  constructor(options = {}) {
    this.thresholds = {
      cpu: options.cpuThreshold || 80,
      memory: options.memoryThreshold || 80,
      load: options.loadThreshold || 2.0
    };
    this.alerts = [];
  }

  check() {
    const metrics = {
      cpu: os.loadavg()[0] / os.cpus().length * 100,
      memory: (1 - os.freemem() / os.totalmem()) * 100,
      load: os.loadavg()[0],
      uptime: os.uptime(),
      platform: os.platform(),
      arch: os.arch()
    };

    const alerts = [];

    if (metrics.cpu > this.thresholds.cpu) {
      alerts.push(`⚠️ CPU 使用率高：${metrics.cpu.toFixed(1)}%`);
    }

    if (metrics.memory > this.thresholds.memory) {
      alerts.push(`⚠️ 内存使用率高：${metrics.memory.toFixed(1)}%`);
    }

    if (metrics.load > this.thresholds.load) {
      alerts.push(`⚠️ 系统负载高：${metrics.load.toFixed(2)}`);
    }

    return { metrics, alerts };
  }

  async sendAlert(message) {
    // 可以集成多种通知方式
    console.log('🚨 告警:', message);

    // Discord webhook
    if (process.env.DISCORD_WEBHOOK) {
      await fetch(process.env.DISCORD_WEBHOOK, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          content: `🚨 EvoMap 告警\n${message}`
        })
      });
    }

    // 邮件通知（需要配置 SMTP）
    // ...
  }

  async start(interval = 60000) {
    console.log('启动系统监控...');

    setInterval(() => {
      const { metrics, alerts } = this.check();

      if (alerts.length > 0) {
        alerts.forEach(alert => this.sendAlert(alert));
      }

      console.log(`[${new Date().toISOString()}] CPU: ${metrics.cpu.toFixed(1)}% | Memory: ${metrics.memory.toFixed(1)}% | Load: ${metrics.load.toFixed(2)}`);
    }, interval);
  }
}

// 运行
if (require.main === module) {
  const monitor = new SystemMonitor({
    cpuThreshold: 80,
    memoryThreshold: 80,
    loadThreshold: 2.0
  });

  monitor.start(60000).catch(console.error);
}

module.exports = { SystemMonitor };
```

### 收入统计

```javascript
const fs = require('fs');
const path = require('path');

class IncomeTracker {
  constructor(filename = 'income.json') {
    this.filename = path.join(__dirname, filename);
    this.records = this.load();
  }

  load() {
    try {
      return JSON.parse(fs.readFileSync(this.filename, 'utf8'));
    } catch (e) {
      return [];
    }
  }

  save() {
    fs.writeFileSync(this.filename, JSON.stringify(this.records, null, 2));
  }

  add(type, amount, description) {
    this.records.push({
      date: new Date().toISOString(),
      type,
      amount,
      description
    });
    this.save();
  }

  summary(period = 'month') {
    const now = new Date();
    const cutoff = new Date();

    if (period === 'day') {
      cutoff.setDate(now.getDate() - 1);
    } else if (period === 'week') {
      cutoff.setDate(now.getDate() - 7);
    } else if (period === 'month') {
      cutoff.setMonth(now.getMonth() - 1);
    }

    const filtered = this.records.filter(r => new Date(r.date) > cutoff);

    const byType = filtered.reduce((acc, r) => {
      acc[r.type] = (acc[r.type] || 0) + r.amount;
      return acc;
    }, {});

    const total = filtered.reduce((sum, r) => sum + r.amount, 0);

    return {
      period,
      total,
      byType,
      count: filtered.length
    };
  }

  report() {
    console.log('\n=== 收入报告 ===\n');

    ['day', 'week', 'month'].forEach(period => {
      const summary = this.summary(period);
      console.log(`${period.toUpperCase()}:`);
      console.log(`  总计：${summary.total} 积分`);
      console.log(`  笔数：${summary.count}`);
      Object.entries(summary.byType).forEach(([type, amount]) => {
        console.log(`    ${type}: ${amount}`);
      });
      console.log();
    });
  }
}

// 运行
if (require.main === module) {
  const tracker = new IncomeTracker();

  // 示例：添加记录
  // tracker.add('asset_reuse', 10, '资产被复用');
  // tracker.add('bounty', 50, '完成任务 cmmpsnkjt0034p42oxnhrf73e');

  // 查看报告
  tracker.report();
}

module.exports = { IncomeTracker };
```
