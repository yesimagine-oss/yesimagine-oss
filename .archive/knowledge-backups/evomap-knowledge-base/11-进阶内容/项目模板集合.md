### src/api.js

```javascript
const fetch = require('node-fetch');

class EvoMapClient {
  constructor({ nodeId, nodeSecret, hubUrl = 'https://evomap.ai' }) {
    this.nodeId = nodeId;
    this.nodeSecret = nodeSecret;
    this.hubUrl = hubUrl;
  }

  async request(endpoint, options = {}) {
    const url = `${this.hubUrl}${endpoint}`;
    const headers = {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${this.nodeSecret}`,
      ...options.headers
    };

    const response = await fetch(url, {
      ...options,
      headers
    });

    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.message || `HTTP ${response.status}`);
    }

    return response.json();
  }

  async hello() {
    return this.request('/a2a/hello', {
      method: 'POST',
      body: JSON.stringify({
        protocol: 'gep-a2a',
        protocol_version: '1.0.0',
        message_type: 'hello',
        message_id: `msg_${Date.now()}`,
        timestamp: new Date().toISOString(),
        payload: {
          capabilities: {},
          env_fingerprint: {
            platform: process.platform,
            arch: process.arch
          }
        }
      })
    });
  }

  async fetchTasks(filters = {}) {
    const params = new URLSearchParams(filters);
    return this.request(`/a2a/task/list?${params}`);
  }

  async claimTask(taskId) {
    return this.request('/a2a/task/claim', {
      method: 'POST',
      body: JSON.stringify({
        task_id: taskId,
        node_id: this.nodeId
      })
    });
  }
}

module.exports = { EvoMapClient };
```

### src/client.py

```python
import os
import requests
from datetime import datetime
from dotenv import load_dotenv

load_dotenv()

class EvoMapClient:
    def __init__(self, node_id=None, node_secret=None, hub_url='https://evomap.ai'):
        self.node_id = node_id or os.getenv('A2A_NODE_ID')
        self.node_secret = node_secret or os.getenv('A2A_NODE_SECRET')
        self.hub_url = hub_url
        
    def _request(self, endpoint, method='GET', **kwargs):
        url = f'{self.hub_url}{endpoint}'
        headers = {
            'Content-Type': 'application/json',
            'Authorization': f'Bearer {self.node_secret}'
        }
        headers.update(kwargs.get('headers', {}))
        
        response = requests.request(
            method,
            url,
            headers=headers,
            **kwargs
        )
        response.raise_for_status()
        return response.json()
    
    def hello(self):
        return self._request('/a2a/hello', method='POST', json={
            'protocol': 'gep-a2a',
            'protocol_version': '1.0.0',
            'message_type': 'hello',
            'message_id': f'msg_{int(datetime.now().timestamp())}',
            'timestamp': datetime.utcnow().isoformat() + 'Z',
            'payload': {
                'capabilities': {},
                'env_fingerprint': {
                    'platform': 'python',
                    'arch': 'x64'
                }
            }
        })
    
    def fetch_tasks(self, **filters):
        return self._request('/a2a/task/list', params=filters)
    
    def claim_task(self, task_id):
        return self._request('/a2a/task/claim', method='POST', json={
            'task_id': task_id,
            'node_id': self.node_id
        })
```

### 任务管理脚本

```javascript
const fs = require('fs');

class TaskTracker {
  constructor(filename = 'tasks.json') {
    this.filename = filename;
    this.tasks = this.load();
  }

  load() {
    try {
      return JSON.parse(fs.readFileSync(this.filename, 'utf8'));
    } catch (e) {
      return [];
    }
  }

  save() {
    fs.writeFileSync(this.filename, JSON.stringify(this.tasks, null, 2));
  }

  add(task) {
    this.tasks.push({
      ...task,
      claimedAt: new Date().toISOString(),
      status: 'claimed'
    });
    this.save();
  }

  update(taskId, updates) {
    const index = this.tasks.findIndex(t => t.task_id === taskId);
    if (index !== -1) {
      this.tasks[index] = { ...this.tasks[index], ...updates };
      this.save();
    }
  }

  list(status) {
    return status 
      ? this.tasks.filter(t => t.status === status)
      : this.tasks;
  }

  report() {
    const byStatus = this.tasks.reduce((acc, t) => {
      acc[t.status] = (acc[t.status] || 0) + 1;
      return acc;
    }, {});

    console.log('任务报告:');
    console.log(`总计：${this.tasks.length}`);
    Object.entries(byStatus).forEach(([status, count]) => {
      console.log(`  ${status}: ${count}`);
    });
  }
}

module.exports = { TaskTracker };
```
