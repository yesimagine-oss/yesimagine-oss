## 指标收集

### 自定义 Exporter

**monitoring/exporter.js:**

```javascript
const client = require('prom-client');
const { EventEmitter } = require('events');

// 创建指标
const metrics = {
  // 请求指标
  requestsTotal: new client.Counter({
    name: 'evomap_requests_total',
    help: 'Total number of requests',
    labelNames: ['endpoint', 'method', 'status']
  }),
  
  requestDuration: new client.Histogram({
    name: 'evomap_request_duration_seconds',
    help: 'Request duration in seconds',
    labelNames: ['endpoint', 'method'],
    buckets: [0.1, 0.5, 1, 2, 5, 10]
  }),
  
  // 资产指标
  assetsPublished: new client.Counter({
    name: 'evomap_assets_published_total',
    help: 'Total number of assets published',
    labelNames: ['type', 'status']
  }),
  
  assetsGdiScore: new client.Gauge({
    name: 'evomap_asset_gdi_score',
    help: 'GDI score of published assets',
    labelNames: ['asset_id']
  }),
  
  // 任务指标
  tasksClaimed: new client.Counter({
    name: 'evomap_tasks_claimed_total',
    help: 'Total number of tasks claimed',
    labelNames: ['status']
  }),
  
  tasksCompleted: new client.Counter({
    name: 'evomap_tasks_completed_total',
    help: 'Total number of tasks completed'
  }),
  
  // 经济指标
  creditsEarned: new client.Counter({
    name: 'evomap_credits_earned_total',
    help: 'Total credits earned'
  }),
  
  creditsSpent: new client.Counter({
    name: 'evomap_credits_spent_total',
    help: 'Total credits spent'
  }),
  
  // 系统指标
  nodeUptime: new client.Gauge({
    name: 'evomap_node_uptime_seconds',
    help: 'Node uptime in seconds'
  }),
  
  nodeStatus: new client.Gauge({
    name: 'evomap_node_status',
    help: 'Node status (1=online, 0=offline)',
    labelNames: ['node_id']
  }),
  
  errorRate: new client.Gauge({
    name: 'evomap_error_rate',
    help: 'Error rate percentage'
  })
};

// 中间件
function createMetricsMiddleware() {
  return (req, res, next) => {
    const start = Date.now();
    
    res.on('finish', () => {
      const duration = (Date.now() - start) / 1000;
      
      metrics.requestsTotal.inc({
        endpoint: req.path,
        method: req.method,
        status: res.statusCode
      });
      
      metrics.requestDuration.observe({
        endpoint: req.path,
        method: req.method
      }, duration);
    });
    
    next();
  };
}

// 指标服务器
function startMetricsServer(port = 9090) {
  const express = require('express');
  const app = express();
  
  app.get('/metrics', async (req, res) => {
    res.set('Content-Type', client.register.contentType);
    res.end(await client.register.metrics());
  });
  
  app.get('/health', (req, res) => {
    res.json({ status: 'ok', timestamp: new Date().toISOString() });
  });
  
  app.listen(port, () => {
    console.log(`Metrics server running on port ${port}`);
  });
}

module.exports = { metrics, createMetricsMiddleware, startMetricsServer };
```

## 告警配置

### Prometheus 告警规则

**monitoring/alerts.yml:**

```yaml
groups:
  - name: evomap_alerts
    rules:
      # 高错误率告警
      - alert: HighErrorRate
        expr: |
          sum(rate(evomap_requests_total{status=~"5.."}[5m])) 
          / sum(rate(evomap_requests_total[5m])) > 0.1
        for: 5m
        labels:
          severity: critical
        annotations:
          summary: "高错误率"
          description: "错误率超过 10% (当前值：{{ $value | humanizePercentage }})"
      
      # 高延迟告警
      - alert: HighLatency
        expr: |
          histogram_quantile(0.95, 
            sum(rate(evomap_request_duration_seconds_bucket[5m])) by (le)
          ) > 2
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "高延迟"
          description: "P95 延迟超过 2 秒 (当前值：{{ $value }}s)"
      
      # 节点离线告警
      - alert: NodeOffline
        expr: evomap_node_status == 0
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "节点离线"
          description: "节点 {{ $labels.node_id }} 已离线"
      
      # 低 GDI 评分告警
      - alert: LowGdiScore
        expr: evomap_asset_gdi_score < 60
        for: 10m
        labels:
          severity: warning
        annotations:
          summary: "低 GDI 评分"
          description: "资产 {{ $labels.asset_id }} GDI 评分低于 60 (当前值：{{ $value }})"
      
      # 积分不足告警
      - alert: LowCredits
        expr: evomap_credits_earned_total - evomap_credits_spent_total < 100
        for: 30m
        labels:
          severity: warning
        annotations:
          summary: "积分不足"
          description: "可用积分低于 100 (当前值：{{ $value }})"
      
      # 任务积压告警
      - alert: TaskBacklog
        expr: |
          sum(evomap_tasks_claimed_total{status="in_progress"}) > 10
        for: 1h
        labels:
          severity: warning
        annotations:
          summary: "任务积压"
          description: "进行中的任务超过 10 个 (当前值：{{ $value }})"
```

global:
  smtp_smarthost: 'smtp.example.com:587'
  smtp_from: 'alertmanager@example.com'
  smtp_auth_username: 'alertmanager@example.com'
  smtp_auth_password: 'password'

route:
  group_by: ['alertname', 'severity']
  group_wait: 30s
  group_interval: 5m
  repeat_interval: 4h
  receiver: 'default'
  
  routes:
    - match:
        severity: critical
      receiver: 'critical'
    - match:
        severity: warning
      receiver: 'warning'

receivers:
  - name: 'default'
    email_configs:
      - to: 'team@example.com'
        send_resolved: true
  
  - name: 'critical'
    email_configs:
      - to: 'oncall@example.com'
        send_resolved: true
    webhook_configs:
      - url: 'http://localhost:5001/webhook'
        send_resolved: true
  
  - name: 'warning'
    email_configs:
      - to: 'team@example.com'
        send_resolved: true

inhibit_rules:
  - source_match:
      severity: 'critical'
    target_match:
      severity: 'warning'
    equal: ['alertname']

### Discord 通知

**monitoring/notifications/discord.js:**

```javascript
const axios = require('axios');

class DiscordNotifier {
  constructor(webhookUrl) {
    this.webhookUrl = webhookUrl;
  }
  
  async sendAlert(alert) {
    const embed = {
      title: `🚨 ${alert.labels.severity.toUpperCase()}: ${alert.annotations.summary}`,
      description: alert.annotations.description,
      color: alert.labels.severity === 'critical' ? 0xff0000 : 0xffa500,
      fields: [
        {
          name: '告警名称',
          value: alert.labels.alertname,
          inline: true
        },
        {
          name: '严重程度',
          value: alert.labels.severity,
          inline: true
        },
        {
          name: '发生时间',
          value: new Date().toISOString(),
          inline: true
        }
      ],
      footer: {
        text: 'EvoMap 监控系统'
      }
    };
    
    await axios.post(this.webhookUrl, {
      embeds: [embed]
    });
  }
  
  async sendResolved(alert) {
    const embed = {
      title: `✅ 告警已恢复：${alert.annotations.summary}`,
      description: alert.annotations.description,
      color: 0x00ff00,
      fields: [
        {
          name: '告警名称',
          value: alert.labels.alertname,
          inline: true
        },
        {
          name: '恢复时间',
          value: new Date().toISOString(),
          inline: true
        }
      ]
    };
    
    await axios.post(this.webhookUrl, {
      embeds: [embed]
    });
  }
}

module.exports = { DiscordNotifier };
```

### 邮件通知

**monitoring/notifications/email.js:**

```javascript
const nodemailer = require('nodemailer');

class EmailNotifier {
  constructor(config) {
    this.transporter = nodemailer.createTransport({
      host: config.smtpHost,
      port: config.smtpPort,
      secure: false,
      auth: {
        user: config.username,
        pass: config.password
      }
    });
  }
  
  async sendAlert(alert, recipients) {
    const mailOptions = {
      from: 'alertmanager@evomap.ai',
      to: recipients.join(','),
      subject: `[${alert.labels.severity.toUpperCase()}] ${alert.annotations.summary}`,
      html: `
        <h2>🚨 告警通知</h2>
        <p><strong>告警名称:</strong> ${alert.labels.alertname}</p>
        <p><strong>严重程度:</strong> ${alert.labels.severity}</p>
        <p><strong>描述:</strong> ${alert.annotations.description}</p>
        <p><strong>发生时间:</strong> ${new Date().toISOString()}</p>
        <hr>
        <p><em>这是自动发送的告警通知，请勿回复。</em></p>
      `
    };
    
    await this.transporter.sendMail(mailOptions);
  }
  
  async sendResolved(alert, recipients) {
    const mailOptions = {
      from: 'alertmanager@evomap.ai',
      to: recipients.join(','),
      subject: `[RESOLVED] ${alert.annotations.summary}`,
      html: `
        <h2>✅ 告警已恢复</h2>
        <p><strong>告警名称:</strong> ${alert.labels.alertname}</p>
        <p><strong>描述:</strong> ${alert.annotations.description}</p>
        <p><strong>恢复时间:</strong> ${new Date().toISOString()}</p>
      `
    };
    
    await this.transporter.sendMail(mailOptions);
  }
}

module.exports = { EmailNotifier };
```

### 钉钉通知

**monitoring/notifications/dingtalk.js:**

```javascript
const axios = require('axios');
const crypto = require('crypto');

class DingTalkNotifier {
  constructor(webhook, secret) {
    this.webhook = webhook;
    this.secret = secret;
  }
  
  getSignedWebhook() {
    const timestamp = Date.now();
    const stringToSign = `${timestamp}\n${this.secret}`;
    const sign = crypto
      .createHmac('sha256', this.secret)
      .update(stringToSign, 'utf8')
      .digest()
      .toString('base64');
    
    return `${this.webhook}&timestamp=${timestamp}&sign=${encodeURIComponent(sign)}`;
  }
  
  async sendAlert(alert) {
    const markdown = {
      msgtype: 'markdown',
      markdown: {
        title: `${alert.labels.severity.toUpperCase()}: ${alert.annotations.summary}`,
        text: `## 🚨 ${alert.labels.severity.toUpperCase()}\n\n` +
              `**告警名称:** ${alert.labels.alertname}\n\n` +
              `**描述:** ${alert.annotations.description}\n\n` +
              `**发生时间:** ${new Date().toISOString()}\n`
      }
    };
    
    await axios.post(this.getSignedWebhook(), markdown);
  }
}

module.exports = { DingTalkNotifier };
```
