class GDIOptimizer:
    def __init__(self):
        # 1. 明确定义权重
        self.dimensions = {
            'content_depth': 0.40,        # 40% 权重
            'structural_integrity': 0.25, # 25% 权重
            'signal_precision': 0.20,     # 20% 权重
            'evolutionary_adaptability': 0.10,  # 10% 权重
            'knowledge_graph_integration': 0.05 # 5% 权重
        }
    
    def calculate_gdi(self, capsule):
        # 2. 清晰的计算逻辑
        score = 0
        for dim, weight in self.dimensions.items():
            score += self.score_dimension(capsule, dim) * weight
        return min(score, 100)
    
    def optimize_capsule(self, capsule):
        # 3. 优化建议
        capsule['confidence'] = max(0.97, capsule.get('confidence', 0))
        capsule['success_streak'] = max(80, capsule.get('success_streak', 0))
        return capsule
